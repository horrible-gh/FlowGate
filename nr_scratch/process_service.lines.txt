65	def _build_bug_template_body(locale: str = "en") -> str:
66	    """Build the default B template body with locale-aware section headings."""
67	    sections = (
68	        _BUG_TEMPLATE_SECTIONS.get(locale)
69	        or _BUG_TEMPLATE_SECTIONS.get((locale or "").split("-")[0])
70	        or _BUG_TEMPLATE_SECTIONS["en"]
71	    )
72	    lines: list[str] = []
73	    for idx, section in enumerate(sections):
74	        lines.append(f"## {section}")
75	        if idx < len(sections) - 1:
76	            lines.append("")
77	    return "\n".join(lines)

228	    """Return docs_root and project_root for the project. Return empty strings when unset."""
229	    env_project_root = os.environ.get(f"FLOWGATE_PROJECT_ROOT_OVERRIDE_{project}", "")
230	    row = db.get_project_settings_by_project(project)
231	    if row:

236	    return "", str(env_project_root or "")

239	def _find_in_allowed_storage_buckets(basename: str) -> str:
240	    """Search allowed buckets under storage and return the absolute path (T051 fallback).
241	
242	    Allowed buckets: inbox, accept, outbox, reject, _rule.
243	    Because transferred files gain a prefix in their name, check not only exact
244	    matches but also "_basename" suffix matches (the post-transfer filename pattern).
245	    Return an empty string when the file cannot be found.
246	    """
247	    if not basename:
248	        return ""
249	    allowed_dirs = [
250	        db.INBOX_DIR,
251	        db.ACCEPT_DIR,
252	        db.OUTBOX_DIR,
253	        db.REJECT_DIR,
254	        os.path.join(db.STORAGE_DIR, "_rule"),
255	    ]
256	    suffix = "_" + basename
257	    for bucket_dir in allowed_dirs:
258	        for dirpath, _dirs, filenames in os.walk(bucket_dir):
259	            for f in filenames:
260	                if f == basename or f.endswith(suffix):
261	                    return os.path.join(dirpath, f)
262	    return ""

265	def _to_docs_root_relative(filename: str, docs_root: str) -> str:
266	    """Convert a filename to a docs_root-relative path (T051/T052).
267	
268	    1. When docs_root is set, search under docs_root and return a relative path
269	       that starts with an allowed prefix (_rule/accept/inbox/outbox/reject).
270	    2. If no location starting with an allowed prefix is found, search the
271	       allowed storage buckets instead (including suffix matching).
272	       - If the fallback absolute path is under docs_root and is an allowed
273	         prefix path, return it as a relative path (T052).
274	       - If it is not under docs_root, return the absolute path as-is.
275	    3. If it cannot be found anywhere, return an empty string. Never return a
276	       bare filename.
277	    """
278	    filename = (filename or "").strip()
279	    if not filename:
280	        return ""
281	    basename = os.path.basename(filename) or filename
282	    docs_root = (docs_root or "").strip()
283	    if docs_root:
284	        for dirpath, _dirs, filenames in os.walk(docs_root):
285	            if basename in filenames:
286	                found_abs = os.path.join(dirpath, basename)
287	                try:
288	                    rel = os.path.relpath(found_abs, docs_root)
289	                except ValueError:
290	                    continue
291	                first_component = rel.split(os.sep)[0]
292	                if first_component in _ALLOWED_PATH_PREFIXES:
293	                    return rel
294	                # Not an allowed prefix → keep searching
295	    # docs_root unset, or file not found in an allowed-prefix location
296	    # → fall back to allowed storage buckets
297	    fallback_abs = _find_in_allowed_storage_buckets(basename)
298	    # T052: if the fallback absolute path is under docs_root, validate the
299	    # allowed prefix and return a relative path
300	    if fallback_abs and docs_root:
301	        try:
302	            rel = os.path.relpath(fallback_abs, docs_root)
303	        except ValueError:
304	            return fallback_abs
305	        if not rel.startswith(".."):
306	            first_component = rel.split(os.sep)[0]
307	            if first_component in _ALLOWED_PATH_PREFIXES:
308	                return rel
309	    return fallback_abs

345	def _resolve_requirement_context(source_doc: dict[str, Any] | None, docs_root: str = "") -> dict[str, Any]:
346	    """Follow the source document's target chain to find the requirement context."""
347	    source_doc = source_doc or {}
348	    current_target_id = str(source_doc.get("target_id") or "").strip()
349	    if not current_target_id:
350	        return {
351	            "requirement_doc_id": "",
352	            "requirement_title": "",
353	            "requirement_file": "",
354	            "reference_qa": [],
355	        }
356	
357	    visited_doc_ids: set[str] = set()
358	    closest_parent_doc: dict[str, Any] | None = None
359	    requirement_doc: dict[str, Any] | None = None
360	
361	    while current_target_id:
362	        if current_target_id in visited_doc_ids:
363	            return {
364	                "requirement_doc_id": "",
365	                "requirement_title": "",
366	                "requirement_file": "",
367	                "reference_qa": [],
368	            }
369	        visited_doc_ids.add(current_target_id)
370	
371	        parent_doc = db.get_document_by_id(current_target_id)
372	        if parent_doc is None:
373	            return {
374	                "requirement_doc_id": "",
375	                "requirement_title": "",
376	                "requirement_file": "",
377	                "reference_qa": [],
378	            }
379	
380	        if closest_parent_doc is None:
381	            closest_parent_doc = parent_doc
382	
383	        if str(parent_doc.get("type") or "").strip() == "R":
384	            requirement_doc = parent_doc
385	            break
386	
387	        current_target_id = str(parent_doc.get("target_id") or "").strip()
388	
389	    requirement_doc = requirement_doc or closest_parent_doc
390	    requirement_doc_id = str((requirement_doc or {}).get("doc_id") or "")
391	    requirement_title = str((requirement_doc or {}).get("title") or "")
392	    raw_req_file = db.get_created_memo_file(requirement_doc_id) or "" if requirement_doc_id else ""
393	    requirement_file = _to_docs_root_relative(raw_req_file, docs_root)
394	
395	    return {
396	        "requirement_doc_id": requirement_doc_id,
397	        "requirement_title": requirement_title,
398	        "requirement_file": requirement_file,
399	        "reference_qa": _build_reference_qa_items(requirement_doc_id, docs_root),
400	    }

403	def _build_next_action_context_fields(source_doc: dict[str, Any] | None, docs_root: str = "") -> dict[str, Any]:
404	    """Build the source/requirement context shared by next-action candidates."""
405	    source_doc = source_doc or {}
406	    source_doc_id = str(source_doc.get("doc_id") or "")
407	    source_title = str(source_doc.get("title") or "")
408	    raw_source_file = db.get_created_memo_file(source_doc_id) or "" if source_doc_id else ""
409	    source_file = _to_docs_root_relative(raw_source_file, docs_root)
410	
411	    context = {
412	        "source_doc_id": source_doc_id,
413	        "source_title": source_title,
414	        "source_file": source_file,
415	        "requirement_doc_id": "",
416	        "requirement_title": "",
417	        "requirement_file": "",
418	        "reference_qa": [],
419	    }
420	    if not source_doc_id:
421	        return context
422	
423	    context.update(_resolve_requirement_context(source_doc, docs_root))
424	    return context

427	def _build_next_action_candidates(
428	    candidate_types: list[str] | tuple[str, ...] | set[str],
429	    source_doc: dict[str, Any] | None,
430	    group_id: str,
431	    project: str,
432	    module: str,
433	    docs_root: str = "",
434	    project_root: str = "",
435	    locale: str = "en",
436	) -> list[dict]:
437	    """Convert the next-action type list into a common payload shape."""
438	    ordered_types: list[str] = []
439	    seen_types: set[str] = set()
440	    for raw_type in candidate_types:
441	        next_type = str(raw_type or "").strip()
442	        if not next_type or next_type in seen_types:
443	            continue
444	        seen_types.add(next_type)
445	        ordered_types.append(next_type)
446	
447	    context_fields = _build_next_action_context_fields(source_doc, docs_root)
448	    return [
449	        {
450	            "type": next_type,
451	            "label": _get_next_label(next_type, locale),
452	            "group_id": group_id,
453	            "project": project,
454	            "module": module,
455	            "docs_root": docs_root,
456	            "project_root": project_root,
457	            **context_fields,
458	        }
459	        for next_type in ordered_types
460	    ]

463	def _find_latest_source_doc_for_stage(
464	    docs: list[dict[str, Any]],
465	    current_action_type: str,
466	) -> dict[str, Any] | None:
467	    """Find the representative document to use as source context for the current action stage."""
468	    preferred_types = GROUP_NEXT_ACTION_SOURCE_TYPES.get(current_action_type, ())
469	    for preferred_type in preferred_types:
470	        for doc in reversed(docs):
471	            if str(doc.get("type") or "").strip() == preferred_type:
472	                return doc
473	
474	    for doc in reversed(docs):
475	        normalized_type = GROUP_NEXT_ACTION_STAGE_BY_TYPE.get(str(doc.get("type") or "").strip())
476	        if normalized_type == current_action_type:
477	            return doc
478	    return None

481	def _resolve_group_next_action_context(
482	    docs: list[dict],
483	    group_id: str,
484	    project: str,
485	    module: str,
486	    locale: str = "en",
487	) -> dict[str, Any]:
488	    """Compute the current-action context and candidate list for the group detail panel."""
489	    current_action_type = ""
490	    for doc in reversed(docs):
491	        normalized_type = GROUP_NEXT_ACTION_STAGE_BY_TYPE.get(str(doc.get("type") or "").strip())
492	        if normalized_type:
493	            current_action_type = normalized_type
494	            break
495	
496	    source_doc = _find_latest_source_doc_for_stage(docs, current_action_type)
497	    next_action_candidates: list[dict] = []
498	    if current_action_type == "DC":
499	        next_action_candidates = _collect_next_stage_candidates(source_doc, group_id, project, module, locale)
500	    elif current_action_type:
501	        existing_types = {
502	            str(doc.get("type") or "").strip()
503	            for doc in docs
504	            if str(doc.get("type") or "").strip()
505	        }
506	        candidate_types = [
507	            next_type
508	            for next_type in GROUP_NEXT_ACTION_FLOW.get(current_action_type, ())
509	            if next_type not in existing_types
510	        ]
511	        docs_root, project_root = _get_project_roots(project)
512	        next_action_candidates = _build_next_action_candidates(
513	            candidate_types,
514	            source_doc,
515	            group_id,
516	            project,
517	            module,
518	            docs_root=docs_root,
519	            project_root=project_root,
520	            locale=locale,
521	        )
522	
523	    selected_next_action_type = ""
524	    preferred_type = str((source_doc or {}).get("next") or "").strip()
525	    candidate_type_set = {candidate["type"] for candidate in next_action_candidates}
526	    if preferred_type in candidate_type_set:
527	        selected_next_action_type = preferred_type
528	
529	    if not selected_next_action_type and next_action_candidates:
530	        selected_next_action_type = str(next_action_candidates[0].get("type") or "")
531	
532	    return {
533	        "current_action_type": current_action_type,
534	        "next_action_candidates": next_action_candidates,
535	        "selected_next_action_type": selected_next_action_type,
536	    }

539	def _collect_next_stage_candidates(
540	    source_doc: dict[str, Any] | None,
541	    group_id: str,
542	    project: str,
543	    module: str,
544	    locale: str = "en",
545	) -> list[dict]:
546	    """Collect the list of next-step candidates for the group.
547	
548	    Returns:
549	        list of {
550	            "type": str,
551	            "label": str,
552	            "group_id": str,
553	            "project": str,
554	            "module": str,
555	            "source_doc_id": str,
556	            "source_title": str,
557	            "source_file": str,
558	            "requirement_doc_id": str,
559	            "requirement_title": str,
560	            "requirement_file": str,
561	            "reference_qa": list[dict],
562	        }
563	        Return [] when there are no candidates.
564	    """
565	    try:
566	        docs = db.get_documents_by_group_id(group_id)
567	        ds_docs = [
568	            d
569	            for d in docs
570	            if str(d.get("type") or "").strip() == "DS" and d.get("status") == "accepted"
571	        ]
572	
573	        candidate_types: set[str] = set()
574	        if ds_docs:
575	            for ds_doc in ds_docs:
576	                next_val = str(ds_doc.get("next") or "").strip()
577	                if not next_val:
578	                    next_val = "N"
579	                candidate_types.add(next_val)
580	        else:
581	            candidate_types = {"N"}
582	
583	        existing_types = {
584	            str(d.get("type") or "").strip()
585	            for d in docs
586	            if str(d.get("type") or "").strip()
587	        }
588	        candidate_types -= existing_types
589	        docs_root, project_root = _get_project_roots(project)
590	        return _build_next_action_candidates(
591	            sorted(candidate_types),
592	            source_doc,
593	            group_id,
594	            project,
595	            module,
596	            docs_root=docs_root,
597	            project_root=project_root,
598	            locale=locale,
599	        )
600	    except Exception:
601	        return []

604	def _parse_template_headings(doc_type: str) -> list[str]:
605	    """Extract the list of ## headings from an NR/TR template file."""
606	    template_name = _REPORT_TEMPLATE_MAP.get(doc_type)
607	    if not template_name:
608	        return []
609	    template_path = ""
610	    for d in _RULE_TEMPLATE_DIRS:
611	        candidate = os.path.join(d, template_name)
612	        if os.path.isfile(candidate):
613	            template_path = candidate
614	            break
615	    if not template_path:
616	        return []
617	    with open(template_path, "r", encoding="utf-8") as f:
618	        content = f.read()
619	    headings: list[str] = []
620	    for line in content.splitlines():
621	        m = re.match(r"^##\s+(.+)$", line)
622	        if m:
623	            headings.append(m.group(1).strip())
624	    return headings

635	def create_workflow_root(
636	    project: str, module: str, title: str,
637	    slug: str, priority: str, body: str,
638	    owner: str = "admin",
639	    group_id: str = "",
640	    new_group_name: str = "",
641	    doc_type: str = "R",
642	    template: str = "default",
643	    locale: str = "en",
644	) -> dict:
645	    """Create an R/B workflow-root document and reserve its group/document ID."""
646	    # Validate input values
647	    errors: list[str] = []
648	    doc_type = (doc_type or "R").strip().upper()
649	    if doc_type not in WORKFLOW_ROOT_TYPES:
650	        errors.append(f"Invalid workflow root type: '{doc_type}'")
651	
652	    title = (title or "").strip()
653	    if not title:
654	        errors.append("title must be between 1 and 100 characters.")
655	    elif len(title) > linter.TITLE_MAX_LEN:
656	        errors.append(f"title must be at most {linter.TITLE_MAX_LEN} characters.")
657	
658	    slug_input = (slug or "").strip()
659	    slug_value = ""
660	    if slug_input:
661	        slug_value = _slugify(slug_input)
662	        if slug_value == "untitled":
663	            errors.append("Please use English letters and numbers for the slug.")
664	
665	    priority = (priority or "medium").strip().lower()
666	    if priority not in linter.VALID_PRIORITIES:
667	        errors.append(f"Invalid priority: '{priority}'")
668	
669	    # Validate project/module
670	    allowed = db.get_allowed_projects()
671	    project_names, project_modules = linter._normalize_allowed_projects(allowed)
672	
673	    if project_names and project not in project_names:
674	        errors.append(f"Unregistered project/module combination: {project}/{module}")
675	    elif project_modules:
676	        if (project, "") not in project_modules and (project, module) not in project_modules:
677	            errors.append(f"Unregistered project/module combination: {project}/{module}")
678	
679	    # Group-related validation
680	    group_id = (group_id or "").strip()
681	    new_group_name = (new_group_name or "").strip()
682	
683	    if group_id and new_group_name:
684	        errors.append("Please specify either an existing group or a new group, not both.")
685	
686	    if errors:
687	        return {"status": "error", "errors": errors}
688	
689	    # Group handling logic
690	    final_group_id = None
691	
692	    if new_group_name:
693	        # Create a new group
694	        try:
695	            group_code = numbering_service.reserve_group(project, module or "none")
696	            final_group_id = f"{project}.{module or 'none'}.{group_code}"
697	            db.insert_group(final_group_id, project, module or "none", new_group_name, priority)
698	        except Exception as e:
699	            return {"status": "error", "errors": [str(e)]}
700	    elif group_id:
701	        # Validate the existing group
702	        existing_group = db.get_group(group_id)
703	        if not existing_group:
704	            errors.append(f"Group does not exist: {group_id}")
705	        elif existing_group.get("project_id") != project:
706	            errors.append(f"The group does not belong to this project: {group_id}")
707	        else:
708	            final_group_id = group_id
709	            # Use the existing group's module as the source of truth.
710	            # It takes precedence over the client-provided module.
711	            group_module = (existing_group.get("module") or "").strip()
712	            if group_module:
713	                module = group_module
714	
715	        if errors:
716	            return {"status": "error", "errors": errors}
717	    else:
718	        # Group unspecified: automatically create a new group
719	        try:
720	            group_code = numbering_service.reserve_group(project, module or "none")
721	            final_group_id = f"{project}.{module or 'none'}.{group_code}"
722	            db.insert_group(final_group_id, project, module or "none", title, priority)
723	        except Exception as e:
724	            return {"status": "error", "errors": [str(e)]}
725	
726	    # A group owns exactly one workflow root, either R or B.
727	    existing_docs = db.get_documents_by_group_id(final_group_id)
728	    for _d in existing_docs:
729	        existing_type = str(_d.get("type_code") or _d.get("type") or "").strip().upper()
730	        if existing_type in WORKFLOW_ROOT_TYPES:
731	            code = (
732	                "group_r_already_exists"
733	                if doc_type == "R" and existing_type == "R"
734	                else "group_root_already_exists"
735	            )
736	            return {
737	                "status": "error",
738	                "errors": [{
739	                    "code": code,
740	                    "message": f"A workflow root ({existing_type}) already exists in this group.",
741	                }],
742	            }
743	
744	    doc_code = numbering_service.reserve_document(
745	        final_group_id, doc_type, module=module or "none"
746	    )
747	    doc_number = doc_code
748	    auto_slug = _slugify(title)
749	    filename_slug = slug_value or auto_slug
750	    filename = f"{doc_code}_{filename_slug}.md"
751	
752	    if doc_type == "B" and template != "none" and not (body or "").strip():
753	        body = _build_bug_template_body(locale)
754	
755	    # Frontmatter is human-readable metadata, so use the short form.
756	    short_group_id = storage_paths.group_dir_name(final_group_id)
757	    project_label = storage_paths.project_dir_name(project)
758	    header_lines = [
759	        "---",
760	        f"group_id: {short_group_id}",
761	        f"type: {doc_type}",
762	        f"doc_number: {doc_number}",
763	        f"project: {project_label}",
764	        f"module: {module}",
765	        f"title: {title}",
766	        f"priority: {priority}",
767	        "target_id:",
768	        "next:",
769	        "---",
770	        "",
771	        body or "",
772	    ]
773	    content = "\n".join(header_lines)
774	
775	    # Write the file to OutBox (AI worker communication channel — planned to switch
776	    # to an API in M016)
777	    outbox = db.outbox_dir(project)
778	    os.makedirs(outbox, exist_ok=True)
779	    filepath = os.path.join(outbox, filename)
780	    with open(filepath, "w", encoding="utf-8") as f:
781	        f.write(content)
782	
783	    # Write the body file into the storage documents tree
784	    # (the actual document base users see in Explorer)
785	    storage_filepath = storage_paths.document_path(
786	        project_id=project,
787	        group_code=final_group_id,
788	        doc_code=doc_number,
789	        filename=f"{filename_slug}.md",
790	        module=module or "none",
791	    )
792	    try:
793	        storage_filepath.parent.mkdir(parents=True, exist_ok=True)
794	        storage_filepath.write_text(content, encoding="utf-8")
795	    except Exception:
796	        if os.path.exists(filepath):
797	            os.remove(filepath)
798	        raise
799	
800	    # Register in the DB
801	    doc_id = f"{final_group_id}.{doc_code}"
802	    try:
803	        db.insert_document(
804	            doc_id=doc_id,
805	            doc_type=doc_type,
806	            project=project,
807	            module=module,
808	            title=title,
809	            group_id=final_group_id,
810	            priority=priority,
811	            status="draft",
812	            direction="outbox",
813	            file_path=str(storage_filepath),
814	        )
815	    except Exception:
816	        if os.path.exists(filepath):
817	            os.remove(filepath)
818	        if storage_filepath.exists():
819	            storage_filepath.unlink()
820	        raise
821	    db.insert_event(doc_id, "created", memo_file=filename)
822	
823	    # groups-table registration was already handled above (for new groups)
824	
825	    return {
826	        "status": "success",
827	        "doc_id": doc_id,
828	        "doc_number": doc_number,
829	        "group_id": final_group_id,
830	        "created_file": os.path.abspath(filepath),
831	        "message": TYPE_ACTION_MESSAGES[doc_type],
832	    }

969	def approve_document(
970	    doc_pk: int,
971	    approved_types: list[str] | None = None,  # DEPRECATED: ignored; kept for flow_gate.py compatibility
972	    confirm_extra: bool = False,
973	) -> dict:
974	    """Approve an InBox document (create AC + transfer files).
975	
976	    For DC type, approve it in bundle mode using approved_files from the DC
977	    header together with DS-based discovery.
978	    """
979	    doc = db.get_document_by_pk(doc_pk)
980	    if doc is None:
981	        return {"status": "error", "message": f"Document not found: doc_id={doc_pk}"}
982	
983	    source_type = doc["type"]
984	    if source_type not in APPROVABLE_TYPES:
985	        return {"status": "error", "message": f"{source_type} is not eligible for approval"}
986	
987	    group_id = doc["group_id"]
988	    project = doc["project"]
989	    module = doc.get("module") or ""
990	    priority = doc.get("priority") or "medium"
991	    source_memo = db.get_created_memo_file(doc["doc_id"])
992	    source_filename = source_memo or ""
993	
994	    migration_results: list[dict] = []
995	
996	    if source_type == "DC":
997	        # 1. Read approved_files from the DC header
998	        source_content = _read_dc_file_content(source_filename)
999	        if source_content is None:
1000	            return {"status": "error", "message": "Unable to read the DC file"}
1001	        header, parse_err = linter.parse_yaml_header(source_content)
1002	        if parse_err or header is None:
1003	            return {"status": "error", "message": f"Failed to parse the DC header: {parse_err}"}
1004	        approved_files = _normalize_approved_files(header.get("approved_files") or [])
1005	        target_id_dc = (header.get("target_id") or "").strip()
1006	
1007	        # 2. Validate approved_files
1008	        validation_issues = _collect_approved_file_validation_issues(
1009	            approved_files, group_id
1010	        )
1011	        if validation_issues:
1012	            return {
1013	                "status": "error",
1014	                "message": "; ".join(issue["message"] for issue in validation_issues),
1015	                "error_file_items": _build_dc_error_file_items(
1016	                    group_id,
1017	                    issues=validation_issues,
1018	                ),
1019	            }
1020	
1021	        # 3. Discover bundle targets
1022	        mismatch_files = _collect_dc_bundle_mismatch_files(approved_files, target_id_dc)
1023	        if mismatch_files:
1024	            return {
1025	                "status": "error",
1026	                "message": "Unable to find bundle target files",
1027	                "error_file_items": _build_dc_error_file_items(
1028	                    group_id,
1029	                    mismatch_files=mismatch_files,
1030	                ),
1031	            }
1032	        bundle_files, extra_files = resolve_bundle_targets(
1033	            approved_files, target_id_dc, group_id
1034	        )
1035	        if not bundle_files:
1036	            return {
1037	                "status": "error",
1038	                "message": "Unable to find bundle target files",
1039	                "error_file_items": _build_dc_error_file_items(
1040	                    group_id,
1041	                    mismatch_files=approved_files,
1042	                ),
1043	            }
1044	
1045	        # 4. C3: extra discovered files — return extra_found when confirm is absent
1046	        if extra_files and not confirm_extra:
1047	            try:
1048	                item_payload = _build_dc_item_payload(group_id, approved_files, extra_files)
1049	            except ValueError as exc:
1050	                return {"status": "error", "message": str(exc)}
1051	            return {
1052	                "status": "extra_found",
1053	                "group_id": group_id,
1054	                "extra_files": extra_files,
1055	                "extra_file_items": item_payload["extra_file_items"],
1056	                "bundle_files": bundle_files,
1057	                "bundle_file_items": item_payload["bundle_file_items"],
1058	                "approved_file_items": item_payload["approved_file_items"],
1059	                "message": "Additional files were found during DS discovery. Please confirm whether to include them in the approval.",
1060	            }
1061	
1062	        # 5. Execute bundle approval
1063	        dc_result = _dc_approve_artifacts(group_id, doc["doc_id"], bundle_files)
1064	        if not dc_result.get("success", True):
1065	            return {"status": "error", "message": dc_result.get("error", "Bundle approval failed")}
1066	        migration_results = dc_result.get("migration", [])
1067	    elif source_type == "VR":
1068	        # Bulk transfer
1069	        targets = _collect_migration_targets(group_id)
1070	        batch_result = _batch_migrate(group_id, targets, db.ACCEPT_DIR, "accepted")
1071	        if not batch_result["success"]:
1072	            return {"status": "error", "message": batch_result.get("error", "Bulk transfer failed")}
1073	        migration_results = batch_result["results"]
1074	    else:
1075	        # Single-file transfer: instructions (DS/N/T) move from inbox, others from processed
1076	        if source_filename:
1077	            source_dir = db.INBOX_DIR if source_type in ("DS", "N", "T") else db.PROCESSED_DIR
1078	            mig = _migrate_single_file(
1079	                group_id, source_filename, source_type, source_dir, db.ACCEPT_DIR,
1080	            )
1081	            if mig:
1082	                migration_results.append(mig)
1083	
1084	    # Update the source document status
1085	    db.update_document_status_by_pk(doc_pk, "accepted")
1086	    db.insert_event(doc["doc_id"], "accepted", memo_file=source_filename)
1087	
1088	    # Create the AC file
1089	    ac_result = _create_ac_file(
1090	        group_id, project, module, priority,
1091	        doc["doc_id"], source_filename, source_type,
1092	    )
1093	
1094	    # Clean up only the OutBox source files corresponding to the current
1095	    # processed document and its direct target document.
1096	    cleaned_outbox_files = _cleanup_related_outbox_files(doc, source_filename)
1097	
1098	    auto_inbox_result: dict[str, str] = {}
1099	    result_type = AUTO_RESULT_DRAFT_TYPES.get(source_type)
1100	    if result_type:
1101	        auto_inbox_result = _create_auto_result_inbox_draft(
1102	            group_id=group_id,
1103	            project=project,
1104	            module=module,
1105	            priority=priority,
1106	            source_doc_id=doc["doc_id"],
1107	            source_type=source_type,
1108	            source_title=doc["title"],
1109	            result_type=result_type,
1110	        )
1111	
1112	    if group_id:
1113	        db.update_group_updated_at(group_id)
1114	
1115	    # next_actions: next-step guidance on DS/DC approval (T040)
1116	    next_actions: list[dict] = []
1117	    if source_type == "DS":
1118	        try:
1119	            next_actions = _collect_next_stage_candidates(doc, group_id, project, module)
1120	        except Exception:
1121	            next_actions = []

1229	def cancel_document(doc_pk: int, cancel_reason: str, followup_action: str) -> dict:
1230	    """Cancel an approved document and clean up files according to the follow-up handling mode.
1231	
1232	    D005 cancellation-allowed types: N / T / TR / NR / D / DS / R / M / P / L
1233	    Behavior when canceling an N-type document:
1234	    - Follow-up handling is fixed to InBox resubmission (resubmit).
1235	    - Automatically cascade-cancel linked NR documents in accepted state (cascade_nr).
1236	    Behavior when canceling a T-type document:
1237	    - Three follow-up handling choices are available (resubmit / discard / user_edit).
1238	    - Automatically cascade-cancel linked TR documents in accepted state
1239	      (cascade_tr, fixed to discard).
1240	    Cancels are allowed only for the PM (user). Automatic worker calls are prohibited.
1241	    """
1242	    cancel_reason = (cancel_reason or "").strip()
1243	    if not cancel_reason:
1244	        return {"status": "error", "errors": ["A cancellation reason is required."]}
1245	
1246	    doc = db.get_document_by_pk(doc_pk)
1247	    if doc is None:
1248	        return {"status": "error", "message": f"Document not found: doc_id={doc_pk}"}
1249	
1250	    current_status = (doc.get("status") or "").strip().lower()
1251	    if current_status != "accepted":
1252	        return {
1253	            "status": "error",
1254	            "message": f"Only approved documents can be cancelled (current status: {doc.get('status')})",
1255	        }
1256	
1257	    doc_type = (doc.get("type") or "").strip().upper()
1258	    group_id = (doc.get("group_id") or "").strip()
1259	
1260	    # T035: only D005-allowed types can be canceled
1261	    if doc_type not in D005_CANCEL_TYPES:
1262	        return {
1263	            "status": "error",
1264	            "message": f"{doc_type} is not eligible for approval cancellation (D005 cancellable types: {', '.join(sorted(D005_CANCEL_TYPES))})",
1265	        }
1266	
1267	    # N type: follow-up handling is fixed to InBox resubmission
1268	    if doc_type == "N":
1269	        action = "resubmit"
1270	    else:
1271	        action = (followup_action or "").strip().lower()
1272	        if action not in CANCEL_FOLLOWUP_ACTION_LABELS:
1273	            return {"status": "error", "errors": ["Please select a valid follow-up action."]}
1274	
1275	    source_memo = db.get_created_memo_file(doc["doc_id"]) or ""
1276	    migration_results: list[dict] = []
1277	
1278	    if action in {"resubmit", "discard"}:
1279	        if not source_memo:
1280	            return {"status": "error", "message": "Original file information could not be found"}
1281	
1282	        accept_file_path = _find_file_in_accept(source_memo)
1283	        if not accept_file_path:
1284	            return {"status": "error", "message": "The original file could not be found in the accept repository"}
1285	
1286	        if action == "resubmit":
1287	            os.makedirs(db.INBOX_DIR, exist_ok=True)
1288	            dest_path = os.path.join(db.INBOX_DIR, source_memo)
1289	            if os.path.exists(dest_path):
1290	                return {"status": "error", "message": f"An identical file already exists in InBox: {source_memo}"}
1291	            _strip_status_header_inplace(accept_file_path)
1292	            shutil.move(accept_file_path, dest_path)
1293	            migration_results.append({
1294	                "original": source_memo,
1295	                "migrated_to": os.path.abspath(dest_path),
1296	                "status": "cancelled",
1297	                "followup_action": action,
1298	            })
1299	        else:
1300	            os.makedirs(db.CANCELLED_DIR, exist_ok=True)
1301	            cancelled_name = _resolve_unique_filename(
1302	                db.CANCELLED_DIR,
1303	                os.path.basename(accept_file_path),
1304	            )
1305	            dest_path = os.path.join(db.CANCELLED_DIR, cancelled_name)
1306	            shutil.move(accept_file_path, dest_path)
1307	            migration_results.append({
1308	                "original": source_memo,
1309	                "migrated_to": os.path.abspath(dest_path),
1310	                "status": "cancelled",
1311	                "followup_action": action,
1312	            })
1313	
1314	    db.update_document_status_by_pk(doc_pk, "cancelled")
1315	    db.insert_event(
1316	        doc["doc_id"],
1317	        "cancelled",
1318	        memo_file=source_memo or None,
1319	        reason=cancel_reason,
1320	        note=action,
1321	    )
1322	
1323	    # On N cancellation: cascade-cancel linked NR documents in accepted state
1324	    # (cascade_nr, D002)
1325	    # On T cancellation: cascade-cancel linked TR documents in accepted state
1326	    # (cascade_tr, fixed to discard, D005)
1327	    cascade_cancelled: list[dict] = []
1328	    if doc_type == "N":
1329	        nr_docs = db.get_documents_by_target_id(
1330	            doc["doc_id"], types=("NR",), statuses=("accepted",)
1331	        )
1332	        for nr_doc in nr_docs:
1333	            db.update_document_status_by_pk(nr_doc["id"], "cancelled")
1334	            db.insert_event(
1335	                nr_doc["doc_id"],
1336	                "cancelled",
1337	                reason=f"N cancellation cascade ({doc['doc_id']})",
1338	                note="cascade_nr",
1339	            )
1340	            cascade_cancelled.append({"doc_id": nr_doc["doc_id"], "type": nr_doc["type"]})
1341	
1342	    elif doc_type == "T":
1343	        # T035: automatically cascade-cancel accepted TR on T cancellation
1344	        # (fixed to discard)
1345	        tr_docs = db.get_documents_by_target_id(
1346	            doc["doc_id"], types=("TR",), statuses=("accepted",)
1347	        )
1348	        for tr_doc in tr_docs:
1349	            # Move the TR file from accept/ → cancelled/ (fixed to discard)
1350	            tr_memo = db.get_created_memo_file(tr_doc["doc_id"])
1351	            if tr_memo:
1352	                tr_accept_path = _find_file_in_accept(tr_memo)
1353	                if tr_accept_path:
1354	                    os.makedirs(db.CANCELLED_DIR, exist_ok=True)
1355	                    cancelled_name = _resolve_unique_filename(
1356	                        db.CANCELLED_DIR, os.path.basename(tr_accept_path)
1357	                    )
1358	                    shutil.move(tr_accept_path, os.path.join(db.CANCELLED_DIR, cancelled_name))
1359	            db.update_document_status_by_pk(tr_doc["id"], "cancelled")
1360	            db.insert_event(
1361	                tr_doc["doc_id"],
1362	                "cancelled",
1363	                reason=f"T cancellation cascade ({doc['doc_id']})",
1364	                note="cascade_tr",
1365	            )
1366	            cascade_cancelled.append({"doc_id": tr_doc["doc_id"], "type": "TR"})
1367	
1368	    if group_id:
1369	        db.update_group_updated_at(group_id)
1370	
1371	    return {
1372	        "status": "success",
1373	        "action": "cancel",
1374	        "group_id": group_id,
1375	        "doc_id": doc["doc_id"],
1376	        "followup_action": action,
1377	        "followup_label": CANCEL_FOLLOWUP_ACTION_LABELS[action],
1378	        "migration": migration_results,
1379	        "cascade_cancelled": cascade_cancelled,
1380	        "message": "Approval cancellation completed.",
1381	    }

1589	    """Read the YAML header from a file path and generate a P-001 §3 handoff message."""

1593	        return {"status": "error", "message": f"File not found: {file_path}"}
1594	
1595	    # T028: normalize based on absolute paths
1596	    # (prevent loss of backslashes or dots)
1597	    real_path = os.path.abspath(real_path)
1598	
1599	    with open(real_path, "r", encoding="utf-8") as f:
1600	        content = f.read()
1601

1603	    if not header:
1604	        return {"status": "error", "message": "Failed to parse YAML header"}
1605

2098	def dispose_group(group_id: str, reason_option: str = "", reason_detail: str = "") -> dict:
2099	    """Discard a group by creating a file-less discard document (AC-style).
2100	
2101	    Mirrors the final-approval AC flow: a documents row with file_path=None records the
2102	    discard as a first-class workflow artifact (header / tab / history), and NO
2103	    out-of-constraint group status is written. The reason is required by the UI and is
2104	    stored on the group_disposed event. Idempotent: one un-reverted discard per group.
2105	    """
2106	    group = db.get_group(group_id)
2107	    if group is None:
2108	        return {"status": "error", "message": f"Group not found: {group_id}"}
2109	
2110	    project = group.get("project_id") or ""
2111	    module = group.get("module") or "none"
2112	
2113	    group_docs = db.get_documents_by_group_id(group_id)
2114	    if any(d.get("type_code") == _GROUP_DISCARD_TYPE for d in group_docs):
2115	        return {"status": "error", "message": f"Group is already disposed: {group_id}"}
2116	
2117	    _ensure_group_discard_type()
2118	
2119	    # Anchor the discard to the group's workflow root (like AC's target_id), when present.
2120	    root_doc = next(
2121	        (d for d in group_docs if d.get("type_code") in WORKFLOW_ROOT_TYPES),
2122	        None,
2123	    )
2124	
2125	    doc_code = numbering_service.reserve_document(group_id, _GROUP_DISCARD_TYPE, module=module)
2126	    doc_id = f"{group_id}.{doc_code}"
2127	
2128	    reason_value, reason_label = _normalize_group_reason_option(reason_option)
2129	    now = datetime.now().strftime("%Y-%m-%d %H:%M")
2130	    note = _build_group_action_note(
2131	        action_label="Group Disposed",
2132	        at=now,
2133	        reason_value=reason_value,
2134	        reason_label=reason_label,
2135	        reason_detail=(reason_detail or "").strip(),
2136	        warning="",
2137	    )
2138	
2139	    # NR0003 §6 (atomicity): previously insert_document committed on its own and the
2140	    # group_disposed event was written afterward. When the event failed (the FK bug
2141	    # below) the DC placeholder document was already committed, and the §2080 idempotency guard then
2142	    # refused to retry — leaving a half-disposed "zombie" group. Wrap the document
2143	    # creation, review-status reset, and the event in ONE transaction (re-entrant: the

3411	def _dc_approve_artifacts(
3412	    group_id: str,
3413	    dc_doc_id: str,
3414	    bundle_files: list[str],
3415	) -> dict:
3416	    """Bulk-approve the artifacts included in bundle_files during DC approval.
3417	
3418	    Roll back everything if transfer fails.
3419	    """
3420	    bundle_set = set(bundle_files)
3421	    docs = db.get_documents_by_group_id(group_id)
3422	    migration_results: list[dict] = []
3423	    rollback_stack: list[dict] = []  # (doc_pk, prev_status, src_path, dst_path)
3424	
3425	    try:
3426	        for d in docs:
3427	            if d["type"] not in _DC_ARTIFACT_TYPES:
3428	                continue
3429	
3430	            memo_file = db.get_created_memo_file(d["doc_id"])
3431	            if not memo_file:
3432	                continue
3433	            if memo_file not in bundle_set:
3434	                continue
3435	
3436	            source_dir = _locate_file(memo_file)
3437	            if not source_dir:
3438	                continue
3439	
3440	            prev_status = d.get("status", "")
3441	            parsed = linter.parse_group_id(group_id)
3442	            if not parsed:
3443	                continue
3444	
3445	            proj = parsed["project"]
3446	            mod = parsed["module"]
3447	            dest_dir = os.path.join(db.ACCEPT_DIR, proj, mod, d["type"])
3448	            os.makedirs(dest_dir, exist_ok=True)
3449	
3450	            seq = _get_migration_type_seq(group_id, d["type"], dest_dir)
3451	            prefixed = _build_migration_filename(group_id, mod, d["type"], seq, memo_file)
3452	            final_name = _resolve_prefix_filename(dest_dir, prefixed)
3453	
3454	            src_path = os.path.join(source_dir, memo_file)
3455	            dst_path = os.path.join(dest_dir, final_name)
3456	            if os.path.exists(src_path):
3457	                shutil.move(src_path, dst_path)
3458	                rollback_stack.append({
3459	                    "doc_pk": d["id"],
3460	                    "prev_status": prev_status,
3461	                    "src": dst_path,
3462	                    "dst": src_path,
3463	                })
3464	
3465	            db.update_document_status_by_pk(d["id"], "accepted")
3466	            db.insert_event(
3467	                d["doc_id"], "accepted", memo_file=memo_file,
3468	                note=f"DC bundle approval (DC: {dc_doc_id})",
3469	            )
3470	
3471	            migration_results.append({
3472	                "original": memo_file,
3473	                "migrated_to": os.path.abspath(dst_path),
3474	                "action": "accepted",
3475	            })
3476	
3477	        return {"success": True, "migration": migration_results}
3478	
3479	    except Exception as e:
3480	        for rb in reversed(rollback_stack):
3481	            try:
3482	                if os.path.exists(rb["src"]):
3483	                    shutil.move(rb["src"], rb["dst"])
3484	                db.update_document_status_by_pk(rb["doc_pk"], rb["prev_status"])
3485	            except Exception:
3486	                pass
3487	        return {"success": False, "error": str(e), "migration": migration_results}

3572	    """Add a suffix when a filename conflict occurs (L-002 §2-4)."""
3573	    if not os.path.exists(os.path.join(dest_dir, prefix_filename)):
3574	        return prefix_filename

3576	    name_part, ext = os.path.splitext(prefix_filename)
3577	    for suffix in range(1, PREFIX_SUFFIX_MAX + 1):
3578	        candidate = f"{name_part}_{suffix}{ext}"
3579	        if not os.path.exists(os.path.join(dest_dir, candidate)):

3771	            content = f.read()

4072	def _compute_tv_aggregate(tv_doc_id: str) -> tuple[str, int, int]:
4073	    """Aggregate scenarios → (tv_status, progress_done, progress_total).
4074	
4075	    Apply the L001 §3 / §4-1 / MEMO_023 Q-08 rules together.
4076	    """
4077	    scenarios = [s for s in db.get_tv_scenarios(tv_doc_id) if not s.get("disabled")]
4078	    total = len(scenarios)
4079	    counts = {"pass": 0, "fail": 0, "skip": 0, "hold": 0, "none": 0}
4080	    for s in scenarios:
4081	        r = (s.get("result") or "").strip().lower()
4082	        if r in counts:
4083	            counts[r] += 1
4084	        else:
4085	            counts["none"] += 1
4086	    done = counts["pass"] + counts["fail"] + counts["skip"] + counts["hold"]
4087	
4088	    if counts["fail"] >= 1:
4089	        return TV_STATUS_FAIL, done, total
4090	    if counts["hold"] >= 1:
4091	        return TV_STATUS_RUNNING, done, total
4092	    if counts["none"] >= 1:
4093	        return TV_STATUS_RUNNING, done, total
4094	    if total >= 1 and counts["pass"] >= 1 and counts["fail"] == 0 and counts["hold"] == 0 and counts["none"] == 0:
4095	        return TV_STATUS_PASS, done, total
4096	    return TV_STATUS_RUNNING, done, total

4111	def _apply_aggregate(tv_doc_id: str) -> tuple[str, int, int]:
4112	    """After aggregation, apply state transitions only within the Running/Pass/Fail range."""
4113	    tv_status_row = db.get_tv_status(tv_doc_id)

4115	    new_status, done, total = _compute_tv_aggregate(tv_doc_id)
4116

4118	    # Moves among Running/Pass/Fail are allowed.

4120	        if current != new_status or (tv_status_row or {}).get("progress_done") != done \
4121	           or (tv_status_row or {}).get("progress_total") != total:
4122	            _transition_tv_status(tv_doc_id, new_status, done, total)
4123	            if new_status in {TV_STATUS_PASS, TV_STATUS_FAIL}:
4124	                db.insert_event(
4125	                    tv_doc_id, "tv_aggregated",
4126	                    note=f"done={done}/total={total} status={new_status}",

5109	_FILE_TREE_NATURAL_RE = re.compile(r"(\d+)")
5110	
5111	
5112	def _file_tree_sort_key(name: str, is_dir: bool) -> tuple:
5113	    """Sort key for file-tree entries (R0003 fix).
5114	
5115	    Folders are grouped before files (folders-first convention), and within
5116	    each group entries are ordered case-insensitively with numeric-aware
5117	    ("natural") comparison so that, e.g., ``file2`` precedes ``file10`` and
5118	    ``Zebra`` is not forced ahead of ``apple``. This replaces the previous bare
5119	    ``sorted(os.listdir(...))`` which interleaved folders and files in raw
5120	    Unicode codepoint order — causing numeric/uppercase-named files to float
5121	    above lowercase-named folders.
5122	    """
5123	    parts = _FILE_TREE_NATURAL_RE.split(name.lower())
5124	    # re.split with a capturing group yields text at even indices and digit
5125	    # runs at odd indices; cast digit runs to int for natural ordering.
5126	    natural = [int(part) if i % 2 else part for i, part in enumerate(parts)]
5127	    return (0 if is_dir else 1, natural)

5130	def get_file_tree(project_id: str) -> dict:
5131	    """Return the project's file tree.
5132	
5133	    Recursively scan the src/{project_name}/{branch} path.
5134	    Return an empty node list when the directory does not exist.
5135	    Entries are ordered folders-first, then by natural case-insensitive name
5136	    (see :func:`_file_tree_sort_key`).
5137	    """
5138	    from modules.flow_gate.db import projects as _proj
5139	    from modules.flow_gate.services import git_service
5140	
5141	    row = _proj.get_by_id(project_id)
5142	    project_name = (row.get("project_name") or "").strip() if row else ""
5143	    settings = _proj.get_settings(project_id)
5144	    branch = (settings.get("branch") or "main").strip() if settings else "main"
5145	
5146	    if not project_name:
5147	        return {"nodes": []}
5148	
5149	    # 0319 B0001: a git-integrated project's base checkout lives under the git
5150	    # base_branch, not project_settings.branch (which stays "main"); resolve it so
5151	    # the base file explorer shows the connected repo's source. A non-integrated or
5152	    # disabled project keeps the settings-branch folder — behaviour unchanged.
5153	    docs_root = str(git_service.base_src_root(project_id, project_name, branch))
5154	    if not os.path.isdir(docs_root):
5155	        return {"nodes": []}
5156	
5157	    nodes: list[dict] = []
5158	    node_id = 0
5159	
5160	    def _list_dir_with_retry(path: str, retries: int = 3, delay: float = 0.3) -> list:
5161	        # 0283 T0004 (NR0003 causes 2 and 3): remote/UNC storage can raise a transient
5162	        # OSError/PermissionError mid-walk — a network hiccup, or a worker writing the
5163	        # same directory (Windows sharing violation). The write path already defends
5164	        # against this (storage.filesystem.safe_rename retries on PermissionError); the
5165	        # read/tree path had no such guard and silently swallowed the error, collapsing
5166	        # the tree into an empty/partial result with no diagnostic. Mirror safe_rename's
5167	        # bounded retry+backoff, and only after it is exhausted log a warning and return
5168	        # an empty listing (degrading this subtree, never the whole request).
5169	        last_exc: Exception | None = None
5170	        for attempt in range(retries):
5171	            try:
5172	                with os.scandir(path) as it:
5173	                    return list(it)
5174	            except (OSError, PermissionError) as exc:
5175	                last_exc = exc
5176	                time.sleep(delay * (attempt + 1))
5177	        logger.warning(
5178	            "get_file_tree: could not list %s after %d attempts: %s",
5179	            path, retries, last_exc,
5180	        )
5181	        return []
5182	
5183	    def walk_directory(path: str, parent_id: str | None = None) -> None:
5184	        nonlocal node_id
5185	
5186	        visible: list[tuple[str, str, bool]] = []
5187	        for entry in _list_dir_with_retry(path):
5188	            name = entry.name
5189	            # Do not expose DB files or hidden items in the tree
5190	            if name.startswith(".") or name.lower().endswith(".db"):
5191	                continue
5192	            try:
5193	                # os.scandir caches the dirent type, so is_dir() usually needs no extra
5194	                # network round-trip (unlike os.path.isdir, which always stat()s).
5195	                is_dir = entry.is_dir()
5196	            except OSError:
5197	                # Transient stat failure on one entry: keep it visible as a file rather
5198	                # than dropping it or aborting the whole walk.
5199	                is_dir = False
5200	            visible.append((name, entry.path, is_dir))
5201	
5202	        # Folders-first + natural, case-insensitive ordering (R0003).
5203	        visible.sort(key=lambda item: _file_tree_sort_key(item[0], item[2]))
5204	
5205	        for entry, full_path, is_dir in visible:
5206	            node_id += 1
5207	            current_id = str(node_id)
5208	
5209	            try:
5210	                rel_path = os.path.relpath(full_path, docs_root)
5211	            except ValueError:
5212	                # Windows: relpath across different drives raises ValueError. Fall back
5213	                # to the absolute path rather than 500-ing the entire tree request.
5214	                rel_path = full_path
5215	
5216	            if is_dir:
5217	                node: dict[str, Any] = {
5218	                    "id": current_id,
5219	                    "parent_id": parent_id,
5220	                    "type": "folder",
5221	                    "name": entry,
5222	                    "label": entry,
5223	                    "path": rel_path,
5224	                    "permissions": ["read"],
5225	                    "children": [],
5226	                }
5227	                nodes.append(node)
5228	                walk_directory(full_path, current_id)
5229	            else:
5230	                nodes.append({
5231	                    "id": current_id,
5232	                    "parent_id": parent_id,
5233	                    "type": "file",
5234	                    "name": entry,
5235	                    "label": entry,
5236	                    "path": rel_path,
5237	                    "permissions": ["read", "download"],
5238	                })
5239	
5240	    walk_directory(docs_root)
5241	    return {"nodes": nodes}

5340	def get_group_tree(project_id: str) -> dict:
5341	    """Return the project's document tree.
5342	
5343	    Following the prototype document-explorer structure, return the
5344	    project -> module -> group -> document hierarchy as a flat node list.
5345	    """
5346	    # Query the group list (based on the main DB)
5347	    groups = db.get_groups_by_projects([project_id])
5348	
5349	    # 0275 NR0003 cause 3: this tree used to issue one documents query per group
5350	    # plus one 'created'-event memo_file query per document (≈ 2 + G + D queries
5351	    # per load — thousands on a real project, ruinous against a remote DB).
5352	    # Batch all three lookups upfront so a tree load stays at a handful of
5353	    # queries regardless of tree size.
5354	    #
5355	    # 0276 NR0003 finding 1: batching alone traded query count for bind-parameter
5356	    # count — the three batched lookups passed the project's whole group list and
5357	    # whole doc_id list as IN(...) parameters (thousands of %s per tree load, and
5358	    # still growing linearly with the project). Every one of those lists WAS the
5359	    # project, so `project_id = ?` expresses the same filter with one parameter,
5360	    # and the grouped/orphan split moves to Python below.
5361	    group_ids = [dict(g)["group_id"] for g in groups]
5362	    known_group_ids = set(group_ids)
5363	    all_tree_docs = [dict(d) for d in db.get_docs_for_tree_by_project(project_id)]
5364	    try:
5365	        memo_files = db.get_created_memo_files_map_by_project(project_id)
5366	    except Exception:
5367	        memo_files = {}
5368	
5369	    def _by_doc_id_desc(rows: list[dict]) -> list[dict]:
5370	        return sorted(rows, key=lambda d: d.get("doc_id") or "", reverse=True)
5371	
5372	    # Reproduces the ordering the replaced queries applied:
5373	    #   grouped -> ORDER BY group_id, doc_id DESC (per-group list: doc_id DESC)
5374	    #   orphans -> ORDER BY module, doc_id DESC   (NULL module first, as in SQL)
5375	    docs_by_group: dict[str, list[dict]] = {gid: [] for gid in group_ids}
5376	    orphan_docs: list[dict] = []
5377	    for doc in all_tree_docs:
5378	        doc_group_id = doc.get("group_id")
5379	        if doc_group_id in known_group_ids:
5380	            docs_by_group[doc_group_id].append(doc)
5381	        else:
5382	            orphan_docs.append(doc)
5383	    for _gid, _docs in docs_by_group.items():
5384	        docs_by_group[_gid] = _by_doc_id_desc(_docs)
5385	    orphan_docs = sorted(
5386	        _by_doc_id_desc(orphan_docs),
5387	        key=lambda d: (d.get("module") is not None, d.get("module") or ""),
5388	    )
5389	
5390	    # Fixed column names from the main DB schema
5391	    doc_type_col = "type_code"
5392	    doc_file_col = "file_path"
5393	
5394	    def _lookup_project(project_key: str) -> dict | None:
5395	        try:
5396	            row = db_projects.get_by_id(project_key)
5397	            if row:
5398	                return row
5399	        except Exception:
5400	            pass
5401	
5402	        server_dir = os.path.abspath(os.path.join(os.path.dirname(__file__), "..", ".."))
5403	        candidate_paths = [
5404	            os.path.join(server_dir, "flowgate.db"),
5405	            os.path.abspath("flowgate.db"),
5406	        ]
5407	        seen_paths: set[str] = set()
5408	        for db_path in candidate_paths:
5409	            db_path = os.path.normpath(db_path)
5410	            if db_path in seen_paths or not os.path.exists(db_path):
5411	                continue
5412	            seen_paths.add(db_path)
5413	            try:
5414	                with sqlite3.connect(f"file:{db_path}?mode=ro", uri=True) as project_conn:
5415	                    project_conn.row_factory = sqlite3.Row
5416	                    has_projects = project_conn.execute(
5417	                        "SELECT 1 FROM sqlite_master WHERE type='table' AND name='projects'"
5418	                    ).fetchone()
5419	                    if not has_projects:
5420	                        continue
5421	                    row = project_conn.execute(
5422	                        "SELECT * FROM projects WHERE project_id = ?",
5423	                        (project_key,),
5424	                    ).fetchone()
5425	                    if row:
5426	                        return dict(row)
5427	            except sqlite3.Error:
5428	                continue
5429	        return None
5430	
5431	    project_label = project_id
5432	    project_row = _lookup_project(project_id)
5433	    if project_row:
5434	        project_name = (project_row.get("project_name") or "").strip()
5435	        if project_name:
5436	            project_label = project_name
5437	
5438	    # 0449 T0004 item 3 (NR0003 E2): `nodes` is a FLAT list and the single source of truth —
5439	    # every node appears in it exactly once and names its parent through `parent_id`. Nodes
5440	    # used to be appended here AND nested into their parent's `children`, so the same object
5441	    # was serialized twice (measured on the live project: 4,210 unique ids serialized as
5442	    # 12,169 entries — 7,959 duplicates in a 4.8 MB response). No consumer read the nested
5443	    # copy: GroupTreeNode.vue computes its children from parent_id. Do not reintroduce a
5444	    # `children` key on these nodes; the duplication is pure transfer/parse cost, and it is
5445	    # what made a retry of a failed refresh so expensive.
5446	    nodes: list[dict] = []
5447	    project_node_id = f"project:{project_id}"
5448	    module_nodes: dict[str, dict[str, Any]] = {}
5449	
5450	    _dir_map = {
5451	        "outbox": db.OUTBOX_DIR,
5452	        "inbox": db.INBOX_DIR,
5453	        "processed": db.PROCESSED_DIR,
5454	        "accept": db.ACCEPT_DIR,
