43	def _safe_filename(filename: str) -> bool:
44	    """Block path traversal attacks."""
45	    return ".." not in filename and "/" not in filename and "\\" not in filename

48	def _move_file(filename: str, src_dir: str, dst_dir: str):
49	    src = os.path.join(src_dir, filename)
50	    dst = os.path.join(dst_dir, filename)
51	    if os.path.exists(src):
52	        os.makedirs(dst_dir, exist_ok=True)
53	        shutil.move(src, dst)

56	def _resolve_duplicate_filename(filename: str) -> str:
57	    """Append a suffix like _2, _3 if the same filename exists in processed/accept."""
58	    name_part, ext = os.path.splitext(filename)
59	    exists_in_processed = os.path.exists(os.path.join(db.PROCESSED_DIR, filename))
60	    exists_in_accept = _file_exists_in_accept(filename)
61	    if not exists_in_processed and not exists_in_accept:
62	        return filename
63	    for suffix in range(2, 100):
64	        candidate = f"{name_part}_{suffix}{ext}"
65	        if not os.path.exists(os.path.join(db.PROCESSED_DIR, candidate)) and \
66	           not _file_exists_in_accept(candidate):
67	            return candidate
68	    raise ValueError(f"Failed to resolve duplicate filename: {filename}")

71	def _file_exists_in_accept(filename: str) -> bool:
72	    """Check whether a filename exists in the accept directory (including subdirectories)."""
73	    for root, _, files in os.walk(db.ACCEPT_DIR):
74	        if filename in files:
75	            return True
76	    return False

79	def _move_to_error(filename: str, errors: list[str]):
80	    """Move a file to error/ and create a failure reason memo."""
81	    _move_file(filename, db.INBOX_DIR, db.ERROR_DIR)
82	    error_memo = os.path.join(db.ERROR_DIR, f"_error_{filename}")
83	    with open(error_memo, "w", encoding="utf-8") as f:
84	        f.write("---\n")
85	        f.write("reason: lint_failed\n")
86	        f.write(f"source_file: {filename}\n")
87	        f.write(f"created_at: {datetime.now().isoformat()}\n")
88	        f.write("---\n\n")
89	        f.write(f"# Lint Failed - {filename}\n\n")
90	        for e in errors:
91	            f.write(f"- {e}\n")

118	def _list_bucket_files(bucket: str, include_sidecar: bool = False) -> list[str]:
119	    folder = _bucket_dir(bucket)
120	    if not folder or not os.path.exists(folder):
121	        return []
122	
123	    files: list[str] = []
124	    for name in sorted(os.listdir(folder)):
125	        path = os.path.join(folder, name)
126	        if not os.path.isfile(path):
127	            continue
128	        if not name.endswith(".md"):
129	            continue
130	        if not include_sidecar and name.startswith("_"):
131	            continue
132	        files.append(name)
133	    return files

136	def _read_bucket_file(bucket: str, filename: str) -> str | None:
137	    folder = _bucket_dir(bucket)
138	    if not folder or not _safe_filename(filename):
139	        return None
140	    path = os.path.join(folder, filename)
141	    if not os.path.exists(path):
142	        return None
143	    with open(path, "r", encoding="utf-8") as f:
144	        return f.read()

147	def _extract_reason_from_sidecar(bucket: str, filename: str) -> dict[str, Any]:
148	    prefix = "_error_" if bucket == "error" else "_conflict_"
149	    sidecar = _read_bucket_file(bucket, f"{prefix}{filename}")
150	    if sidecar is None:
151	        return {"reason": "", "note": "", "related_target_id": ""}
152	
153	    header, _ = linter.parse_yaml_header(sidecar)
154	    note_lines = [
155	        line.strip()[2:].strip()
156	        for line in sidecar.splitlines()
157	        if line.strip().startswith("- ")
158	    ]
159	    return {
160	        "reason": (header or {}).get("reason", ""),
161	        "note": "\n".join(note_lines),
162	        "related_target_id": (header or {}).get("related_target_id", ""),
163	    }

166	def _split_header_and_body(content: str) -> tuple[dict | None, str]:
167	    header, _ = linter.parse_yaml_header(content)
168	    stripped = content.strip()
169	    if not stripped.startswith("---"):
170	        return header, stripped
171	
172	    end_idx = stripped.find("---", 3)
173	    if end_idx == -1:
174	        return header, stripped
175	
176	    body = stripped[end_idx + 3 :].strip()
177	    return header, body

207	    content = read_inbox_file(filename)

210	                "errors": ["File not found"], "header": None, "content": ""}

217	        "errors": errors,
218	        "header": header,
219	        "content": content,

247	def requeue_file(bucket: str, filename: str) -> dict:
248	    """Return an error/conflict file back to inbox."""
249	    if bucket not in ("error", "conflict"):
250	        return {"success": False, "message": f"Unsupported bucket: {bucket}"}
251	    if not _safe_filename(filename):
252	        return {"success": False, "message": "Invalid filename"}
253	
254	    src_dir = _bucket_dir(bucket) or ""
255	    src_path = os.path.join(src_dir, filename)
256	    if not os.path.exists(src_path):
257	        return {"success": False, "message": f"File not found: {filename}"}
258	
259	    _move_file(filename, src_dir, db.INBOX_DIR)
260	    reason = _extract_reason_from_sidecar(bucket, filename)
261	    db.insert_event(
262	        f"memo:{filename}",
263	        "requeued",
264	        memo_file=filename,
265	        reason=reason.get("reason") or None,
266	        related_target_id=reason.get("related_target_id") or None,
267	        note=f"{bucket} -> inbox",
268	    )
269	    return {"success": True, "bucket": bucket, "filename": filename, "moved_to": "inbox"}

272	def reprocess_file(bucket: str, filename: str) -> dict:
273	    """Restore an error/conflict file to inbox and immediately apply it."""
274	    requeue = requeue_file(bucket, filename)
275	    if not requeue.get("success"):
276	        return {"success": False, "step": "requeue", "error": requeue.get("message", "requeue failed")}
277	
278	    result = apply_file(filename)
279	    return {
280	        "success": bool(result.get("success")),
281	        "step": "apply",
282	        "bucket": bucket,
283	        "filename": filename,
284	        "apply_result": result,
285	    }

301	def get_document_detail(doc_id: str) -> dict | None:
302	    """Return document details (document/events/links/source)."""
303	    if not _safe_doc_id(doc_id):
304	        return None
305	
306	    doc = db.get_document_by_id(doc_id)
307	    if doc is None:
308	        return None
309	
310	    events = db.get_events_by_doc_id(doc_id)
311	    linked_docs = db.get_linked_result_documents(doc_id)
312	    target_id = (doc.get("target_id") or "").strip()
313	    target_doc = db.get_document_by_id(target_id) if target_id else None
314	
315	    source_memo = db.get_created_memo_file(doc_id)
316	    source_content: str | None = None
317	    source_bucket = ""
318	    if source_memo:
319	        doc_type = (doc.get("type") or "").strip()
320	        if doc_type in ("TV", "TVR"):
321	            # TV/TVR source files may be in 90_test_reports or the archive.
322	            for base, label in (
323	                (db.TEST_REPORTS_DIR, "test_reports"),
324	                (db.TEST_REPORTS_ARCHIVE_DIR, "test_reports_archive"),
325	            ):
326	                candidate = os.path.join(base, source_memo)
327	                if os.path.exists(candidate):
328	                    try:
329	                        with open(candidate, "r", encoding="utf-8") as f:
330	                            source_content = f.read()
331	                        source_bucket = label
332	                        break
333	                    except OSError:
334	                        source_content = None
335	        else:
336	            source_content = _read_bucket_file("processed", source_memo)
337	            if source_content is not None:
338	                source_bucket = "processed"
339	    source_header, source_body = _split_header_and_body(source_content or "") if source_content else (None, "")
340	
341	    # ── Group Chain (L001 §4) ─────────────────────────────────────
342	    group_id = (doc.get("group_id") or "").strip() or None
343	    chain: dict | None = None
344	    chain_unavailable_reason: str | None = None
345	    if group_id:
346	        chain_docs = db.get_documents_by_group_id(group_id)
347	        chain_nodes, chain_tree_warnings = build_chain_tree(chain_docs, doc_id)
348	        gap_warnings = detect_gaps(group_id)
349	        chain = {
350	            "group_id": group_id,
351	            "nodes": chain_nodes,
352	            "gap_warnings": gap_warnings,
353	            "chain_warnings": chain_tree_warnings,
354	        }
355	    else:
356	        chain_unavailable_reason = "no group_id"
357	
358	    # ── TV/TVR-specific Data ──────────────────────────────────────
359	    tv_extra: dict = {}
360	    doc_type = (doc.get("type") or "").strip()
361	    if doc_type == "TV":
362	        tv_extra["tv_status_row"] = db.get_tv_status(doc_id)
363	        tv_extra["tv_scenarios"] = db.get_tv_scenarios(doc_id)
364	        tv_extra["tv_clear_scope"] = db.get_tv_clear_scope(doc_id)
365	        # TVR link list
366	        tvr_docs = db.get_documents_by_target_id(doc_id, types=("TVR",))
367	        tv_extra["tvr_docs"] = list(tvr_docs)
368	    elif doc_type == "TVR":
369	        target_tv_id = (doc.get("target_id") or "").strip()
370	        if target_tv_id:
371	            tv_extra["target_tv_doc"] = db.get_document_by_id(target_tv_id)
372	            tv_extra["target_tv_status_row"] = db.get_tv_status(target_tv_id)
373	            tv_extra["target_tv_scenarios"] = db.get_tv_scenarios(target_tv_id)
374	        else:
375	            tv_extra["target_tv_doc"] = None
376	            tv_extra["target_tv_status_row"] = None
377	            tv_extra["target_tv_scenarios"] = []
378	
379	    return {
380	        "doc": doc,
381	        "events": events,
382	        "linked_documents": linked_docs,
383	        "target_document": target_doc,
384	        "chain": chain,
385	        "chain_unavailable_reason": chain_unavailable_reason,
386	        "source": {
387	            "bucket": source_bucket,
388	            "filename": source_memo or "",
389	            "header": source_header,
390	            "body": source_body,
391	            "raw": source_content or "",
392	        },
393	        **tv_extra,
394	    }

397	def get_memo_detail(bucket: str, filename: str) -> dict | None:
398	    """Return the original memo content (header/body/related info) in a bucket."""
399	    if bucket not in BUCKET_DIRS:
400	        return None
401	    if not _safe_filename(filename):
402	        return None
403	
404	    content = _read_bucket_file(bucket, filename)
405	    if content is None:
406	        return None
407	
408	    header, body = _split_header_and_body(content)
409	    doc_id = None
410	    related_events: list[dict] = []
411	    target_id = (header or {}).get("target_id") if header else None
412	
413	    if bucket == "processed":
414	        docs = db.get_all_documents()
415	        for d in docs:
416	            if db.get_created_memo_file(d["doc_id"]) == filename:
417	                doc_id = d["doc_id"]
418	                related_events = db.get_events_by_doc_id(doc_id)
419	                break
420	
421	    reason = _extract_reason_from_sidecar(bucket, filename) if bucket in ("error", "conflict") else {
422	        "reason": "",
423	        "note": "",
424	        "related_target_id": "",
425	    }
426	
427	    return {
428	        "bucket": bucket,
429	        "filename": filename,
430	        "header": header,
431	        "body": body,
432	        "raw": content,
433	        "doc_id": doc_id,
434	        "target_id": target_id,
435	        "related_events": related_events,
436	        "reason": reason,
437	    }

440	def get_filtered_documents(filters: dict[str, str]) -> dict:
441	    """Build document filter/search results and option candidates."""
442	    project = (filters.get("project") or "").strip()
443	    doc_type = (filters.get("type") or "").strip()
444	    status = (filters.get("status") or "").strip()
445	    owner = (filters.get("owner") or "").strip()
446	    priority = (filters.get("priority") or "").strip().lower()
447	    query = (filters.get("q") or "").strip()
448	    view = (filters.get("view") or "").strip()
449	
450	    owner_missing = owner == "__missing__"
451	    if owner_missing:
452	        owner = ""
453	
454	    docs = db.get_documents_filtered(
455	        project=project or None,
456	        doc_type=doc_type or None,
457	        status=status or None,
458	        owner=owner or None,
459	        priority=priority or None,
460	        query=query or None,
461	    )
462	
463	    if owner_missing:
464	        docs = [d for d in docs if not (d.get("owner") or "").strip()]
465	
466	    if view == "n_without_t":
467	        filtered_docs: list[dict] = []
468	        for d in docs:
469	            if d.get("type") != "N":
470	                continue
471	            followers = db.get_documents_by_target_id(d.get("doc_id", ""), types=("T",))
472	            if not followers:
473	                filtered_docs.append(d)
474	        docs = filtered_docs
475	    elif view == "high_priority":
476	        docs = [d for d in docs if (d.get("priority") or "").lower() in ("high", "urgent")]
477	
478	    all_docs = db.get_all_documents()
479	    options = {
480	        "project": sorted({(d.get("project") or "") for d in all_docs if d.get("project")}),
481	        "type": sorted({(d.get("type") or "") for d in all_docs if d.get("type")}),
482	        "status": sorted({(d.get("status") or "") for d in all_docs if d.get("status")}),
483	        "owner": sorted({(d.get("owner") or "") for d in all_docs if d.get("owner")}),
484	        "priority": sorted({(d.get("priority") or "") for d in all_docs if d.get("priority")}),
485	    }
486	
487	    return {
488	        "filters": {
489	            "project": project,
490	            "type": doc_type,
491	            "status": status,
492	            "owner": owner,
493	            "priority": priority,
494	            "q": query,
495	            "view": view,
496	        },
497	        "documents": docs,
498	        "count": len(docs),
499	        "options": options,
500	    }

585	        "closed": sum(1 for d in docs if d.get("status") == "closed"),
586	        "rejected": sum(1 for d in docs if d.get("status") == "rejected"),
587	    }
588	    status_counts["conflict"] = len(_list_bucket_files("conflict"))
589	
590	    recent_24h = 0
591	    for ev in db.get_recent_events(500):
592	        if ev.get("event_type") != "created":
593	            continue
594	        created_at = _parse_iso_datetime_safe(ev.get("created_at"))
595	        if created_at is None:
596	            continue
597	        if (now - created_at).total_seconds() <= 86400:
598	            recent_24h += 1
599	
600	    by_project: dict[str, dict] = {}
601	    for d in docs:
602	        project = d.get("project") or "(unknown)"
603	        bucket = by_project.setdefault(
604	            project,
605	            {"total": 0, "open": 0, "closed": 0, "rejected": 0},
606	        )
607	        bucket["total"] += 1
608	        status = d.get("status")
609	        if status in ("open", "closed", "rejected"):
610	            bucket[status] += 1
611	
612	    return {
613	        "status_counts": status_counts,
614	        "recent_created_24h": recent_24h,
615	        "projects": by_project,
616	        "updated_at": now.isoformat(),
617	    }

733	
734	def _assign_group_id(
735	    doc_type: str,
736	    target_id: str | None,
737	    project: str,
738	    module: str | None,
739	) -> tuple[str | None, list[str]]:
740	    """Issue or inherit a group_id based on L001 §2-1.
741	
742	    Returns:
743	        (group_id, gap_warnings) — group_id may be None
744	    Raises:
745	        ValueError: Numbering sequence upper limit exceeded (L001 §5)
746	    """
747	    warnings: list[str] = []
748	
749	    if not target_id:
750	        if doc_type in _ROOT_TYPES:
751	            group_id = db.issue_group_id(project, module or "")  # propagate ValueError
752	            return group_id, warnings
753	        else:
754	            warnings.append(f"Not a root type + no target_id — group not assigned ({doc_type})")
755	            return None, warnings
756	
757	    parent = db.get_document_by_id(target_id)
758	    if parent is None:
759	        warnings.append(f"Parent document not found — {target_id}")
760	        return None, warnings
761	
762	    parent_group_id = (parent.get("group_id") or "").strip() or None
763	    if not parent_group_id:
764	        warnings.append(f"Parent group_id missing — {target_id}")
765	        return None, warnings
766	
767	    return parent_group_id, warnings

770	def detect_gaps(group_id: str) -> list[str]:
771	    """Return the list of workflow gap warnings within the group based on L001 §4-2."""
772	    docs = db.get_documents_by_group_id(group_id)
773	    warnings: list[str] = []
774	
775	    # G-01: No TR for T
776	    for doc in docs:
777	        if doc.get("type") == "T":
778	            tr_exists = any(
779	                d.get("type") == "TR" and d.get("target_id") == doc["doc_id"]
780	                for d in docs
781	            )
782	            if not tr_exists:
783	                warnings.append(f"TR not registered — no TR for {doc['doc_id']}")
784	
785	    # G-02: No NR for N
786	    for doc in docs:
787	        if doc.get("type") == "N":
788	            nr_exists = any(
789	                d.get("type") == "NR" and d.get("target_id") == doc["doc_id"]
790	                for d in docs
791	            )
792	            if not nr_exists:
793	                warnings.append(f"NR not registered — no NR for {doc['doc_id']}")
794	
795	    # G-03: No follow-up document for M
796	    for doc in docs:
797	        if doc.get("type") == "M":
798	            followup_exists = any(
799	                d.get("target_id") == doc["doc_id"] and d.get("type") in _M_FOLLOWUP_TYPES
800	                for d in docs
801	            )
802	            if not followup_exists:
803	                warnings.append(f"No follow-up document — no follow-up for {doc['doc_id']}")
804	
805	    # G-04: Only root exists
806	    if len(docs) == 1:
807	        warnings.append("No follow-up document — only root document in group")
808	
809	    return warnings

812	def build_chain_tree(
813	    docs: list[dict],
814	    current_doc_id: str,
815	) -> tuple[list[dict], list[str]]:
816	    """Return BFS-ordered chain node list based on L001 §4-3."""
817	    warnings: list[str] = []
818	    if not docs:
819	        return [], warnings
820	
821	    doc_ids = {d["doc_id"] for d in docs}
822	
823	    # 1. Find root
824	    root_candidates = [d for d in docs if not (d.get("target_id") or "").strip()]
825	    if not root_candidates:
826	        root_candidates = [
827	            d for d in docs
828	            if (d.get("target_id") or "").strip() not in doc_ids
829	        ]
830	    if not root_candidates:
831	        root_candidates = sorted(docs, key=lambda d: d.get("created_at") or "")
832	
833	    if len(root_candidates) > 1:
834	        warnings.append("Multiple roots detected")
835	        root_candidates = sorted(root_candidates, key=lambda d: d.get("created_at") or "")
836	    root = root_candidates[0]
837	
838	    # 2. Build children map (created_at ASC)
839	    children_map: dict[str, list[dict]] = {}
840	    for doc in docs:
841	        pid = (doc.get("target_id") or "").strip()
842	        children_map.setdefault(pid, []).append(doc)
843	    for children in children_map.values():
844	        children.sort(key=lambda d: d.get("created_at") or "")
845	
846	    # 3. BFS traversal
847	    result: list[dict] = []
848	    queue = [root]
849	    depth = 0
850	    visited: set[str] = set()
851	
852	    while queue:
853	        depth += 1
854	        if depth > _CHAIN_MAX_DEPTH:
855	            warnings.append("Chain view depth limit exceeded — some nodes omitted")
856	            break
857	        next_queue: list[dict] = []
858	        for node in queue:
859	            nid = node["doc_id"]
860	            if nid in visited:
861	                continue
862	            visited.add(nid)
863	            result.append({
864	                "doc_id": nid,
865	                "type": node.get("type", ""),
866	                "parent_id": (node.get("target_id") or "").strip() or None,
867	                "title": node.get("title", ""),
868	                "status": node.get("status", ""),
869	                "is_current": nid == current_doc_id,
870	            })
871	            next_queue.extend(children_map.get(nid, []))
872	        queue = next_queue
873	
874	    return result, warnings

925	        "title": header["title"],
926	        "status": "open",
927	    }
928	
929	    # If target_id is present, show link info regardless of type
930	    target_id = (header.get("target_id") or "").strip()
931	    if target_id:
932	        target_doc = db.get_document_by_id(target_id) if target_id else None
933	        preview["target_id"] = target_id
934	        preview["target_transition"] = (
935	            f"{target_id}: {target_doc['status']} → closed"
936	            if target_doc else
937	            f"{target_id}: document not found (will error on Apply)"

943	# ── Apply ─────────────────────────────────────────────────────────
944	
945	def apply_file(filename: str, _depth: int = 0) -> dict:
946	    """Numbering → DB save → inbox → processed move."""
947	    if _depth > 1:
948	        return {"success": False, "error": "CONFIRM chain depth exceeded", "filename": filename}
949	
950	    if not _safe_filename(filename):
951	        return {"success": False, "error": "Invalid filename", "filename": filename}
952	
953	    if db.is_file_processed(filename):
954	        return {"success": False, "error": "File already processed", "filename": filename}
955	
956	    result = lint_inbox_file(filename)
957	    if not result["valid"]:
958	        _move_to_error(filename, result["errors"])
959	        db.insert_event(
960	            f"memo:{filename}",
961	            "lint_failed",
962	            memo_file=filename,
963	            reason="lint_failed",
964	            note="; ".join(result["errors"]),
965	        )
966	        return {"success": False, "error": "Lint validation failed",
967	                "errors": result["errors"], "filename": filename}
968	
969	    file_hash = hashlib.sha256(result["content"].encode("utf-8")).hexdigest()
970	    if db.is_hash_processed(file_hash):
971	        return {
972	            "success": False,
973	            "error": "File already processed",
974	            "filename": filename,
975	        }
976	
977	    header = result["header"]
978	    doc_type = header["type"]
979	
980	    # ── D/DB/P/L single-item Apply blocked: must be processed via DC ──
981	    if doc_type in ("D", "DB", "P", "L"):
982	        return {
983	            "success": False,
984	            "error": f"{doc_type} type cannot be applied individually. Submit via DC (design complete).",
985	            "filename": filename,
986	        }
987	
988	    # ── CONFIRM Processing ──
989	    if doc_type == "CONFIRM":
990	        target = header.get("target", "")
991	        if not target:
992	            _move_to_error(filename, ["CONFIRM memo has no target"])
993	            db.insert_event(
994	                f"memo:{filename}",
995	                "confirm_failed",
996	                memo_file=filename,
997	                reason="confirm_target_missing",
998	                note="CONFIRM memo has no target",
999	            )
1000	            return {"success": False, "error": "CONFIRM target missing", "filename": filename}
1001	
1002	        target_path = os.path.join(db.INBOX_DIR, target)
1003	        if not os.path.exists(target_path):
1004	            _move_to_error(filename, [f"CONFIRM target '{target}' is not in inbox"])
1005	            db.insert_event(
1006	                f"memo:{filename}",
1007	                "confirm_failed",
1008	                memo_file=filename,
1009	                reason="confirm_target_not_found",
1010	                related_doc_id=target,
1011	                note=f"CONFIRM target '{target}' is not in inbox",
1012	            )
1013	            return {"success": False, "error": f"target not found: {target}", "filename": filename}
1014	
1015	        target_result = apply_file(target, _depth=_depth + 1)
1016	        _move_file(filename, db.INBOX_DIR, db.PROCESSED_DIR)
1017	        db.insert_event(
1018	            f"memo:{filename}",
1019	            "confirm_processed",
1020	            memo_file=filename,
1021	            reason="confirm_applied",
1022	            related_doc_id=target,
1023	            note=f"target={target}, success={target_result.get('success')}",
1024	        )
1025	        return {"success": target_result["success"],
1026	                "confirm_target": target, "target_result": target_result,
1027	                "filename": filename}
1028	
1029	    # ── General Document Processing ──
1030	    project = header["project"]
1031	    module = (header.get("module") or "").strip()
1032	    doc_id = db.get_next_doc_id(project, module, doc_type)
1033	    title = header["title"]
1034	    target_id = (header.get("target_id") or "").strip() or None
1035	    owner = (header.get("owner") or "").strip() or None
1036	    priority = (header.get("priority") or "").strip().lower() or None
1037	    due_date = (header.get("due_date") or "").strip() or None
1038	
1039	    # ── group_id: used directly from header ──
1040	    group_id = (header.get("group_id") or "").strip() or None
1041	    assign_warnings: list[str] = []
1042	
1043	    # ── NR/TR: transition target document status to closed ──
1044	    if doc_type in linter.NR_TR_TYPES:
1045	        if not target_id:
1046	            _move_to_error(filename, [f"{doc_type} type requires target_id"])
1047	            db.insert_event(
1048	                f"memo:{filename}",
1049	                "apply_failed",
1050	                memo_file=filename,
1051	                reason="target_missing",
1052	                note=f"{doc_type} type requires target_id",
1053	            )
1054	            return {"success": False, "error": "target_id missing", "filename": filename}
1055	
1056	        target_doc = db.get_document_by_id(target_id)
1057	        if target_doc is None:
1058	            _move_to_error(filename, [f"target_id '{target_id}' document not found"])
1059	            db.insert_event(
1060	                f"memo:{filename}",
1061	                "apply_failed",
1062	                memo_file=filename,
1063	                reason="target_not_found",
1064	                related_target_id=target_id,
1065	                note=f"target_id '{target_id}' document not found",
1066	            )
1067	            return {"success": False, "error": f"target not found: {target_id}", "filename": filename}
1068	
1069	        has_conflict, open_result_doc_ids = db.has_open_result_for_target(target_id)
1070	        if has_conflict:
1071	            reason = (
1072	                f"there are already open result documents for target '{target_id}': "
1073	                f"{', '.join(open_result_doc_ids)}"
1074	            )
1075	            _move_to_conflict(filename, reason, related_target_id=target_id)
1076	            db.insert_event(
1077	                target_id,
1078	                "conflict_detected",
1079	                memo_file=filename,
1080	                file_hash=file_hash,
1081	                reason="conflict_open_result_exists",
1082	                related_target_id=target_id,
1083	                related_doc_id=",".join(open_result_doc_ids),
1084	                note=reason,
1085	            )
1086	            return {"success": False, "error": reason, "filename": filename}
1087	
1088	        ok, msg = db.update_document_status(target_id, "closed")
1089	        if not ok:
1090	            _move_to_error(filename, [f"target '{target_id}' status transition failed: {msg}"])
1091	            db.insert_event(
1092	                f"memo:{filename}",
1093	                "apply_failed",
1094	                memo_file=filename,
1095	                reason="target_transition_failed",
1096	                related_target_id=target_id,
1097	                note=msg,
1098	            )
1099	            return {"success": False, "error": msg, "filename": filename}
1100	
1101	        db.insert_event(target_id, "status_changed",
1102	                        memo_file=filename, note=f"{msg} (by {doc_type})")
1103	
1104	    db.insert_document(
1105	        doc_id,
1106	        doc_type,
1107	        project,
1108	        module,
1109	        title,
1110	        target_id=target_id,
1111	        group_id=group_id,
1112	        owner=owner,
1113	        priority=priority,
1114	        due_date=due_date,
1115	    )
1116	    db.insert_event(doc_id, "created", memo_file=filename, file_hash=file_hash, note=f"hash:{file_hash}")
1117	
1118	    # Directive types (DS/N/T) remain in inbox — moved to accept/reject on approval/rejection
1119	    # Q type moves directly to accept/ after apply (T030)
1120	    if doc_type in ("DS", "N", "T"):
1121	        actual_filename = filename
1122	        moved_to = "inbox"
1123	    elif doc_type == "Q":
1124	        # Q moves inbox→accept/ directly (T030: process/ passthrough not needed)
1125	        actual_filename = _resolve_duplicate_filename(filename)
1126	        if actual_filename != filename:
1127	            src = os.path.join(db.INBOX_DIR, filename)
1128	            tmp = os.path.join(db.INBOX_DIR, actual_filename)
1129	            if os.path.exists(src):
1130	                os.rename(src, tmp)
1131	        _move_file(actual_filename, db.INBOX_DIR, db.ACCEPT_DIR)
1132	        moved_to = "accept"
1133	    else:
1134	        # Duplicate filename handling: append suffix if same filename exists in processed/accept
1135	        actual_filename = _resolve_duplicate_filename(filename)
1136	        if actual_filename != filename:
1137	            src = os.path.join(db.INBOX_DIR, filename)
1138	            tmp = os.path.join(db.INBOX_DIR, actual_filename)
1139	            if os.path.exists(src):
1140	                os.rename(src, tmp)
1141	        _move_file(actual_filename, db.INBOX_DIR, db.PROCESSED_DIR)
1142	        moved_to = "processed"
1143	
1144	    gap_warnings = detect_gaps(group_id) if group_id else []
1145	    all_warnings = assign_warnings + gap_warnings
1146	
1147	    return {

1223	# ── Briefing ─────────────────────────────────────────────────────────
1224	
1225	class BriefContext:
1226	    """Per-request prefetch shared by the brief / gaps / queue / handover path.
1227	
1228	    0276 NR0003 finding 3: these four entry points each re-queried the same data.
1229	    A single /brief request ran get_open_documents() up to four times, executed
1230	    detect_workflow_gaps() twice in full, and issued five separate N+1 cascades
1231	    (one query per open document, per queue item, per conflict event) — 100+
1232	    queries for 30 open documents, growing linearly with the document count.
1233	
1234	    Design decision (CH0004): the shared state is an *explicit object* threaded
1235	    as an optional argument, not a module-level request-scoped cache. The server
1236	    has no request-scope plumbing to hang a cache on, and a module-level one
1237	    would leak a stale snapshot across requests and threads. Passing it also
1238	    keeps the data flow visible at each call site.
1239	
1240	    Every lookup is lazy and memoised, so each underlying query runs at most once
1241	    per context and only if something actually asks for it — a standalone
1242	    detect_workflow_gaps() does not pay for the linked-document scan it never
1243	    reads, while a /brief request shares all of them.
1244	
1245	    Callers that pass no context keep the previous behaviour: each public
1246	    function builds a private one, so signatures stay backward compatible.
1247	    """
1248	
1249	    def __init__(self):
1250	        self._open_docs: list[dict] | None = None
1251	        self._linked_by_target: dict[str, list[dict]] | None = None
1252	        self._followups_by_target: dict[str, set] | None = None
1253	        self._latest_events: dict[str, dict | None] = {}
1254	        self._workflow_gaps: dict | None = None
1255	
1256	    @property
1257	    def open_docs(self) -> list[dict]:
1258	        """The open-document snapshot every consumer shares.
1259	
1260	        Sharing it also makes one response internally consistent; the four
1261	        separate queries could each observe a different commit.
1262	        """
1263	        if self._open_docs is None:
1264	            self._open_docs = db.get_open_documents()
1265	        return self._open_docs
1266	
1267	    @property
1268	    def linked_by_target(self) -> dict[str, list[dict]]:
1269	        """target_id -> NR/TR documents. One scan instead of N+1 ①②③."""
1270	        if self._linked_by_target is None:
1271	            self._linked_by_target = db.get_linked_result_documents_map()
1272	        return self._linked_by_target
1273	
1274	    @property
1275	    def followups_by_target(self) -> dict[str, set]:
1276	        """target_id -> follow-up type codes. One query instead of N+1 ④.
1277	
1278	        NR is included so get_handover()'s pending list resolves from it too.
1279	        """
1280	        if self._followups_by_target is None:
1281	            self._followups_by_target = db.get_followup_type_map(("T", "TR", "NR"))
1282	        return self._followups_by_target
1283	
1284	    def linked(self, doc_id: str) -> list[dict]:
1285	        """NR/TR documents referencing doc_id (same shape as db.get_linked_result_documents)."""
1286	        return self.linked_by_target.get(doc_id, [])
1287	
1288	    def has_followup(self, doc_id: str, types: tuple[str, ...]) -> bool:
1289	        """Whether a follow-up document of any of `types` targets doc_id."""
1290	        existing = self.followups_by_target.get(doc_id)
1291	        return bool(existing and existing.intersection(types))
1292	
1293	    def latest_events(self, doc_ids: list[str]) -> dict[str, dict | None]:
1294	        """Memoised latest-event map; queries only ids not seen yet."""
1295	        missing = [d for d in doc_ids if d and d not in self._latest_events]
1296	        if missing:
1297	            # Record every requested id first so documents without any event are
1298	            # remembered as "looked up, none found" and never re-queried.
1299	            for doc_id in missing:
1300	                self._latest_events[doc_id] = None
1301	            self._latest_events.update(db.get_latest_events_map(missing))
1302	        return self._latest_events
1303	
1304	    def workflow_gaps(self) -> dict:
1305	        """detect_workflow_gaps() computed at most once per request."""
1306	        if self._workflow_gaps is None:
1307	            self._workflow_gaps = detect_workflow_gaps(ctx=self)
1308	        return self._workflow_gaps

1546	def detect_workflow_gaps(now: datetime | None = None, ctx: "BriefContext | None" = None) -> dict:
1547	    """Detect operational gaps (stale/overdue/follow-up/review bottlenecks)."""
1548	    now_dt = now or datetime.now()
1549	    today = now_dt.date()
1550	    ctx = ctx or BriefContext()
1551	    open_docs = ctx.open_docs
1552	
1553	    stale_open: list[dict] = []
1554	    overdue: list[dict] = []
1555	    unassigned_important: list[dict] = []
1556	    missing_followup_from_n: list[dict] = []
1557	    missing_followup_from_t: list[dict] = []
1558	    review_stuck: list[dict] = []
1559	
1560	    for doc in open_docs:
1561	        doc_id = doc.get("doc_id") or ""
1562	        doc_type = doc.get("type")
1563	        priority = (doc.get("priority") or "").strip().lower()
1564	        owner = (doc.get("owner") or "").strip()
1565	
1566	        updated_days = _days_since(doc.get("updated_at"), now_dt)
1567	        created_days = _days_since(doc.get("created_at"), now_dt)
1568	
1569	        if updated_days is not None and updated_days >= WORKFLOW_GAP_RULES["stale_open_days"]:
1570	            _append_gap_item(
1571	                stale_open,
1572	                doc,
1573	                f"open for {int(updated_days)}+ days since last updated_at",
1574	            )
1575	
1576	        due = _parse_iso_date(doc.get("due_date"))
1577	        if due is not None:
1578	            overdue_days = (today - due).days
1579	            if overdue_days > WORKFLOW_GAP_RULES["overdue_grace_days"]:
1580	                _append_gap_item(
1581	                    overdue,
1582	                    doc,
1583	                    f"due date {due.isoformat()} passed (+{overdue_days} days)",
1584	                )
1585	
1586	        if priority in ("high", "urgent") and not owner:
1587	            _append_gap_item(unassigned_important, doc, f"{priority} priority document has no owner")
1588	
1589	        if doc_type == "N" and created_days is not None and created_days >= WORKFLOW_GAP_RULES["followup_grace_days_from_n"]:
1590	            if not ctx.has_followup(doc_id, ("T",)):
1591	                _append_gap_item(
1592	                    missing_followup_from_n,
1593	                    doc,
1594	                    f"no follow-up T within {int(created_days)} days of creation",
1595	                )
1596	
1597	        if doc_type == "T" and created_days is not None and created_days >= WORKFLOW_GAP_RULES["followup_grace_days_from_t"]:
1598	            if not ctx.has_followup(doc_id, ("TR",)):
1599	                _append_gap_item(
1600	                    missing_followup_from_t,
1601	                    doc,
1602	                    f"no follow-up TR within {int(created_days)} days of creation",
1603	                )
1604	
1605	        if doc_type in ("NR", "TR") and updated_days is not None and updated_days >= WORKFLOW_GAP_RULES["review_stuck_days"]:
1606	            _append_gap_item(
1607	                review_stuck,
1608	                doc,
1609	                f"review document has been open for {int(updated_days)}+ days",
1610	            )
1611	
1612	    categories = {
1613	        "stale_open": _build_gap_category(stale_open, ctx),
1614	        "overdue": _build_gap_category(overdue, ctx),
1615	        "unassigned_important": _build_gap_category(unassigned_important, ctx),
1616	        "missing_followup_from_n": _build_gap_category(missing_followup_from_n, ctx),
1617	        "missing_followup_from_t": _build_gap_category(missing_followup_from_t, ctx),
1618	        "review_stuck": _build_gap_category(review_stuck, ctx),
1619	    }
1620	
1621	    counts = {name: cat["count"] for name, cat in categories.items()}
1622	    return {
1623	        "rules": dict(WORKFLOW_GAP_RULES),
1624	        "counts": counts,
1625	        "total": sum(counts.values()),
1626	        "categories": categories,
1627	    }

1648	def build_action_queue(ctx: BriefContext | None = None) -> dict:
1649	    """Build the operational priority queue by category."""
1650	    ctx = ctx or BriefContext()
1651	    open_docs = ctx.open_docs
1652	    workflow_gaps = ctx.workflow_gaps()
1653	    open_n = [d for d in open_docs if d.get("type") == "N"]
1654	    open_t = [d for d in open_docs if d.get("type") == "T"]
1655	    open_review = db.get_documents_by_status_and_types("open", ("NR", "TR"))
1656	    rejected = db.get_documents_by_status("rejected")
1657	    stale_open = workflow_gaps["categories"]["stale_open"]["items"]
1658	
1659	    needs_dispatch = _find_target_docs_with_no_followup(open_n, ("T",), ctx)
1660	    needs_result = _find_target_docs_with_no_followup(open_t, ("NR", "TR"), ctx)
1661	
1662	    # One query for every conflict event's document instead of one per event
1663	    # (0276 NR0003 finding 3, N+1 ⑤). Insertion order still follows the event order.
1664	    conflict_docs_map: dict[str, dict] = {}
1665	    conflict_events = db.get_conflict_events(50)
1666	    conflict_doc_map = db.get_documents_by_ids(
1667	        [e.get("doc_id") for e in conflict_events if e.get("doc_id")]
1668	    )
1669	    for e in conflict_events:
1670	        doc_id = e.get("doc_id")
1671	        if not doc_id or doc_id in conflict_docs_map:
1672	            continue
1673	        doc = conflict_doc_map.get(doc_id)
1674	        if doc:
1675	            conflict_docs_map[doc_id] = doc
1676	
1677	    if os.path.exists(db.CONFLICT_DIR):
1678	        for fname in sorted(os.listdir(db.CONFLICT_DIR)):
1679	            if fname.startswith("_"):
1680	                continue
1681	            base = {
1682	                "doc_id": f"conflict-file:{fname}",
1683	                "type": "-",
1684	                "project": "-",
1685	                "module": "",
1686	                "title": fname,
1687	                "status": "conflict",
1688	                "updated_at": "",
1689	                "target_id": "",
1690	            }
1691	            conflict_docs_map[base["doc_id"]] = base
1692	
1693	    categories_raw = {
1694	        "needs_dispatch": needs_dispatch,
1695	        "needs_result": needs_result,
1696	        "needs_review": open_review,
1697	        "rejected_followup": rejected,
1698	        "conflict": list(conflict_docs_map.values()),
1699	        "stale_open": stale_open,
1700	    }
1701	
1702	    # Prime the latest-event cache for every category in one query, so the
1703	    # per-category _with_latest_event() calls below are all cache hits.
1704	    ctx.latest_events([
1705	        d.get("doc_id") for docs in categories_raw.values() for d in docs if d.get("doc_id")
1706	    ])
1707	
1708	    categories: dict[str, dict] = {}
1709	    for name, docs in categories_raw.items():
1710	        enriched = _with_latest_event(docs, ctx)
1711	        with_actions: list[dict] = []
1712	        for item in enriched:
1713	            doc_id = item.get("doc_id") or ""
1714	            if doc_id.startswith("conflict-file:"):
1715	                item["next_action"] = "Check conflict reason, then decide to reprocess or discard"
1716	            else:
1717	                linked_docs = ctx.linked(doc_id) if doc_id else []
1718	                _, desc = _suggest_next_action(item, linked_docs)
1719	                item["next_action"] = desc
1720	            with_actions.append(item)
1721	        sorted_items = _sort_queue_items(with_actions)
1722	        categories[name] = {
1723	            "count": len(sorted_items),
1724	            "items": sorted_items,
1725	            "sample": sorted_items[:3],
1726	        }
1727	
1728	    return {
1729	        "stale_open_hours": STALE_OPEN_HOURS,
1730	        "workflow_gaps": workflow_gaps,
1731	        "categories": categories,
1732	    }

1754	def _suggest_next_action(doc: dict, linked_docs: list[dict]) -> tuple[str, str]:
1755	    """Suggest the next action based on document status/links."""
1756	    doc_type = doc.get("type")
1757	    status = doc.get("status")
1758	
1759	    if status == "rejected":
1760	        return "followup_rework", "Check rejection reason and request rework"
1761	
1762	    if status == "open" and doc_type == "N":
1763	        has_t = any(d.get("type") == "T" for d in linked_docs)
1764	        if not has_t:
1765	            return "dispatch", "Draft follow-up T work order"
1766	
1767	    if status == "open" and doc_type == "T":
1768	        has_tr = any(d.get("type") == "TR" for d in linked_docs)
1769	        if not has_tr:
1770	            return "result_request", "Draft follow-up TR result request"
1771	
1772	    if status == "open" and doc_type in ("NR", "TR"):
1773	        return "review_request", "Draft review request"
1774	
1775	    return "monitor", "Monitor status and provide further instructions"

1778	def build_worker_draft(doc_id: str) -> dict | None:
1779	    """Build a dispatch draft for the operations leader to send to a worker."""
1780	    doc = db.get_document_by_id(doc_id)
1781	    if not doc:
1782	        return None
1783	
1784	    linked_docs = db.get_linked_result_documents(doc_id)
1785	    target_id = (doc.get("target_id") or "").strip()
1786	    target_doc = db.get_document_by_id(target_id) if target_id else None
1787	    recent_events = db.get_recent_events_by_doc_id(doc_id, 5)
1788	    latest_event = recent_events[0] if recent_events else None
1789	
1790	    action_type, action_description = _suggest_next_action(doc, linked_docs)
1791	
1792	    project = doc.get("project", "")
1793	    module = doc.get("module") or ""
1794	    module_text = module if module else "(all)"
1795	    title = doc.get("title", "")
1796	    status = doc.get("status", "")
1797	
1798	    worker_message = (
1799	        "[FlowGate Work Order Draft]\n"
1800	        f"- Target Document: {doc_id} ({doc.get('type', '')})\n"
1801	        f"- Project/Module: {project}/{module_text}\n"
1802	        f"- Title: {title}\n"
1803	        f"- Current Status: {status}\n"
1804	        f"- Suggested Action: {action_type} ({action_description})\n"
1805	        f"- target_id: {target_id or '-'}\n"
1806	        "- Linked Documents: "
1807	        + (
1808	            ", ".join(f"{d.get('doc_id')}[{d.get('type')}/{d.get('status')}]" for d in linked_docs)
1809	            if linked_docs else
1810	            "none"
1811	        )

1818	        + "- Allowed Files/Restricted Scope: [operator to confirm]\n"
1819	        + "- Request: Please review the draft based on the above information and reply with your work plan."
1820	    )
1821	
1822	    return {
1823	        "doc": {
1824	            "doc_id": doc_id,
1825	            "type": doc.get("type"),
1826	            "title": title,
1827	            "project": project,
1828	            "module": module,
1829	            "status": status,
1830	            "updated_at": doc.get("updated_at"),
1831	            "target_id": target_id,
1832	        },
1833	        "target_document": target_doc,
1834	        "linked_documents": linked_docs,
1835	        "recent_events": recent_events,
1836	        "latest_event": latest_event,
1837	        "suggested_action": {
1838	            "type": action_type,
1839	            "description": action_description,
1840	        },
1841	        "worker_message_template": worker_message,
