65	def _reject_if_group_disposed(doc: dict) -> None:
66	    """Reject a forward modification when the document's group has been disposed (409).
67	
68	    TR0079.0003 rework (2nd pass). The inbox ingestion guard (rev3) made the AI-worker
69	    edit path inert for a discarded group, but a logged-in user editing through the web
70	    UI reaches the document-content / workflow / field-update / transition / conversation
71	    endpoints in THIS router, which only checked is_document_editable (status / final
72	    approval) and never the disposed-group signal. That is the exact rejected symptom —
73	    "documents in a discarded group can still be edited just fine" (test.test.0024.0001-R edits fine after the group's
74	    DC discard). Shares process_service.is_group_disposed with the inbox and workflow
75	    guards (single source of truth: the file-less DC marker), so it fails open for live
76	    groups and on any lookup failure — legitimate work on a live group is never blocked.
77	    """
78	    from modules.flow_gate import process_service as _process_service
79	
80	    if _process_service.is_group_disposed(doc.get("group_id")):
81	        raise HTTPException(
82	            status_code=409,
83	            detail="Modification not allowed: the group has been disposed.",
84	        )

134	def _try_close_parent_on_child_created(
135	    parent_doc_id: str,
136	    actor_user_id: str,
137	) -> None:
138	    """Automatically transition parent document (R/M) from open → closed on child creation (T528).
139	
140	    On transition failure, only log and ignore (T518 pattern).
141	    """
142	    from modules.flow_gate.db import documents as _db_docs
143	    from modules.flow_gate.workflow.pipeline_service import (
144	        transition_document as _pipeline_transition,
145	    )
146	
147	    try:
148	        parent = _db_docs.get_by_id(parent_doc_id)
149	        if parent is None:
150	            return
151	        if parent.get("type_code") not in _PARENT_CLOSE_TYPE_CODES:
152	            return
153	        if parent.get("status") != "open":
154	            return
155	        _pipeline_transition(
156	            doc_id=parent_doc_id,
157	            action="child_created",
158	            actor_user_id=actor_user_id,
159	            user_permissions={"document.update", "document.approve"},
160	        )
161	    except Exception as _e:
162	        _log.warning(
163	            "open→closed (child_created) transition failed for parent %s: %s",
164	            parent_doc_id,
165	            _e,
166	            exc_info=True,
167	        )

178	def _document_file_path(doc: dict) -> Path:
179	    """Resolve a document's stored file_path to a jailed absolute Path.
180	
181	    Delegates to the unified storage.paths.resolve_storage_path() (L0054.0002 §4).
182	    Kept as a thin wrapper so callers retain the HTTPException contract: a missing
183	    column → 404 "missing", an unresolvable/escaping path → 404 "Not found".
184	    """
185	    raw = (doc.get("file_path") or "").strip()
186	    if not raw:
187	        raise HTTPException(status_code=404, detail="Document file path is missing.")
188	
189	    project_id = doc.get("project_id")
190	    branch = (doc.get("branch") or "main") or "main"
191	    resolved = storage_paths.resolve_storage_path(raw, project_id, branch=branch)
192	    if resolved is None:
193	        # Fall back to a best-effort jailed path so existing 404-on-missing-file
194	        # behaviour at call sites (which separately check .is_file()) is preserved.
195	        root = get_storage_root(project_id).resolve()
196	        cand = (Path(raw) if (Path(raw).is_absolute() or _re.match(r"^[A-Za-z]:", raw))
197	                else root / raw).resolve(strict=False)
198	        if not storage_paths._within_allowed_roots(cand, project_id):
199	            raise HTTPException(status_code=404, detail="Not found")
200	        return cand
201	    return resolved

204	def _short_doc_code(doc: dict) -> str:
205	    """Derive the short ``{seq}-{type}`` doc_code from a document's full doc_id.
206	
207	    Mirrors inbox_routes._resolve_storage_path: strip the group_id prefix so the
208	    recomputed canonical filename matches the rest of the storage layout.
209	    """
210	    doc_id = doc.get("doc_id") or ""
211	    group_id = doc.get("group_id") or ""
212	    if group_id and (doc_id.startswith(group_id + ".") or doc_id.startswith(group_id + "-")):
213	        return doc_id[len(group_id) + 1:]
214	    return doc_id

217	def _regenerate_target_path(doc: dict) -> Path:
218	    """Resolve the path a regenerated file should be written to (R0001 / NR0003).
219	
220	    Unlike :func:`_document_file_path`, this never raises and never requires the file
221	    to already exist — the whole point is that the file is *missing*. When ``file_path``
222	    is present it reconstructs the originally-intended absolute location (so recovery
223	    lands where the document expects it); when ``file_path`` is empty or escapes the
224	    storage jail it recomputes the canonical D013 §5 path from DB metadata.
225	    """
226	    raw = (doc.get("file_path") or "").strip()
227	    project_id = doc.get("project_id")
228	    branch = (doc.get("branch") or "main") or "main"
229	    if raw:
230	        resolved = storage_paths.resolve_storage_path(raw, project_id, branch=branch)
231	        if resolved is not None:
232	            return resolved  # file unexpectedly exists / branch-drift hit
233	        root = get_storage_root(project_id).resolve()
234	        cand = (
235	            Path(raw) if (Path(raw).is_absolute() or _re.match(r"^[A-Za-z]:", raw))
236	            else root / raw
237	        ).resolve(strict=False)
238	        if storage_paths._within_allowed_roots(cand, project_id):
239	            return cand
240	        # escaping path → fall through to a recomputed jailed path
241	    filename = "document.md"
242	    if str(doc.get("type_code") or "").upper() == WORK_PLAN_TYPE:
243	        # A work plan's canonical body is JSON (P0009 §2.6 decision 2). Recomputing a `.md`
244	        # name here would recover the document into a file its own reader cannot open.
245	        from modules.flow_gate.services import work_plan_service as _wp
246	
247	        filename = _wp.DOCUMENT_FILENAME
248	    return storage_paths.document_path(
249	        project_id=project_id,
250	        group_code=doc.get("group_id") or "",
251	        doc_code=_short_doc_code(doc),
252	        filename=filename,
253	        module=doc.get("module") or "none",
254	        branch=branch,
255	    )

258	def _latest_revision_body(doc_id: str, project_id: Optional[str]) -> Optional[str]:
259	    """Return the body of the newest readable revision backup, or None (NR0003 §3a).
260	
261	    On every inbox edit the prior file is copied to ``revisions/{doc_id}.r{n}.md`` and
262	    recorded in ``document_revisions``. When the live file is lost we can restore the
263	    last-saved body from the newest backup that still resolves to a real file. Returns
264	    None when there is no backup (e.g. revision_no=0 / pruned by delete_old).
265	    """
266	    from modules.flow_gate.db import document_revisions as _db_rev
267	    try:
268	        revisions = _db_rev.list_by_doc(doc_id)  # newest first (revision_no DESC)
269	    except Exception:
270	        return None
271	    for row in revisions:
272	        backup_path = (row.get("backup_path") or "").strip()
273	        if not backup_path:
274	            continue
275	        resolved = storage_paths.resolve_storage_path(backup_path, project_id)
276	        if resolved is not None and resolved.is_file():
277	            try:
278	                return resolved.read_text(encoding="utf-8")
279	            except OSError:
280	                continue
281	    return None

466	def _parse_doc_workflow(doc: dict) -> dict:
467	    """Parse the workflow_steps JSON string from a documents row into a list.
468	
469	    For non-root documents, find the R/B workflow root in the same group and
470	    attach its workflow metadata for sequence/action-bar display.
471	    """
472	    raw = doc.get("workflow_steps")
473	    parsed: Optional[list] = None
474	    if isinstance(raw, str):
475	        try:
476	            parsed = _json.loads(raw)
477	        except Exception:
478	            parsed = None
479	    elif isinstance(raw, list):
480	        parsed = raw
481	
482	    out: dict = {
483	        **doc,
484	        "workflow_steps": parsed,
485	    }
486	    if doc.get("type_code") in WORKFLOW_ROOT_TYPES:
487	        out["workflow_root_type"] = doc.get("type_code")
488	
489	    # rejection_history: DB TEXT → list
490	    raw_history = doc.get("rejection_history")
491	    if isinstance(raw_history, str):
492	        try:
493	            parsed_history = _json.loads(raw_history)
494	            if not isinstance(parsed_history, list):
495	                parsed_history = []
496	        except (ValueError, TypeError):
497	            parsed_history = []
498	    elif isinstance(raw_history, list):
499	        parsed_history = raw_history
500	    else:
501	        parsed_history = []
502	    out["rejection_history"] = parsed_history
503	
504	    # 0291 T3: read the group's document list **once, first**, and let the R/B root lookup and
505	    # the head decision below share that result. Previously the root was fetched by two narrow
506	    # queries (R, then B) and then the same group's whole document list was read again — even
507	    # though the latter fully contains the former. Up to 3 → 1 per document response.
508	    #
509	    # ``list_documents`` sorts identically for both calls (``ORDER BY updated_at DESC``), so
510	    # picking the first R (or first B) from the list yields the same row as ``type_code=R,
511	    # limit=1`` — **only when the list was not truncated.** Past 100 documents in a group the
512	    # root may lie outside those 100, and it falls back to the old narrow query.
513	    group_head = None
514	    candidates: list = []  # all group docs; also used for 'pending' derivation below
515	    if doc.get("project_id") and doc.get("group_id"):
516	        from modules.flow_gate.db import documents as _db_docs_chain
517	        try:
518	            candidates = _db_docs_chain.list_documents(
519	                project_id=doc["project_id"],
520	                group_id=doc["group_id"],
521	                limit=_GROUP_DOC_SCAN_LIMIT,   # group sizes are tiny; we filter in Python below
522	            )
523	        except Exception:
524	            candidates = []
525	    _candidates_truncated = len(candidates) >= _GROUP_DOC_SCAN_LIMIT
526	
527	    if doc.get("type_code") not in WORKFLOW_ROOT_TYPES and doc.get("group_id") and doc.get("project_id"):
528	        from modules.flow_gate.db import documents as _db_docs
529	        try:
530	            roots = [c for c in candidates if c.get("type_code") == "R"][:1]
531	            if not roots:
532	                roots = [c for c in candidates if c.get("type_code") == "B"][:1]
533	            if not roots and (_candidates_truncated or not candidates):
534	                # The list was truncated or unreadable — absence cannot be concluded, so check.
535	                roots = _db_docs.list_documents(
536	                    project_id=doc["project_id"],
537	                    group_id=doc["group_id"],
538	                    type_code="R",
539	                    limit=1,
540	                ) or _db_docs.list_documents(
541	                    project_id=doc["project_id"],
542	                    group_id=doc["group_id"],
543	                    type_code="B",
544	                    limit=1,
545	                )
546	            if roots:
547	                parent_root = roots[0]
548	                root_doc_id = parent_root.get("doc_id")
549	                out["parent_root_doc_id"] = root_doc_id
550	                out["workflow_root_type"] = parent_root.get("type_code")
551	                # Kept for older clients until the response contract is fully renamed.
552	                out["parent_r_doc_id"] = root_doc_id
553	                if not parsed:
554	                    parent_raw = parent_root.get("workflow_steps")
555	                    if isinstance(parent_raw, str):
556	                        try:
557	                            out["workflow_steps"] = _json.loads(parent_raw)
558	                        except Exception:
559	                            pass
560	                    elif isinstance(parent_raw, list):
561	                        out["workflow_steps"] = parent_raw
562	        except Exception:
563	            pass
564	
565	    # Resolve group head by direct documents-table lookup (PM 4-step spec, T818).
566	    # NOT using workflow_sequence_items.result_doc_id (rejected by PM thrice).
567	    # 0291 T3: the document list was already read above (``candidates``); this only filters it.
568	    #
569	    # The two constants are defined here, not inside the try. They used to sit in the lookup
570	    # block, so if the lookup raised, the head decision below blew up with NameError — a worse
571	    # failure than not reading the list. They are independent of the list and must always exist.
572	    NON_HEAD_TYPES = WORKFLOW_ROOT_TYPES | {"Q"} | AUTO_COMPLETE_TYPES
573	    APPROVED_STATUSES = {"approved", "wf_done"}
574	    # Filter: workflow-step docs only, not yet approved.
575	    # Auto-complete types (memos) are never an actionable head — invariant
576	    # guard so an existing memo can never surface as the head regardless of
577	    # its stored status (B / defence-in-depth alongside the reopen guard).
578	    in_progress = [
579	        c for c in candidates
580	        if c.get("type_code") not in NON_HEAD_TYPES
581	        and c.get("status") != "archived"
582	        and (c.get("doc_review_status") is None
583	             or c.get("doc_review_status") not in APPROVED_STATUSES)
584	    ]
585	    # Pick the earliest-seq in-progress doc (the next step in workflow order).
586	    in_progress.sort(key=lambda c: (c.get("seq") or 0))
587	    group_head = in_progress[0] if in_progress else None
588	
589	    final_approved = (
590	        doc.get("doc_review_status") == "wf_done"
591	        or any(
592	            c.get("type_code") in WORKFLOW_ROOT_TYPES
593	            and c.get("doc_review_status") == "wf_done"
594	            for c in candidates
595	        )
596	    )
597	    out["is_final_approved"] = final_approved
598	    # TR0079.0003 (rework): a disposed group is recorded by a file-less DC document
599	    # (process_service.dispose_group) — the requirement stays wf_in_progress and the
600	    # group status is untouched, so the DC document is the only reliable discard marker
601	    # (same signal the dashboard exclusion uses). Surface it so the client can collapse
602	    # the review/workflow action bar for every document in a discarded group.
603	    out["group_disposed"] = any(
604	        c.get("type_code") == "DC" for c in candidates
605	    )
606	    out["is_editable"] = document_service.is_document_editable(
607	        doc,
608	        final_approved=final_approved,
609	    )
610	
611	    # D030 §3.5: single-shot sequence lookup — reused for 'pending' derivation below
612	    # and workflow_steps / next_step_exists population (T818; no sequence-walk).
613	    root_doc_id_for_head = (
614	        doc["doc_id"]
615	        if doc.get("type_code") in WORKFLOW_ROOT_TYPES
616	        else out.get("parent_root_doc_id") or out.get("parent_r_doc_id")
617	    )
618	    seq_items: list = []
619	    _seq_found = False
620	    if root_doc_id_for_head:
621	        from modules.flow_gate.db import workflow_sequences as _db_wfseq
622	        try:
623	            _seq = _db_wfseq.get_sequence_by_doc_id(root_doc_id_for_head)
624	            if _seq:
625	                seq_items = _db_wfseq.get_sequence_items(_seq["id"]) or []
626	                _seq_found = True
627	        except Exception:
628	            pass
629	
630	    def _effective_head_from_seq_items(items: list[dict]) -> dict | None:
631	        review_by_doc_id = {
632	            c.get("doc_id"): c.get("doc_review_status")
633	            for c in candidates
634	            if c.get("doc_id")
635	        }
636	
637	        # B0001 (group 0105): a *pending* (not-yet-created) auto-approve slot (M / CH)
638	        # IS an actionable "create next document" step — it just auto-approves on
639	        # creation. Excluding it from `pending` made the resolver return None when the
640	        # only remaining slot was a memo/chat, which collapsed the head to the synthetic
641	        # AC gate (action bar showed [final approval] instead of [create document]).
642	        # The canonical SSOT (workflow_sequences.get_effective_head) does NOT type-filter
643	        # the `result_doc_id IS NULL` branch, so the action-bar head diverged from the
644	        # head that create_next_empty / worker inbox actually advance to. Align them:
645	        # keep excluding REALIZED auto-approve docs (T818 invariant — an existing memo
646	        # never re-surfaces as head) but allow PENDING M / CH as head candidates.
647	        pending_non_head = NON_HEAD_TYPES - AUTO_COMPLETE_TYPES  # roots + Q only
648	        linked: list[tuple[dict, int]] = []
649	        pending: list[tuple[dict, int]] = []
650	        for index, item in enumerate(items):
651	            item_type = item.get("type")
652	            result_doc_id = item.get("result_doc_id")
653	            if result_doc_id:
654	                result_review = item.get("result_doc_review_status")
655	                if result_review is None:
656	                    result_review = review_by_doc_id.get(result_doc_id)
657	                if item_type not in NON_HEAD_TYPES and result_review not in APPROVED_STATUSES:
658	                    linked.append((item, index))
659	            elif item_type not in pending_non_head:
660	                pending.append((item, index))
661	
662	        sort_key = lambda pair: (pair[0].get("sort_order", pair[1]), pair[1])
663	        if linked:
664	            linked.sort(key=sort_key)
665	            return linked[0][0]
666	        if pending:
667	            pending.sort(key=sort_key)
668	            return pending[0][0]
669	        return None
670	
671	    # Premature-AC guard: an already-opened AC (final-approval) document lingers
672	    # as an in_progress group doc after new pending steps are inserted before it
673	    # (sequence edit). It must NOT win head resolution while earlier actionable
674	    # slots are still unrealized — otherwise head_index lands at len(steps) and the
675	    # strip paints those empty steps 'done'. When the sequence still has an
676	    # unrealized non-AC slot, defer to the sequence head (resolved in the branch
677	    # below). The legitimate "all steps realized → AC head" case is preserved:
678	    # get_effective_head returns None there, so group_head stays the AC doc.
679	    if (
680	        group_head is not None
681	        and group_head.get("type_code") == "AC"
682	        and _seq_found
683	    ):
684	        _eff = _effective_head_from_seq_items(seq_items)
685	        if _eff and _eff.get("type") not in (None, "AC"):
686	            group_head = None
687	
688	    if group_head is not None:
689	        out["workflow_head_doc_id"]            = group_head["doc_id"]
690	        out["workflow_head_doc_number"]        = group_head["doc_id"]
691	        out["workflow_head_doc_title"]         = group_head.get("title")
692	        out["workflow_head_doc_review_status"] = group_head.get("doc_review_status")
693	        out["workflow_head_type"]              = group_head.get("type_code")
694	        out["workflow_head_status"]            = "in_progress"
695	    elif doc.get("project_id") and doc.get("group_id"):
696	        # 'pending' if sequence has unrealized steps, 'done' if complete.
697	        # Applies to all doc types (NR158: removes T829's R-only restriction).
698	        if seq_items:
699	            # Next pending step from the sequence SLOTS (each has its own identity
700	            # + result_doc_id), via the same SSOT as create's get_effective_head.
701	            # NOT by matching document TYPE — type matching collapses repeated types
702	            # (e.g. M appearing twice) to one and skips the later occurrence, which
703	            # made the head disagree with create_next_empty for such workflows.
704	            eff_head = _effective_head_from_seq_items(seq_items)
705	            head_type = eff_head.get("type") if eff_head else None
706	            if head_type is not None:
707	                out["workflow_head_status"] = "pending"
708	                out["workflow_head_type"] = head_type
709	            else:
710	                # All actionable steps realized. The final approval (AC) is an
711	                # explicit step, not a document: keep the head at AC/pending until
712	                # the R doc is finalized (wf_done), then the workflow is done.
713	                root_done = any(
714	                    c.get("type_code") in WORKFLOW_ROOT_TYPES
715	                    and c.get("doc_review_status") == "wf_done"
716	                    for c in candidates
717	                )
718	                # Mandatory final-approval (AC) gate (M042 / group 0104): every workflow ends
719	                # in an explicit final approval, even memo / auto-approved instruction chains.
720	                # When all document steps are realized + approved but final approval has NOT
721	                # happened yet, keep the head at AC/pending so the action bar surfaces the
722	                # [final approval] control. Only report 'done' once final approval is actually
723	                # performed — the root is finalized (wf_done) or an approved AC document exists.
724	                # (b39f6b8 had collapsed this to always-'done', auto-finalizing the workflow
725	                #  and removing the final-approval action — the regression in group 0104.)
726	                ac_done = any(
727	                    c.get("type_code") == "AC"
728	                    and c.get("status") != "archived"
729	                    and c.get("doc_review_status") in APPROVED_STATUSES
730	                    for c in candidates
731	                )
732	                if root_done or ac_done:
733	                    out["workflow_head_status"] = "done"
734	                else:
735	                    out["workflow_head_type"] = "AC"
736	                    out["workflow_head_status"] = "pending"
737	        else:
738	            # seq_items is empty. Two very different situations collapse here — keep
739	            # them apart (0119 B0001 / NR0009 §6.1):
740	            #  • _seq_found True  → the sequence ROW exists but every item was deleted
741	            #    (a decided workflow emptied of all steps — the B0001 "decided but empty" zombie).
742	            #    This is NOT a finished workflow: the only legitimate terminal state is the
743	            #    AC (final-approval) gate (M042 / group 0104). Reporting 'done' here made the
744	            #    strip paint [done] and let the group chain advance past a broken/empty
745	            #    workflow. Report a distinct 'empty' status so the client routes to recovery
746	            #    ([edit sequence]) instead of [done]/auto-advance.
747	            #  • _seq_found False → no sequence at all (undecided root, or a non-workflow
748	            #    doc with no R parent). Preserve the pre-existing terminal 'done' fallback
749	            #    (NR158 — test_n158_non_r_no_seq_items_fallback_done).
750	            out["workflow_head_status"] = "empty" if _seq_found else "done"
751	
752	    # Populate workflow_steps and next_step_exists from the already-fetched seq_items.
753	    if _seq_found:
754	        if not out.get("workflow_steps") and seq_items:
755	            steps = [item.get("type") for item in seq_items if item.get("type")]
756	            if steps:
757	                out["workflow_steps"] = steps
758	        # Resolve the head's POSITION in the sequence (workflow_head_index) by
759	        # slot IDENTITY, so the strip can colour the correct cell when a type
760	        # repeats (e.g. M twice). Matching by type alone hits the first M and
761	        # mis-colours the strip.
762	        eff_head_type = out.get("workflow_head_type")
763	        head_doc_id = out.get("workflow_head_doc_id")
764	        head_idx = None
765	        if head_doc_id:
766	            # in-progress existing doc → the slot it is registered to
767	            for i, item in enumerate(seq_items):
768	                if item.get("result_doc_id") == head_doc_id:
769	                    head_idx = i
770	                    break
771	        if head_idx is None and eff_head_type:
772	            # pending step → first not-yet-realized slot of this type
773	            for i, item in enumerate(seq_items):
774	                if item.get("type") == eff_head_type and not item.get("result_doc_id"):
775	                    head_idx = i
776	                    break
777	        if eff_head_type == "AC" and head_idx is None:
778	            head_idx = len([it for it in seq_items if it.get("type")])
779	        out["workflow_head_index"] = head_idx
780	        # Resolve the VIEWED doc's own slot index (workflow_self_index) by slot
781	        # identity, so the DocInfoPanel "next" box can find the step that follows
782	        # THIS doc even when its type repeats (e.g. M at slots 1 and 4). A client
783	        # type-based indexOf collapses repeats to the first slot. None when the
784	        # viewed doc is not a realized step slot (R / AC / not yet registered).
785	        self_idx = None
786	        _self_doc_id = doc.get("doc_id")
787	        for i, item in enumerate(seq_items):
788	            if item.get("result_doc_id") == _self_doc_id:
789	                self_idx = i
790	                break
791	        out["workflow_self_index"] = self_idx
792	        # next_step_exists = there IS a head step to advance to — NOT "a step after the
793	        # head". A present head_idx already means an advanceable step (the next doc to
794	        # create/approve, or the synthetic AC final-approval head whose head_idx is
795	        # len(steps)). b39f6b8 changed this to `head_idx < step_count - 1`, which goes
796	        # False whenever the head is the LAST sequence slot — i.e. the last real doc
797	        # before the (synthetic, off-sequence) AC gate. Concretely, group test.test.0007
798	        # has steps [N,NR,T,TR] with TR pending (head_idx=3, step_count=4): 3 < 3 is
799	        # False, so the action bar blanked on every approved doc whose next step is TR
800	        # (next-next = final approval). This is the group 0104 regression the prior fix
801	        # missed — restored to the pre-b39f6b8 semantics.
802	        if eff_head_type == "AC":
803	            out["next_step_exists"] = True
804	        else:
805	            out["next_step_exists"] = head_idx is not None
806	
807	    return out

878	def _slugify_title(text: str) -> str:
879	    text = text.strip().lower()
880	    text = _re.sub(r"[\s\-]+", "_", text)
881	    text = _re.sub(r"[^a-z0-9_]", "", text)
882	    text = _re.sub(r"_+", "_", text).strip("_")
883	    return text[:50] or "untitled"

1906	def restore_workflow(
1907	    body: _RestoreBody,
1908	    current_user: dict = Depends(get_current_user),
1909	) -> dict:
1910	    """Reverse time-machine restore: re-approve unchanged rollback steps."""
1911	    from modules.flow_gate.db import documents as _db_docs
1912	    from modules.flow_gate.db import workflow_return_points as _db_rp
1913	    from modules.flow_gate.db.connection import get_store
1914	    from modules.flow_gate.workflow import event_logger as _event_logger
1915	
1916	    doc = _db_docs.get_by_id(body.doc_id)
1917	    if doc is None:
1918	        raise HTTPException(status_code=404, detail=f"Document not found: {body.doc_id}")
1919	    group_id = doc.get("group_id")
1920	    project_id = doc.get("project_id")
1921	    if not project_id or not group_id:
1922	        raise HTTPException(status_code=400, detail="Document has no project/group")
1923	    _reject_if_group_disposed(doc)
1924	    _reject_if_group_ai_running(doc)
1925	
1926	    with get_store().transaction():
1927	        rp = _db_rp.get_by_group(group_id)
1928	        root_doc = _group_workflow_root_doc(doc)
1929	        root_status = (root_doc or doc).get("doc_review_status") or "wf_in_progress"
1930	        if rp is None:
1931	            return {
1932	                "ok": True,
1933	                "restored": [],
1934	                "stopped_at": None,
1935	                "stopped_doc_id": None,
1936	                "reached_front": False,
1937	                "root_status": root_status,
1938	                "return_point_cleared": False,
1939	            }
1940	
1941	        front_seq = int(rp["front_seq"])
1942	        current_min = _db_rp.current_pending_min_seq(rp["id"])
1943	        lower_bound = current_min if current_min is not None else front_seq
1944	        destination = body.destination_seq if body.destination_seq is not None else front_seq
1945	        destination = max(lower_bound, min(int(destination), front_seq))
1946	
1947	        restored: list[str] = []
1948	        stopped_at: int | None = None
1949	        stopped_doc_id: str | None = None
1950	        for candidate in _db_rp.list_candidates(rp["id"], destination):
1951	            fingerprint = _content_fingerprint(candidate)
1952	            if fingerprint is None or fingerprint != candidate.get("fingerprint"):
1953	                stopped_at = int(candidate.get("seq") or 0)
1954	                stopped_doc_id = candidate.get("doc_id")
1955	                break
1956	            restore_status = candidate.get("prev_status") or "approved"
1957	            _db_docs.update(candidate["doc_id"], {"doc_review_status": restore_status})
1958	            restored.append(candidate["doc_id"])
1959	
1960	        reached_front = stopped_at is None and destination == front_seq
1961	        return_point_cleared = False
1962	        if reached_front:
1963	            # Finalize (restore the root's pre-rewind status, clear the return point) ONLY when the
1964	            # restore actually reconstructed the approved baseline. If any snapshot doc is still
1965	            # pending_review — a pre-existing polluted return point whose prev_status was itself
1966	            # pending — keep the return point and leave the root untouched rather than finalizing over
1967	            # a pending step and laundering the pollution on the next rewind (0142 T0013).
1968	            still_pending = _db_rp.current_pending_min_seq(rp["id"])
1969	            if still_pending is None:
1970	                # Honest re-declaration: walk the root back to whatever it held before the rewind.
1971	                # A genuinely-completed workflow becomes wf_done again; a workflow rewound mid-flight
1972	                # is restored to wf_in_progress instead of being falsely declared done (0158). Legacy
1973	                # return points predating root_prev_status default to wf_done (their old behaviour).
1974	                target_root_status = rp.get("root_prev_status") or "wf_done"
1975	                if root_doc is not None and root_doc.get("doc_review_status") != target_root_status:
1976	                    _db_docs.update(root_doc["doc_id"], {"doc_review_status": target_root_status})
1977	                    root_doc = _db_docs.get_by_id(root_doc["doc_id"])
1978	                _db_rp.delete(rp["id"])
1979	                return_point_cleared = True
1980	
1981	        final_root = root_doc or _group_workflow_root_doc(doc) or doc
1982	        root_status = final_root.get("doc_review_status") or root_status
1983	        try:
1984	            _event_logger.log_event(
1985	                event_type="reverse_time_machine",
1986	                project_id=project_id,
1987	                actor_user_id=current_user["user_id"],
1988	                group_id=group_id,
1989	                document_id=final_root.get("id"),
1990	                metadata={
1991	                    "restored": restored,
1992	                    "stopped_at": stopped_at,
1993	                    "destination_seq": destination,
1994	                    "front_seq": front_seq,
1995	                    "return_point_cleared": return_point_cleared,
1996	                },
1997	            )
1998	        except Exception as exc:  # pragma: no cover - best-effort event trail

2620	def regenerate_document_file(
2621	    doc_id: str,
2622	    current_user: dict = Depends(get_current_user),
2623	) -> dict:
2624	    """Recreate the Markdown file for a document whose stored file is missing (R0001).
2625	
2626	    This is a *recovery* operation, not a content edit, so it deliberately does NOT
2627	    apply the editable/final-approval gate (NR0003 §5) — a missing file must be
2628	    recoverable regardless of workflow state (R0001 itself is ``closed``). Safety
2629	    instead comes from two guards: the document row must exist, and an existing file
2630	    is never clobbered (409).
2631	
2632	    Recovery source priority (NR0003 §3): the newest readable revision backup (restores
2633	    the last-saved body) → else a frontmatter stub synthesized from DB metadata (always
2634	    works, but the body is lost — surfaced as ``body_lost: true``).
2635	    """
2636	    doc = document_service.get_document(doc_id)
2637	    if doc is None:
2638	        raise HTTPException(status_code=404, detail=f"Document not found: {doc_id}")

2640	    _reject_if_group_ai_running(doc)
2641	
2642	    target = _regenerate_target_path(doc)
2643	    if target.is_file():
2644	        # The file is present — there is nothing to recover, and overwriting it would
2645	        # destroy good data. Treat as a no-op conflict (NR0003 §7-3).
2646	        raise HTTPException(
2647	            status_code=409,
2648	            detail="Document file already exists; nothing to regenerate.",
2649	        )
2650	
2651	    project_id = doc.get("project_id")
2652	    body = _latest_revision_body(doc_id, project_id)
2653	    if body is not None:

2655	    elif str(doc.get("type_code") or "").upper() == WORK_PLAN_TYPE:
2656	        # The metadata stub below is Markdown frontmatter, which a work plan reader can
2657	        # only report as "this work plan cannot be opened as a table". Recover a valid but undecided
2658	        # plan instead: every countable type present, every count 0. The response still
2659	        # says body_lost, so nobody mistakes it for the plan that was there.
2660	        from modules.flow_gate.services import work_plan_service as _wp
2661	
2662	        restored_from = "metadata"
2663	        body = _wp.dumps(_wp.empty_recovery_body(project_id))
2664	    else:
2665	        restored_from = "metadata"
2666	        body = _build_next_empty_content(
2667	            project_id=project_id,
2668	            module=doc.get("module") or "none",
2669	            group_id=doc.get("group_id") or "",
2670	            type_code=doc.get("type_code") or "",
2671	            doc_code=_short_doc_code(doc),
2672	            title=doc.get("title") or "",
2673	            target_id=doc.get("target_id") or "",
2674	            next_type="",
2675	        )
2676	
2677	    try:
2678	        target.parent.mkdir(parents=True, exist_ok=True)
2679	        target.write_text(body, encoding="utf-8")
2680	    except OSError as exc:
2681	        raise HTTPException(status_code=500, detail=f"Storage error: {exc}") from exc
2682	
2683	    # L0054.0002: persist the path relative so recovery never re-introduces an
2684	    # absolute path (B0001 regression).

2689	    )

2691	    _broadcast_document_refresh(doc)
