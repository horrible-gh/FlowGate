40	    row = _tpl.get_document_type(type_id)
41	    if not row:
42	        return False

51	_TEMPLATE_STORE_DIR = os.path.join(
52	    os.path.dirname(__file__), "..", "..", "..", "templates", "document_type_templates"
53	)
54	_MAX_TEMPLATE_SIZE = 500 * 1024

58	def list_templates(project_id: str, type_code: str | None = None) -> list[dict]:
59	    store = get_store()
60	    sql = (
61	        "SELECT id, project_id, type_code, is_active, uploaded_by, uploaded_at "
62	        "FROM document_type_templates WHERE project_id = ?"
63	    )
64	    params: list = [project_id]
65	    if type_code:
66	        sql += " AND type_code = ?"
67	        params.append(type_code)
68	    sql += " ORDER BY uploaded_at DESC, id DESC"
69	    rows = store._fetch_all(sql, params)
70	    # E1: held locales per row; template_path is omitted (AC-1 — no path in mgmt UI).
71	    for row in rows:
72	        row["locales"] = _tpl.available_locales(row["id"])
73	    return rows

77	def upload_template(
78	    project_id: str,
79	    type_code: str,
80	    filename: str,
81	    content: bytes,
82	    uploaded_by: str | None = None,
83	) -> dict:
84	    if len(content) > _MAX_TEMPLATE_SIZE:
85	        raise ValueError("Template file size must not exceed 500KB.")
86	
87	    safe_filename = os.path.basename(filename or "template.md")
88	    if not safe_filename.lower().endswith(".md"):
89	        raise ValueError("Template files must use the .md extension.")
90	
91	    os.makedirs(_TEMPLATE_STORE_DIR, exist_ok=True)
92	    safe_name = f"{project_id}_{type_code}_{uuid.uuid4().hex[:8]}_{safe_filename}"
93	    dest = os.path.join(_TEMPLATE_STORE_DIR, safe_name)
94	    with open(dest, "wb") as file_handle:
95	        file_handle.write(content)
96	
97	    store = get_store()
98	    store._execute(
99	        "INSERT INTO document_type_templates (project_id, type_code, template_path, is_active, uploaded_by, uploaded_at)"
100	        " VALUES (?, ?, ?, ?, ?, datetime('now'))"
101	        " ON CONFLICT(project_id, type_code) DO UPDATE SET"
102	        " template_path=excluded.template_path, is_active=excluded.is_active,"
103	        " uploaded_by=excluded.uploaded_by, uploaded_at=excluded.uploaded_at",
104	        [project_id, type_code, dest, 0, uploaded_by],
105	    )
106	    row = store._fetch_one(
107	        "SELECT * FROM document_type_templates WHERE project_id = ? AND type_code = ?",
108	        [project_id, type_code],
109	    )
110	    return row

114	def update_template(project_id: str, template_id: int, data: dict) -> dict | None:
115	    row = _tpl.get_template(template_id)
116	    if not row or row.get("project_id") != project_id:
117	        return None
118	
119	    allowed = {"is_active"}
120	    updates = {k: v for k, v in data.items() if k in allowed}
121	    if "is_active" in updates and updates["is_active"]:
122	        store = get_store()
123	        store._execute(
124	            "UPDATE document_type_templates SET is_active = 0"
125	            " WHERE project_id = ? AND type_code = ? AND id != ?",
126	            [project_id, row["type_code"], template_id],
127	        )
128	    if updates:
129	        store = get_store()
130	        set_clause = ", ".join(f"{k} = ?" for k in updates)
131	        store._execute(
132	            f"UPDATE document_type_templates SET {set_clause} WHERE id = ?",
133	            [*updates.values(), template_id],
134	        )
135	    return _tpl.get_template(template_id)

139	def delete_template(project_id: str, template_id: int) -> bool:
140	    row = _tpl.get_template(template_id)
141	    if not row or row.get("project_id") != project_id:
142	        return False
143	    template_path = row.get("template_path")
144	    _tpl.delete_template(template_id)
145	    if template_path and os.path.exists(template_path):
146	        try:
147	            os.remove(template_path)
148	        except OSError:
149	            pass
150	    return True

186	def register_template_content(
187	    project_id: str, type_code: str, locale: str, content: str, user_id: str | None
188	) -> dict:
189	    """E4 — ensure registry (is_active preserved) + first/replacement body.
190	
191	    Raises TemplateValidationError (422) on invalid locale/content.
192	    Returns the E1-shaped registry summary (id, type_code, is_active, locales, …).
193	    """
194	    _tp.validate_locale(locale)
195	    _tp.validate_content(content)
196	    with get_store().transaction():
197	        template_id = _tpl.ensure_registry(project_id, type_code, user_id)
198	        _tpl.upsert_content(template_id, locale, content, user_id)
199	    return _tpl.registry_with_locales(project_id, type_code)  # type: ignore[return-value]

202	def put_template_content(
203	    project_id: str, template_id: int, locale: str, content: str, user_id: str | None
204	) -> dict | None:
205	    """E5 — register/replace a locale body (no redeploy). None if not owned.
206	
207	    Raises TemplateValidationError (422). Returns {locale, bytes, updated_*, created}.
208	    """
209	    _tp.validate_locale(locale)
210	    _tp.validate_content(content)
211	    if not _tpl.template_owned_by(template_id, project_id):
212	        return None
213	    with get_store().transaction():
214	        created = _tpl.upsert_content(template_id, locale, content, user_id)
215	    row = _tpl.get_content_row(template_id, locale)
216	    return {
217	        "template_id": template_id,
218	        "locale": locale,
219	        "bytes": _tp.bytelen(row["content"]),  # type: ignore[index]
220	        "updated_by": row["updated_by"],  # type: ignore[index]
221	        "updated_at": row["updated_at"],  # type: ignore[index]
222	        "created": created,
223	    }

226	def delete_template_content(
227	    project_id: str, template_id: int, locale: str
228	) -> bool | None:
229	    """E6 — delete a locale body. None if not owned; raises KoFallbackProtected
230	    (409) if deleting the last 'ko' while other locales remain. True if deleted,
231	    False if the locale had no body.
232	    """
233	    if not _tpl.template_owned_by(template_id, project_id):
234	        return None
235	    if locale == _tp.FALLBACK_LOCALE and _tpl.count_other_locales(template_id, locale) > 0:
236	        raise KoFallbackProtected(
237	            "Cannot delete the 'ko' fallback content while other locales exist."
238	        )
239	    with get_store().transaction():
240	        return _tpl.delete_content(template_id, locale)

249	def update_project_settings(project_id: str, data: dict) -> dict:
250	    allowed = {
251	        "group_structure",
252	        "digits_group",
253	        "digits_sub_group",
254	        "digits_type",
255	        "storage_root_override",
256	        "source_mode_override",
257	    }
258	    updates = {k: v for k, v in data.items() if k in allowed}
259	    mode = updates.get("source_mode_override")
260	    if mode is not None and mode not in {"local", "remote"}:
261	        raise ValueError("source_mode_override must be one of: local, remote, null")
262	    current = _proj.get_settings(project_id) or {}
263	    merged = {**current, **updates, "project_id": project_id}
264	    return _proj.upsert_settings(project_id, merged)

268	def get_numbering_impact(project_id: str) -> dict:
269	    """Calculate the number of items affected by reformatting."""
270	    store = get_store()
271	    docs = store._fetch_one("SELECT COUNT(*) as cnt FROM documents WHERE project_id = ?", [project_id])
272	    groups = store._fetch_one("SELECT COUNT(*) as cnt FROM groups WHERE project_id = ?", [project_id])
273	    subs = store._fetch_one(
274	        "SELECT COUNT(*) as cnt FROM sub_groups sg"
275	        " INNER JOIN groups g ON g.group_id = sg.group_id"
276	        " WHERE g.project_id = ?",
277	        [project_id],
278	    )
279	    return {
280	        "documents": docs["cnt"] if docs else 0,
281	        "groups": groups["cnt"] if groups else 0,
282	        "sub_groups": subs["cnt"] if subs else 0,
283	    }

287	def enqueue_numbering_migrate(
288	    project_id: str,
289	    target: str,
290	    from_width: int,
291	    to_width: int,
292	    requested_by: str,
293	) -> dict:
294	    """Enqueue a numbering-digit change job. Raise 409 when another job is already running."""
295	    running = _nj.list_by_project(project_id, status="running")
296	    queued = _nj.list_by_project(project_id, status="queued")
297	    if running or queued:
298	        raise ValueError(f"Project {project_id} already has a numbering job in progress.")
299	    return _nj.create(
300	        {
301	            "project_id": project_id,
302	            "requested_by": requested_by,
303	            "target": target,
304	            "from_width": from_width,
305	            "to_width": to_width,
306	            "status": "queued",
307	        }
308	    )
