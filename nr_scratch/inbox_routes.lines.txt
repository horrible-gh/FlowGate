93	def _truthy(v) -> bool:
94	    """Normalize a dry_run flag defensively (L0007 §2.1).
95	
96	    `bool(body.get("dry_run"))` would treat the string "false" as truthy. JSON booleans
97	    are the normal path, but a client may send a string/int, so accept those explicitly.
98	    None / missing / anything else → False (backward-compatible: existing callers unaffected).
99	    """
100	    if isinstance(v, bool):
101	        return v
102	    if isinstance(v, (int, float)):
103	        return v != 0
104	    if isinstance(v, str):
105	        return v.strip().lower() in ("true", "1", "yes", "on")
106	    return False

125	def _maybe_dry_run(
126	    body: dict, token_rec: dict, would_register: dict
127	) -> Optional[JSONResponse]:
128	    """Shared dry-run short-circuit for all three inbox handlers (L0007 §3, P0006).
129	
130	    Returns:
131	      - None when this is not a dry-run → caller proceeds to the real side-effect path,
132	        byte-for-byte unchanged (backward compatibility).
133	      - HTTP 200 dry-run response (validation passed; every side effect skipped except the
134	        per-token counter +1) when under the limit.
135	      - HTTP 429 when the per-token dry-run limit is already reached (no counter bump).
136	
137	    MUST be invoked only after Step 1~5 validation has fully passed and immediately before
138	    the first side effect, so a dry-run can never reach numbering/storage/DB/consume/SSE.
139	    This ordering is the structural guarantee behind the side-effect-zero invariant (L0007 §7).
140	    Validation *failures* never reach here (handlers already returned _fail), so a failed
141	    dry-run carries the exact same status/message as a real submission and is not counted
142	    (L0007 §5.1).
143	    """
144	    if not _truthy(body.get("dry_run")):
145	        return None
146	
147	    limit = _dryrun_max()
148	    locale = token_rec.get("continuation_locale")
149	    copy = _DRY_RUN_COPY.get(locale) or _DRY_RUN_COPY["ko"]
150	    cnt = int(token_rec.get("dry_run_count") or 0)
151	    if cnt >= limit:
152	        # P0006 §3.5: limit-exceeded body carries the counters alongside the _fail shape.
153	        return JSONResponse(status_code=429, content={
154	            "ok": False,
155	            "http_status": 429,
156	            "error_message": copy["limit"].format(limit=limit),
157	            "help_url": _help_url(),
158	            "dry_run_count": cnt,
159	            "dry_run_remaining": 0,
160	        })
161	
162	    # The only side effect a dry-run is allowed to have (L0007 §5.1 / DB0008 §3).
163	    token_service.increment_dry_run(token_rec["token_id"])
164	    return JSONResponse(status_code=200, content={
165	        "ok": True,
166	        "dry_run": True,
167	        "would_register": would_register,
168	        "dry_run_count": cnt + 1,
169	        "dry_run_remaining": limit - (cnt + 1),
170	        "message": copy["ok"],
171	    })

190	def validate_doc_path(doc_path: str, scratch_dir: str) -> bool:
191	    """Verify that doc_path is within the scratch_dir prefix."""
192	    p = pathlib.Path(doc_path)
193	    if not p.is_absolute():
194	        return False
195	    if str(p).startswith("\\\\"):  # UNC
196	        return False
197	    real = os.path.realpath(doc_path)
198	    if real != os.path.abspath(doc_path):
199	        return False  # symlink
200	    return real.lower().startswith(os.path.realpath(scratch_dir).lower())

271	def _editable_source_root(project_id: str, group_id: Optional[str]) -> tuple[pathlib.Path, Optional[str]]:
272	    from modules.flow_gate.services import git_service
273	
274	    if group_id:
275	        group = db_groups.get_by_id(group_id)
276	        if group is None or group.get("project_id") != project_id:
277	            raise HTTPException(status_code=404, detail="Group not found in this project")
278	        root, reason = git_service.effective_src_root_ex(project_id, group_id)
279	        if root is None:
280	            raise HTTPException(
281	                status_code=409,
282	                detail=f"Group worktree is unavailable ({reason}); base checkout was not used",
283	            )
284	        return root.resolve(), group_id
285	
286	    project = db_projects.get_by_id(project_id)
287	    project_name = (project.get("project_name") or "").strip() if project else ""
288	    settings = db_projects.get_settings(project_id)
289	    branch = (settings.get("branch") or "main").strip() if settings else "main"
290	    if not project_name:
291	        raise HTTPException(status_code=404, detail="Not found")
292	    # Resolve through the storage module at call time so deployments/tests that
293	    # redirect the configured source root keep the same compatibility seam as
294	    # the legacy src-content handler.
295	    from modules.flow_gate.storage import paths as storage_paths
296	    base_branch = git_service.base_branch_for(project_id) or branch
297	    return storage_paths.src_root(project_name, base_branch).resolve(), None

300	def _editable_source_path(
301	    project_id: str, path: str, group_id: Optional[str]
302	) -> tuple[pathlib.Path, pathlib.Path, Optional[str]]:
303	    root, resolved_group_id = _editable_source_root(project_id, group_id)
304	    normalized = path.replace("\\", "/")
305	    parts = normalized.split("/")
306	    if (
307	        not normalized
308	        or normalized.startswith("/")
309	        or (len(normalized) >= 2 and normalized[1] == ":")
310	        or ".." in parts
311	    ):
312	        raise HTTPException(status_code=403, detail="Forbidden")
313	
314	    candidate = root.joinpath(*parts)
315	    current = root
316	    try:
317	        candidate.relative_to(root)
318	        for part in candidate.relative_to(root).parts:
319	            current = current / part
320	            if current.is_symlink():
321	                raise HTTPException(status_code=400, detail="Symbolic links are not allowed")
322	        resolved = candidate.resolve()
323	        resolved.relative_to(root)
324	    except HTTPException:
325	        raise
326	    except (ValueError, OSError):
327	        raise HTTPException(status_code=403, detail="Forbidden")
328	    return resolved, root, resolved_group_id

366	@router.api_route(
367	    "/projects/{project_id}/files/src-content",
368	    methods=["GET", "HEAD"],
369	    response_class=PlainTextResponse,
370	)
371	def get_editable_source_content(
372	    request: Request,
373	    project_id: str,
374	    path: str = Query(..., description="relative path from source root"),
375	    group_id: Optional[str] = Query(None, description="target group worktree"),
376	):
377	    """Read current source content from the base checkout or an exact group worktree."""
378	    full_path, _root, resolved_group_id = _editable_source_path(project_id, path, group_id)
379	    if not full_path.is_file():
380	        raise HTTPException(status_code=404, detail="Not found")
381	    raw = full_path.read_bytes()
382	    headers = {
383	        "Content-Length": str(len(raw)),
384	        "ETag": _source_etag(raw),
385	    }
386	    if resolved_group_id:
387	        headers["X-FlowGate-Group-Id"] = resolved_group_id
388	    if request.method == "HEAD":
389	        return PlainTextResponse("", headers=headers)
390	    return PlainTextResponse(raw.decode("utf-8", errors="replace"), headers=headers)

393	@router.patch("/projects/{project_id}/files/src-content", response_class=JSONResponse)
394	def update_editable_source_content(
395	    request: Request,
396	    project_id: str,
397	    body: EditableSourceUpdate,
398	    path: str = Query(..., description="relative path from source root"),
399	    group_id: Optional[str] = Query(None, description="target group worktree"),
400	):
401	    """Update an existing source file, fail-closed when a group worktree is requested."""
402	    full_path, _root, resolved_group_id = _editable_source_path(project_id, path, group_id)
403	    if not full_path.is_file():
404	        raise HTTPException(status_code=404, detail="Not found")
405	
406	    before = full_path.read_bytes()
407	    if_match = (request.headers.get("if-match") or "").strip()
408	    if if_match and if_match != _source_etag(before):
409	        raise HTTPException(
410	            status_code=409,
411	            detail="File changed since it was opened; reload before saving",
412	        )
413	
414	    encoded = body.content.encode("utf-8")
415	    full_path.write_bytes(encoded)
416	    _emit_source_edit_refresh(project_id, resolved_group_id, path)
417	
418	    response: dict = {
419	        "path": path,
420	        "content_length": len(body.content),
421	        "etag": _source_etag(encoded),
422	    }
423	    if resolved_group_id:
424	        response["group_git"] = _source_group_meta(project_id, resolved_group_id)
425	    else:
426	        from modules.flow_gate.services import git_service
427	        base_git = git_service.base_checkout_dirty_status(project_id)
428	        # Preserve the established src-content response shape. Untracked files
429	        # are refreshed through the project status endpoint, not this existing-file edit.
430	        base_git.pop("untracked", None)
431	        response["base_git"] = base_git
432	    return response

435	def _group_path_is_deleted(project_id: str, group_id: str, path: str) -> bool:
436	    """Best-effort deletion check for stale/forged group blob requests."""
437	    from modules.flow_gate.services import git_service
438	
439	    normalized = path.replace("\\", "/")
440	    try:
441	        payload = git_service.read_group_changes(project_id, group_id)
442	    except Exception:
443	        return False
444	    return any(
445	        change.get("path", "").replace("\\", "/") == normalized
446	        and change.get("status") == "D"
447	        for change in payload.get("data", {}).get("changes", [])
448	    )

454	    group_id: str,
455	    body: DeletedGroupFileRestore,
456	    user=Depends(require_permission("project.settings.edit", "project_id")),
457	):
458	    """Restore one deleted tracked file from the group's current HEAD."""
459	    from modules.flow_gate.services import git_service
460	
461	    normalized = body.path.replace("\\", "/")
462	    full_path, root, _resolved_group_id = _editable_source_path(

466	        raise HTTPException(

468	            detail={

473	
474	    proc = git_service._run_git(
475	        ["checkout", "HEAD", "--", normalized],
476	        cwd=root,
477	    )
478	    if proc.returncode != 0 or not full_path.is_file():
479	        raise HTTPException(
480	            status_code=500,
481	            detail={
482	                "code": "RESTORE_FAILED",
483	                "message": "Failed to restore the deleted file",
484	            },
485	        )
486	
487	    _emit_source_edit_refresh(project_id, group_id, normalized)
488	    return {
489	        "ok": True,
490	        "data": {
491	            "group_id": group_id,
492	            "path": normalized,
493	            "restored": True,
494	        },
495	    }

498	@router.get("/projects/{project_id}/git/groups/{group_id}/blob")
499	def get_editable_group_blob(
500	    project_id: str,
501	    group_id: str,
502	    path: str,
503	    ref: Optional[str] = None,
504	    user=Depends(require_permission("project.settings.read", "project_id")),
505	):
506	    """Return the latest group-worktree bytes so unstaged tracked edits stay visible."""
507	    if ref and not re.fullmatch(r"[0-9a-fA-F]{40}", ref):
508	        raise HTTPException(status_code=400, detail="ref must be a full 40-hex commit sha")
509	    full_path, _root, _resolved_group_id = _editable_source_path(
510	        project_id, path, group_id
511	    )
512	    if not full_path.is_file():
513	        # 0340 T0004 (B0001 / NR0003 §3): a deleted path still exists in the
514	        # branch HEAD tree, so a stale editor tab can ask for it even though the
515	        # group worktree no longer has bytes. Distinguish that intentional deletion
516	        # from an unknown path and never fall back to the committed blob.
517	        if _group_path_is_deleted(project_id, group_id, path):
518	            raise HTTPException(
519	                status_code=410,
520	                detail={"code": "FILE_DELETED", "message": "File was deleted in this group"},
521	            )
522	        raise HTTPException(status_code=404, detail="Not found")
523	
524	    from modules.flow_gate.services import git_service
525	
526	    size = full_path.stat().st_size
527	    with full_path.open("rb") as handle:
528	        head = handle.read(git_service.BLOB_MAX_RETURN_BYTES)
529	    meta = _source_group_meta(project_id, group_id)
530	    if b"\x00" in head[:git_service.BLOB_BINARY_SNIFF_BYTES]:
531	        return {"ok": True, "data": {
532	            **meta,
533	            "path": path,
534	            "size": size,
535	            "binary": True,
536	            "truncated": False,
537	            "encoding": None,
538	            "content": None,
539	            "worktree": True,
540	        }}
541	    return {"ok": True, "data": {
542	        **meta,
543	        "path": path,
544	        "size": size,
545	        "binary": False,
546	        "truncated": size > git_service.BLOB_MAX_RETURN_BYTES,
547	        "encoding": "utf-8",
548	        "content": head.decode("utf-8", errors="replace"),
549	        "worktree": True,
550	    }}

572	def _git_archive_ref_segment(group_id: str) -> str:
573	    # Refs live in the project-wide base checkout, so the terminal number alone
574	    # is not unique (default.0001 and fileforge.0001 would overwrite each other).
575	    # Drop only the project prefix and retain module + group number.
576	    parts = (group_id or "").split(".")
577	    group_segment = "_".join(parts[1:] if len(parts) > 1 else parts)
578	    segment = re.sub(r"[^A-Za-z0-9._-]+", "_", group_segment).strip(".")
579	    while ".." in segment:
580	        segment = segment.replace("..", "._.")
581	    return segment.replace("@{", "_")

607	def _project_git_archives(project_id: str) -> list[dict]:
608	    items: list[dict] = []
609	    for row in db_settings.list_settings():
610	        if not str(row.get("setting_key") or "").startswith(_GIT_ARCHIVE_KEY_PREFIX):
611	            continue
612	        try:
613	            record = json.loads(row.get("setting_value") or "")
614	        except (TypeError, ValueError):
615	            continue
616	        if (
617	            not isinstance(record, dict)
618	            or record.get("project_id") != project_id
619	            or record.get("status") not in _GIT_ARCHIVE_STATUSES
620	        ):
621	            continue
622	        group = db_groups.get_by_id(record.get("group_id") or "")
623	        enriched = dict(record)
624	        enriched["title"] = (group or {}).get("title") or record.get("group_id")
625	        items.append(enriched)
626	    items.sort(key=lambda item: item.get("archived_at") or "", reverse=True)
627	    return items

663	def _git_archive_existing_ref(base_root: pathlib.Path, ref_name: str) -> Optional[str]:
664	    """Return an existing partial-archive ref without treating it as foreign data."""
665	    from modules.flow_gate.services import git_service
666	
667	    proc = git_service._run_git(
668	        ["show-ref", "--verify", "--quiet", ref_name], cwd=base_root
669	    )
670	    if proc.returncode == 1:
671	        return None
672	    if proc.returncode != 0:
673	        _git_archive_error(
674	            500,
675	            "archive_ref_failed",
676	            f"could not validate archive ref '{ref_name}': "
677	            f"{git_service._last_line(proc.stderr)}",
678	        )
679	    resolved = _git_archive_run(
680	        ["rev-parse", "--verify", ref_name],
681	        cwd=base_root,
682	        code="archive_ref_failed",
683	        message=f"could not resolve partial archive ref '{ref_name}'",
684	    )
685	    return (resolved.stdout or "").strip() or None

718	def _archive_group_git(
719	    group_id: str,
720	    reason: Optional[str] = None,
721	    actor_user_id: Optional[str] = None,
722	) -> dict:
723	    """Pin branch/worktree state to named refs, then release and hide the slot."""
724	    from modules.flow_gate.services import git_service
725	
726	    group = db_groups.get_by_id(group_id)
727	    if group is None:
728	        _git_archive_error(404, "not_found", f"group '{group_id}' not found")
729	    project_id = group.get("project_id") or (group_id.split(".", 1)[0] if group_id else "")
730	    cfg = git_service.db_git.get_config(project_id)
731	    if cfg is None or not cfg.get("enabled"):
732	        _git_archive_error(
733	            409, "invalid_state",
734	            f"git integration is not enabled for project '{project_id}'",
735	        )
736	
737	    existing = _git_archive_record(group_id)
738	    state = git_service.db_git.get_state(group_id)
739	    if existing and existing.get("status") == "archived":
740	        if state is not None and state.get("worktree_registered"):
741	            _git_archive_error(
742	                409,
743	                "archive_already_exists",
744	                f"group '{group_id}' already has a completed archive; "
745	                "restore or permanently delete it before archiving new work",
746	            )
747	        return {"ok": True, "result": {**existing, "idempotent": True}}
748	    if existing and existing.get("status") != "archiving":
749	        _git_archive_error(
750	            409,
751	            "invalid_state",
752	            f"archive for '{group_id}' has unsupported status "
753	            f"'{existing.get('status') or 'none'}'",
754	        )
755	
756	    # A fresh request must be a real finalize choice.  An interrupted request
757	    # already has durable refs and resumes below without taking a second stash.
758	    if existing is None:
759	        finalized = _GIT_ARCHIVE_ORIGINAL_GET_FINALIZE(group_id)
760	        current = (finalized.get("state") or {}).get("status")
761	        if current not in ("awaiting_choice", "waiting"):
762	            _git_archive_error(
763	                409, "invalid_state",
764	                f"group '{group_id}' cannot be archived from git state '{current or 'none'}'",
765	            )
766	
767	    holder = f"archive:{uuid.uuid4()}"
768	    if not git_service._acquire_lock(project_id, holder):
769	        _git_archive_error(409, "git_busy", "another Git operation is in progress")
770	    try:
771	        record = _git_archive_record(group_id)
772	        state = git_service.db_git.get_state(group_id)
773	        if record and record.get("status") == "archived":
774	            if state is not None and state.get("worktree_registered"):
775	                _git_archive_error(
776	                    409,
777	                    "archive_already_exists",
778	                    f"group '{group_id}' already has a completed archive; "
779	                    "restore or permanently delete it before archiving new work",
780	                )
781	            return {"ok": True, "result": {**record, "idempotent": True}}
782	        if record and record.get("status") != "archiving":
783	            _git_archive_error(
784	                409,
785	                "invalid_state",
786	                f"archive for '{group_id}' has unsupported status "
787	                f"'{record.get('status') or 'none'}'",
788	            )
789	        if record is None:
790	            if state is None or not state.get("worktree_registered"):
791	                _git_archive_error(
792	                    409, "invalid_state",
793	                    f"Git integration is not active for group '{group_id}'",
794	                )
795	            base_root, branch, head_sha = git_service.resolve_group_ref(project_id, group_id)
796	            worktree_root, root_reason = git_service.effective_src_root_ex(project_id, group_id)
797	            if worktree_root is None:
798	                _git_archive_error(
799	                    409, "invalid_state",
800	                    f"group worktree is unavailable ({root_reason})",
801	                )
802	
803	            ref_segment = _git_archive_ref_segment(group_id)
804	            head_ref = f"refs/flowgate/archive/{ref_segment}/head"
805	            stash_ref = f"refs/flowgate/archive/{ref_segment}/stash"
806	            # A module-qualified ref with no archive record can only be this
807	            # group's interrupted transaction.  Re-pin head to the current
808	            # branch and carry a completed partial stash forward instead of
809	            # permanently blocking the group or deleting preserved bytes.
810	            partial_stash_sha = _git_archive_existing_ref(base_root, stash_ref)
811	            _git_archive_run(
812	                ["update-ref", head_ref, head_sha], cwd=base_root,
813	                code="archive_ref_failed", message="could not preserve the group branch",
814	            )
815	
816	            status_proc = _git_archive_run(
817	                ["status", "--porcelain=v1", "-z"], cwd=worktree_root,
818	                code="archive_status_failed", message="could not inspect the group worktree",
819	            )
820	            changed_entries = [
821	                item for item in (status_proc.stdout or "").split("\0") if item
822	            ]
823	            stash_sha: Optional[str] = partial_stash_sha
824	            if partial_stash_sha and changed_entries:
825	                _git_archive_error(
826	                    409,
827	                    "archive_retry_conflict",
828	                    "a partial archive stash and newer worktree changes both exist; "
829	                    "commit or otherwise clear the newer changes before retrying",
830	                )
831	            if changed_entries:
832	                _git_archive_run(
833	                    [
834	                        "stash", "push", "--include-untracked",
835	                        "--message", f"flowgate archive {group_id}",
836	                    ],
837	                    cwd=worktree_root,
838	                    code="archive_snapshot_failed",
839	                    message="could not preserve uncommitted changes",
840	                )
841	                stash_proc = _git_archive_run(
842	                    ["rev-parse", "--verify", "refs/stash"], cwd=base_root,
843	                    code="archive_snapshot_failed",
844	                    message="could not resolve the preserved changes",
845	                )
846	                stash_sha = (stash_proc.stdout or "").strip()
847	                _git_archive_run(
848	                    ["update-ref", stash_ref, stash_sha], cwd=base_root,
849	                    code="archive_ref_failed", message="could not pin the preserved changes",
850	                )
851	                # The project mutex makes this shared-stack operation exclusive.
852	                _git_archive_run(
853	                    ["stash", "drop", "stash@{0}"], cwd=base_root,
854	                    code="archive_snapshot_failed",
855	                    message="could not release the temporary stash entry",
856	                )
857	
858	            base_branch = (cfg.get("base_branch") or "main").strip() or "main"
859	            base_proc = git_service._run_git(
860	                ["merge-base", base_branch, branch], cwd=base_root
861	            )
862	            base_sha = (
863	                (base_proc.stdout or "").strip() if base_proc.returncode == 0 else None
864	            )
865	            commits_proc = git_service._run_git(
866	                ["rev-list", "--count", f"{base_branch}..{branch}"], cwd=base_root
867	            )
868	            try:
869	                commit_count = int((commits_proc.stdout or "0").strip())
870	            except ValueError:
871	                commit_count = 0
872	
873	            record = {
874	                "status": "archiving",
875	                "project_id": project_id,
876	                "group_id": group_id,
877	                "branch": branch,
878	                "git_status": state.get("status") or "awaiting_choice",
879	                "base_branch": base_branch,
880	                "reason": re.sub(r"\s+", " ", reason or "").strip()[:500] or None,
881	                "actor_user_id": actor_user_id,
882	                "archived_at": now_iso(),
883	                "base_sha": base_sha,
884	                "head_sha": head_sha,
885	                "stash_sha": stash_sha,
886	                "head_ref": head_ref,
887	                "stash_ref": stash_ref if stash_sha else None,
888	                "commit_count": commit_count,
889	                "changed_file_count": len(changed_entries),
890	            }
891	            _save_git_archive_record(record, actor_user_id)
892	
893	        # Existing cleanup handles live, missing, and half-removed worktrees.  It
894	        # is safe to force-discard because the named refs now own every byte.
895	        state = git_service.db_git.get_state(group_id)
896	        if state is not None and state.get("worktree_registered"):
897	            if not git_service._cleanup_group_slot(
898	                project_id, group_id, force_discard=True
899	            ):
900	                _git_archive_error(
901	                    500, "archive_cleanup_failed",
902	                    "the archive refs are safe, but the worktree could not be released; retry",
903	                )
904	
905	        # An archived slot is inactive. Keeping awaiting_choice here made an AC
906	        # approval race a stale git_action into the precheck and fail with 422.
907	        # The pre-archive status is retained in the archive record for restore.
908	        if git_service.db_git.get_state(group_id) is not None:
909	            git_service.db_git.set_status(group_id, "none")
910	
911	        # Do not soft-delete the group row. The document tree fetches project
912	        # documents independently of active groups, so deleting only the group
913	        # turned every retained document into a visible "Uncategorized" orphan.
914	        # Final-approved groups already use the explorer's normal hidden toggle;
915	        # the archive record and inactive Git slot are the archive source of truth.
916	        record["status"] = "archived"
917	        _save_git_archive_record(record, actor_user_id)
918	    finally:
919	        git_service.db_git.release_lock(project_id, holder)
920	
921	    _emit_git_archive_refresh(project_id, group_id, "archived")
922	    return {"ok": True, "result": record}

925	def _restore_group_git_archive(group_id: str, actor_user_id: Optional[str]) -> dict:
926	    from modules.flow_gate.services import git_service
927	
928	    record = _git_archive_record(group_id)
929	    if record is None or record.get("status") not in _GIT_ARCHIVE_STATUSES:
930	        _git_archive_error(404, "archive_not_found", f"archive for '{group_id}' not found")
931	    group = db_groups.get_by_id(group_id)
932	    if group is None:
933	        _git_archive_error(404, "not_found", f"group '{group_id}' not found")
934	    project_id = record.get("project_id") or group.get("project_id")
935	    cfg = git_service.db_git.get_config(project_id)
936	    if cfg is None or not cfg.get("enabled"):
937	        _git_archive_error(409, "invalid_state", "git integration is not enabled")
938	    state = git_service.db_git.get_state(group_id)
939	
940	    holder = f"archive-restore:{uuid.uuid4()}"
941	    if not git_service._acquire_lock(project_id, holder):
942	        _git_archive_error(409, "git_busy", "another Git operation is in progress")
943	    try:
944	        project_name = git_service._project_name(project_id)
945	        if not project_name:
946	            _git_archive_error(404, "not_found", f"project '{project_id}' not found")
947	        base_branch = record.get("base_branch") or cfg.get("base_branch") or "main"
948	        base_root = git_service.src_root(project_name, base_branch)
949	        if not (base_root / ".git").exists():
950	            _git_archive_error(409, "invalid_state", "base checkout is not provisioned")
951	
952	        branch = str(record.get("branch") or "").strip()
953	        head_ref = str(record.get("head_ref") or "").strip()
954	        stash_ref = str(record.get("stash_ref") or "").strip()
955	        if not branch or not head_ref or not git_service._ref_exists(base_root, head_ref):
956	            _git_archive_error(409, "archive_corrupt", "the archived branch ref is missing")
957	        if git_service._ref_exists(base_root, f"refs/heads/{branch}"):
958	            _git_archive_error(409, "restore_conflict", f"local branch '{branch}' already exists")
959	        worktree_root = git_service.src_root(project_name, branch)
960	        if worktree_root.exists():
961	            _git_archive_error(
962	                409, "restore_conflict", f"worktree path already exists: {worktree_root}",
963	            )
964	
965	        _git_archive_run(
966	            ["update-ref", f"refs/heads/{branch}", head_ref], cwd=base_root,
967	            code="restore_failed", message="could not recreate the archived branch",
968	        )
969	        added = git_service._run_git(
970	            ["worktree", "add", str(worktree_root), branch], cwd=base_root
971	        )
972	        if added.returncode != 0:
973	            git_service._run_git(["update-ref", "-d", f"refs/heads/{branch}"], cwd=base_root)
974	            _git_archive_error(
975	                500, "restore_failed",
976	                f"could not recreate the worktree: {git_service._last_line(added.stderr)}",
977	            )
978	
979	        if stash_ref:
980	            applied = git_service._run_git(
981	                ["stash", "apply", "--index", stash_ref], cwd=worktree_root
982	            )
983	            if applied.returncode != 0:
984	                # Roll back only the worktree created from immutable archive refs.
985	                git_service._run_git(["reset", "--hard", "HEAD"], cwd=worktree_root)
986	                git_service._run_git(["clean", "-fd"], cwd=worktree_root)
987	                git_service._run_git(
988	                    ["worktree", "remove", "--force", str(worktree_root)],
989	                    cwd=base_root,
990	                    timeout=git_service.GIT_WORKTREE_RM_TIMEOUT_SEC,
991	                )
992	                git_service._run_git(
993	                    ["update-ref", "-d", f"refs/heads/{branch}"], cwd=base_root
994	                )
995	                _git_archive_error(
996	                    409, "restore_conflict",
997	                    "the archived working changes could not be applied; the archive is intact",
998	                    details={"git": git_service._last_line(applied.stderr)},
999	                )
1000	
1001	        git_service.db_git.register_worktree(group_id, project_id, branch)
1002	        restored_status = (
1003	            record.get("git_status")
1004	            or (state or {}).get("status")
1005	            or "awaiting_choice"
1006	        )
1007	        git_service.db_git.set_status(group_id, restored_status)
1008	        # Backward compatibility for archives created before group soft deletion
1009	        # was removed: restoring an old record also repairs its tree membership.
1010	        db_groups.update(group_id, {"deleted_at": None})
1011	        git_service._run_git(["update-ref", "-d", head_ref], cwd=base_root)
1012	        if stash_ref:
1013	            git_service._run_git(["update-ref", "-d", stash_ref], cwd=base_root)
1014	        db_settings.delete(_git_archive_key(group_id))
1015	    finally:
1016	        git_service.db_git.release_lock(project_id, holder)
1017	
1018	    _emit_git_archive_refresh(project_id, group_id, "restored")
1019	    return {
1020	        "ok": True,
1021	        "result": {
1022	            "group_id": group_id,
1023	            "status": "restored",
1024	            "branch": record.get("branch"),
1025	            "head_sha": record.get("head_sha"),
1026	            "changed_file_count": record.get("changed_file_count", 0),
1027	        },
1028	    }

1249	def _extract_title_from_content(content: Optional[str]) -> Optional[str]:
1250	    """Extract display title from document body.
1251	
1252	    Priority:
1253	      1. `title:` line inside a frontmatter (`---` block), excluding placeholder `<Title here>`
1254	      2. First `# Heading` line in the body
1255	    Returns None if extraction fails.
1256	    """
1257	    if not content:
1258	        return None
1259	    lines = content.splitlines()
1260	    if lines and lines[0].strip() == "---":
1261	        for line in lines[1:]:
1262	            stripped = line.strip()
1263	            if stripped == "---":
1264	                break
1265	            if stripped.lower().startswith("title:"):
1266	                value = stripped[len("title:"):].strip()
1267	                if value and value != "<Title here>":
1268	                    return value
1269	    for line in lines:
1270	        stripped = line.strip()
1271	        if stripped.startswith("# "):
1272	            return stripped[2:].strip()
1273	    return None

1276	# ── Duplicate-body guard (B0106 / NR0003) ──────────────────────────────────────
1277	# The investigation proved the server faithfully writes whatever body the submitter
1278	# sends; the contamination it surfaced — four separate NR documents in different
1279	# groups all carrying 0082's 5 KB report body, each with its own correct title —
1280	# happened in the submission (worker/client) layer, not here. Since that layer is an
1281	# ad-hoc AI worker assembling a POST from scratch files rather than one fixed client,
1282	# the durable defense lives at the gate: refuse to let a *substantial* body land
1283	# byte-identical to an existing document in a *different* group (the contamination
1284	# signature). Short bodies (approval stubs, one-line memos, boilerplate) legitimately
1285	# repeat across groups and are exempt, so the guard fires only on the signature.

1294	    DB schema. The observed contamination was 5397 bytes — far above any reasonable
1295	    threshold — so 1024 leaves a wide margin while exempting approval stubs.
1296	    """
1297	    try:
1298	        return int(os.environ.get("FLOWGATE_INBOX_DUP_MIN_CHARS", _DUP_MIN_CHARS_DEFAULT))
1299	    except (ValueError, TypeError):

1303	def _content_fingerprint(text: Optional[str]) -> Optional[str]:
1304	    """sha256 hex of a substantial body, else None.
1305	
1306	    Returns None for non-str input and for short bodies (< _dup_min_chars stripped
1307	    chars): approval stubs ("Investigation Directive has been approved."), short memos
1308	    and shared boilerplate legitimately repeat across groups and must never trip the
1309	    cross-group duplicate guard. The contamination this catches is a full report copied
1310	    verbatim, always well above the threshold.
1311	    """
1312	    if not isinstance(text, str):
1313	        return None
1314	    if len(text.strip()) < _dup_min_chars():
1315	        return None
1316	    return hashlib.sha256(text.encode("utf-8")).hexdigest()

1319	def _normalize_body(text: str) -> str:
1320	    """Whitespace-insensitive canonical form for near-duplicate detection.
1321	
1322	    Collapses every run of whitespace (spaces, tabs, newlines) to a single space
1323	    and strips the ends. The byte-exact guard (`content_sha256`) misses a clone
1324	    that only differs by reflow / indentation / trailing-newline churn — the
1325	    natural next evasion once exact-match is enforced (NR0003 §5.1c). Normalizing
1326	    away whitespace catches those while staying conservative: it does not touch
1327	    word characters, so two genuinely different reports never collide.
1328	    """
1329	    return re.sub(r"\s+", " ", text).strip()

1332	def _content_fingerprints(text: Optional[str]) -> dict:
1333	    """Both body fingerprints for a substantial body, else an empty dict.
1334	
1335	    Returns ``{"content_sha256": <exact>, "content_sha256_norm": <normalized>}``
1336	    so the duplicate-body guard can match either a byte-exact clone or a
1337	    whitespace-only near-duplicate (NR0003 §5.1c). The two keys are always
1338	    present together or not at all, gated by the same `_dup_min_chars` threshold
1339	    that exempts short approval stubs / boilerplate.
1340	    """
1341	    exact = _content_fingerprint(text)
1342	    if exact is None or not isinstance(text, str):
1343	        return {}
1344	    return {
1345	        "content_sha256": exact,
1346	        "content_sha256_norm": hashlib.sha256(
1347	            _normalize_body(text).encode("utf-8")
1348	        ).hexdigest(),
1349	    }

1391	def _write_submission_body(stored_path: pathlib.Path, body: str) -> None:
1392	    """Persist a submission body verbatim — no newline translation.
1393	
1394	    ``Path.write_text(..., encoding="utf-8")`` opens in text mode, where every
1395	    "\\n" becomes os.linesep on write. On Windows that stores an LF body as CRLF
1396	    and an already-CRLF body as CRCRLF, so the bytes on disk stop being the
1397	    bytes the guards checked and a no-op CRLF resubmission is corrupted. The
1398	    read side (_submission_text) is symmetric: newline="" there too.
1399	    """
1400	    with open(stored_path, "w", encoding="utf-8", newline="") as fh:
1401	        fh.write(body)

1435	def _frontmatter_identity_mismatch(
1436	    text: Optional[str],
1437	    *,
1438	    expected_project: str,
1439	    expected_module: str,
1440	    expected_group_id: str,
1441	    expected_doc_type: str,
1442	    expected_doc_id: Optional[str] = None,
1443	    expected_target_id: Optional[str] = None,
1444	) -> Optional[str]:
1445	    """Return an identity mismatch description for a submitted frontmatter, if any.
1446	
1447	    The duplicate-body guard catches exact body clones. This guard catches the more
1448	    durable contamination signal: a submitted body whose own YAML frontmatter declares
1449	    another document's identity. Missing identity fields are left to existing document
1450	    validation; present conflicting fields are rejected.
1451	    """
1452	    # BOM-aware: `text.lstrip().startswith("---")` is not, because U+FEFF is not
1453	    # whitespace, so a BOM-prefixed foreign frontmatter used to skip this guard
1454	    # entirely while the normalizer happily recognized the very same block.
1455	    if not _linter.looks_like_frontmatter(text):
1456	        return None
1457	    header, parse_error = _linter.parse_yaml_header(text)
1458	    if parse_error or not isinstance(header, dict):
1459	        return None
1460	
1461	    mismatches: list[str] = []
1462	
1463	    def _present(value: object) -> Optional[str]:
1464	        if value is None:
1465	            return None
1466	        s = str(value).strip()
1467	        if not s or s in {"<auto>", "<AUTO>", "<Title here>"}:
1468	            return None
1469	        return s
1470	
1471	    declared_project = _present(header.get("project"))
1472	    if declared_project and declared_project != expected_project:
1473	        mismatches.append(f"project={declared_project!r} expected {expected_project!r}")
1474	
1475	    declared_module = _present(header.get("module"))
1476	    if declared_module and declared_module != expected_module:
1477	        mismatches.append(f"module={declared_module!r} expected {expected_module!r}")
1478	
1479	    for key in ("group_id", "group"):
1480	        declared_group = _present(header.get(key))
1481	        if declared_group and declared_group not in {expected_group_id, _group_short_code(expected_group_id)}:
1482	            mismatches.append(f"{key}={declared_group!r} expected {expected_group_id!r}")
1483	
1484	    declared_type = _present(header.get("type"))
1485	    if declared_type and declared_type != expected_doc_type:
1486	        mismatches.append(f"type={declared_type!r} expected {expected_doc_type!r}")
1487	
1488	    declared_doc_number = _present(header.get("doc_number"))
1489	    if declared_doc_number:
1490	        if expected_doc_id:
1491	            expected_code = _doc_id_suffix(expected_doc_id)
1492	            if declared_doc_number not in _doc_code_alternates(expected_code):
1493	                mismatches.append(
1494	                    f"doc_number={declared_doc_number!r} expected {expected_code!r}"
1495	                )
1496	        else:
1497	            declared_code_type = _doc_code_type(declared_doc_number)
1498	            if declared_code_type and declared_code_type != expected_doc_type:
1499	                mismatches.append(
1500	                    f"doc_number={declared_doc_number!r} has type {declared_code_type!r} "
1501	                    f"expected {expected_doc_type!r}"
1502	                )
1503	
1504	    declared_target_id = _present(header.get("target_id"))
1505	    if declared_target_id and expected_target_id and declared_target_id != expected_target_id:
1506	        # 0226 B0001 / NR0003 §3.1: the mention's §2 header instructs the SHORT form
1507	        # ("target_id: B0001") while expected_target_id is the canonical id, so an
1508	        # unmanned worker that followed the instruction verbatim was structurally
1509	        # 409'd. Accept the same alternate spellings doc_number already gets.
1510	        expected_code = _doc_id_suffix(expected_target_id)
1511	        if declared_target_id not in _doc_code_alternates(expected_code):
1512	            mismatches.append(f"target_id={declared_target_id!r} expected {expected_target_id!r}")
1513	
1514	    return "; ".join(mismatches) if mismatches else None

1541	    text: Optional[str],
1542	    *,
1543	    expected_project: str,
1544	    expected_module: str,
1545	    expected_group_id: str,
1546	    expected_doc_type: str,
1547	    expected_doc_id: Optional[str] = None,
1548	    expected_target_id: Optional[str] = None,
1549	    locale: str = "ko",
1550	) -> Optional[JSONResponse]:
1551	    """422 for an ambiguous/malformed --- frontmatter, 409 for a declared
1552	    identity conflict, else None (nothing to reject).
1553	
1554	    `_frontmatter_identity_mismatch` alone silently returns None (no
1555	    mismatch) both when there is genuinely no conflict AND when the parse
1556	    failed/partially succeeded — the two cases must not be conflated (NR0003:
1557	    a collapsed frontmatter line that starts with an unchecked key, e.g.
1558	    title, currently 200s with no identity fields ever compared). This runs
1559	    `frontmatter_parse_is_ambiguous` first so the ambiguous case is rejected
1560	    explicitly instead of falling through to "no mismatch found".
1561	    """
1562	    # Same BOM-aware test as the identity comparison below (and as the
1563	    # normalizer): all three must agree on what counts as frontmatter, or a
1564	    # BOM-prefixed malformed body walks past whichever one disagrees.
1565	    if not _linter.looks_like_frontmatter(text):
1566	        return None
1567	    if _linter.frontmatter_parse_is_ambiguous(text):
1568	        normalized_locale = template_provision.normalize_locale(locale)

1735	def _encoding_guard(
1736	    *,
1737	    fields: dict[str, Optional[str]],
1738	    fingerprint_field: Optional[str],
1739	    body_sha256: Optional[str],
1740	    body_chars,
1741	    force_encoding_reason: Optional[str],
1742	    locale: str = "ko",
1743	) -> Optional[JSONResponse]:
1744	    reason = (force_encoding_reason or "").strip()
1745	    if len(reason.replace(" ", "")) >= 10:
1746	        return None
1747	
1748	    from modules.flow_gate.services import workflow_decision_service as _wf_decision
1749	
1750	    normalized_locale = template_provision.normalize_locale(locale)
1751	    copy = _ENCODING_GUARD_COPY[normalized_locale]
1752	    labels = _ENCODING_FIELD_LABELS[normalized_locale]
1753	
1754	    check_fields = dict(fields)
1755	    if fingerprint_field and (body_sha256 or body_chars is not None):
1756	        text = check_fields.pop(fingerprint_field, None) or ""
1757	        actual_sha256 = hashlib.sha256(text.encode("utf-8")).hexdigest()
1758	        actual_chars = len(text)
1759	        mismatches = []
1760	        if body_sha256 and str(body_sha256).strip().lower() != actual_sha256:
1761	            mismatches.append(copy["sha256_mismatch"].format(expected=body_sha256, actual=actual_sha256))
1762	        if body_chars is not None:
1763	            try:
1764	                if int(body_chars) != actual_chars:
1765	                    mismatches.append(copy["chars_mismatch"].format(expected=body_chars, actual=actual_chars))
1766	            except (TypeError, ValueError):
1767	                mismatches.append(copy["chars_invalid"].format(value=repr(body_chars)))
1768	        if mismatches:
1769	            return _fail(422, copy["fingerprint_mismatch"].format(mismatches="; ".join(mismatches)))
1770	
1771	    for name, value in check_fields.items():
1772	        if _wf_decision._text_is_corrupted(value):
1773	            display_name = labels.get(name, name)
1774	            return _fail(422, copy["corrupted"].format(field=display_name))
1775	    return None

1965	def _prior_tr_declared(group_id: str, exclude_doc_id: Optional[str] = None) -> list[str]:
1966	    """Return the union of paths already reported by earlier mutating-type docs in this group.
1967	
1968	    0390 TR0005 rev1: this used to hard-compare ``type_code == "TR"``, so a TS
1969	    document's own declared changed-files were invisible to every other document's
1970	    evaluate() call -- the same file could come back flagged "unconfirmed" for a
1971	    sibling TR/TS submission even though a TS already reported it. Widened to the
1972	    same tool_registry.MUTATING_STEP_TYPES membership used everywhere else in this
1973	    gate, so T/TR/TSR/TS all feed and read the same declared-paths pool.
1974	    """
1975	    declared: set[str] = set()
1976	    for document in db_docs.get_documents_by_group_id(group_id):
1977	        if str(document.get("type_code") or "").upper() not in tool_registry.MUTATING_STEP_TYPES:
1978	            continue
1979	        if exclude_doc_id and document.get("doc_id") == exclude_doc_id:
1980	            continue
1981	        verdict = tr_scope_service.verdict_from_meta(document.get("meta"))
1982	        if not verdict:
1983	            continue
1984	        reported = verdict.get("reported")
1985	        if not isinstance(reported, dict):
1986	            continue
1987	        for path in reported.get("items") or []:
1988	            if isinstance(path, str) and path:
1989	                declared.add(path)
1990	    return sorted(declared)

2510	def _normalize_continuation_target(
2511	    target_seq: int,
2512	    instruction_mode: Optional[str],
2513	    completed_item: Optional[dict],
2514	    wfseq,
2515	) -> int:
2516	    """Return the effective continuous boundary for the selected instruction mode.
2517	
2518	    In ``auto_approved`` mode N/T are generated and approved by the server without an AI
2519	    worker. A target on either instruction therefore means its paired AI report (NR/TR);
2520	    otherwise the automatic instruction advance necessarily crosses the displayed boundary
2521	    before the first worker can submit anything. ``ai_direct`` keeps N/T as independent AI
2522	    targets. Any legacy/incomplete sequence data falls back to the original boundary.
2523	    """
2524	    if (instruction_mode or "auto_approved") != "auto_approved":
2525	        return target_seq
2526	    sequence_id = (completed_item or {}).get("sequence_id")
2527	    if sequence_id is None:
2528	        return target_seq
2529	
2530	    try:
2531	        items = wfseq.get_sequence_items(sequence_id)
2532	    except Exception:
2533	        return target_seq
2534	
2535	    target_idx = next(
2536	        (idx for idx, item in enumerate(items) if item.get("item_seq") == target_seq),
2537	        None,
2538	    )
2539	    if target_idx is None:
2540	        return target_seq
2541	
2542	    instruction_type = str(items[target_idx].get("type") or "").upper()
2543	    report_type = {"N": "NR", "T": "TR"}.get(instruction_type)
2544	    if report_type is None or target_idx + 1 >= len(items):
2545	        return target_seq
2546	
2547	    paired = items[target_idx + 1]
2548	    if str(paired.get("type") or "").upper() != report_type:
2549	        return target_seq
2550	    try:
2551	        return int(paired["item_seq"])
2552	    except (KeyError, TypeError, ValueError):
2553	        return target_seq

2799	def _continuation_self_chain(
2800	    request: Request,
2801	    token_rec: dict,
2802	    project: str,
2803	    canonical_doc_id: str,
2804	    doc_type: str,
2805	) -> Optional[dict]:
2806	    """Server-driven self-chaining for continuous (unmanned) work (group 0051 / NR0003 option B).
2807	
2808	    After an inbox `new` submission consumes a *continuation* token, decide what the
2809	    worker does next without a human re-issuing a token:
2810	
2811	      • Ordinary token (no continuation_target_seq) → return None: nothing changes.
2812	      • Review mode (group 0086 TR0004 rework rev4) → PAUSE. Review mode is the pre-flight
2813	        Q-registration phase, not "go": it never auto-approves or advances. The worker
2814	        registers clarifying Qs (or a "no blockers, confirm to proceed" Q) and the run waits
2815	        for the human go (review mode off → non-review auto-run).
2816	      • Non-review (go) → auto-approve the just-submitted document (NR0003 §6-①: the
2817	        deliberate gate relaxation the FE warning dialog made the user accept) so the head
2818	        advances, then, if the target is reached, stop with the last step already approved
2819	        (point 2: up until final approval, the last step must also end up approved);
2820	        otherwise advance_workflow
2821	        mints the next token + continuous mention. Auto-approve uses the issuer's REAL
2822	        document.approve permission (P0005 §4 — approve is never bypassed); if they genuinely
2823	        lack it, pause honestly instead of forcing it.
2824	
2825	    Returns an envelope merged into the inbox 201 response, or None for ordinary tokens.
2826	    Any failure degrades to a paused envelope (the submitted document is already saved;
2827	    only the *continuation* stops), never a 500.
2828	    """
2829	    target_seq = token_rec.get("continuation_target_seq")
2830	    if target_seq is None:
2831	        return None  # ordinary token — not a continuation chain
2832	
2833	    review_mode = bool(token_rec.get("continuation_review_mode"))
2834	    instruction_mode = token_rec.get("continuation_instruction_mode")
2835	    # 0352 T0004 §3.4: db/tokens.py decodes this back to a list on every read, so the
2836	    # consumed token record already carries a plain list (possibly empty) here.
2837	    auto_approve_item_seqs = token_rec.get("continuation_auto_approve_item_seqs") or []
2838	    spine_doc_ref = token_rec.get("doc_ref")
2839	    actor_user_id = token_rec["issued_to"]
2840	    chain_group = token_rec.get("group_id")
2841	
2842	    envelope: dict = {
2843	        "continuation": True,
2844	        "continuation_review_mode": review_mode,
2845	        "continuation_target_seq": target_seq,
2846	        "continuation_instruction_mode": instruction_mode or "auto_approved",
2847	        "continuation_auto_approve_item_seqs": auto_approve_item_seqs,
2848	    }
2849	
2850	    def _stop(stop_code: str, *, detail: Optional[str] = None,
2851	              item_seq: Optional[int] = None) -> dict:
2852	        """Every branch below that does not hand out a token ends through here (§2.12)."""
2853	        from modules.flow_gate.services import ai_invoke_service as _ai_invoke
2854	        return _ai_invoke.stamp_chain_stop(
2855	            envelope,
2856	            stop_code,
2857	            project_id=project,
2858	            group_id=chain_group,
2859	            actor_user_id=actor_user_id,
2860	            anchor_doc_id=canonical_doc_id,
2861	            token_id=token_rec.get("token_id"),
2862	            item_seq=item_seq,
2863	            detail=detail,
2864	        )
2865	
2866	    # Review mode is the PRE-FLIGHT Q-registration phase, not "go" (group 0086 TR0004 rework
2867	    # rev4: review mode = not go yet, it is pre-flight question-registration time). It must
2868	    # NOT auto-approve or advance:
2869	    # the worker registers clarifying Qs (or a "no blockers, confirm to proceed" Q) and the run
2870	    # waits for the human to give the explicit go (turn review mode off → non-review auto-run).
2871	    # The review-phase mention already tells the worker not to create documents, so reaching
2872	    # this self-chain in review mode is unexpected; pause honestly rather than producing work.
2873	    if review_mode:
2874	        envelope["continuation_paused"] = True
2875	        envelope["continuation_reason"] = (
2876	            "review mode (pre-flight Q phase): register clarifying questions; the run advances "
2877	            "only after the human gives the go."
2878	        )
2879	        return _stop("review_hold")
2880	
2881	    from modules.flow_gate.db import workflow_sequences as _wfseq
2882	    completed_item = _wfseq.get_item_by_result_doc_id(canonical_doc_id)
2883	    completed_seq = completed_item.get("item_seq") if completed_item else None
2884	    target_seq = _normalize_continuation_target(
2885	        target_seq,
2886	        instruction_mode,
2887	        completed_item,
2888	        _wfseq,
2889	    )
2890	    envelope["continuation_target_seq"] = target_seq
2891	
2892	    # 0226 B0001 / NR0003 §1.2 (§5-3): a submission that did NOT fill the current head
2893	    # slot (e.g. a doc_type differing from the head) proves no progress toward the
2894	    # target — the old flow skipped the target-reached check below entirely and
2895	    # advanced anyway, minting another token every hop past the target (the actual
2896	    # "endless run"). Pause honestly instead: the stray document stays saved (and
2897	    # unapproved — this runs before the auto-approve) for the human to triage.
2898	    if completed_seq is None:
2899	        envelope["continuation_paused"] = True
2900	        envelope["continuation_reason"] = (
2901	            f"submitted document ({doc_type}) did not fill the current workflow head "
2902	            "slot; progress toward the target cannot be verified, so the chain pauses "
2903	            "instead of advancing."
2904	        )
2905	        return _stop("head_slot_mismatch")
2906	
2907	    # 0317 TR0011 (Q153 opt-1): HOW the chain continues depends on who drives it.
