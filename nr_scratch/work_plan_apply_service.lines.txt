118	def _ordered(items: Iterable[dict]) -> list[dict]:
119	    return sorted((dict(x) for x in (items or [])), key=lambda x: _int(x.get("item_seq")))

122	def _progress(item: dict) -> str:
123	    if item.get("status") == "done":
124	        return "done"
125	    if item.get("status") == "in_progress" or item.get("result_doc_id") is not None:
126	        return "in_progress"
127	    return "pending"

130	# L0010 §2.4
131	def build_workflow_tag(sequence: Optional[dict], items: Iterable[dict]) -> str:
132	    if not sequence:
133	        return "none"
134	    lines = []
135	    ordered = _ordered(items)
136	    for item in ordered:
137	        lines.append(":".join([
138	            str(item.get("item_seq") or ""),
139	            str(item.get("type") or "").upper(),
140	            str(item.get("status") or ""),
141	            str(item.get("sort_order") if item.get("sort_order") is not None else ""),
142	            "1" if item.get("result_doc_id") is not None else "0",
143	            str(item.get("label") or ""),
144	        ]))
145	    canonical = "\n".join([
146	        str(sequence.get("id") or ""),
147	        str(sequence.get("head_advanced_at") or ""),
148	        *lines,
149	    ])
150	    digest = int.from_bytes(hashlib.sha256(canonical.encode("utf-8")).digest()[:8], "big")
151	    return f"seq{sequence.get('id')}-r{digest % WORKFLOW_TAG_DIGEST_MOD}-i{len(ordered)}"

154	def _poured_block(items: Iterable[dict], plan_doc_id: Optional[str]) -> Optional[list[dict]]:
155	    """The rows this very plan put in the sequence, reports included.
156	
157	    0408 M0019 re-rejection 3: L0010 §2.5 counts "the Nth row of type X" over the WHOLE sequence,
158	    which predates the provenance columns 0399 added. A sequence that already finished an
159	    N/NR pair before the plan poured its own therefore matched the plan's N#1 onto the old
160	    finished row, and the plan's mention for that step reached nothing. A row poured by this
161	    plan says so (``source_doc_id``); its automatic report row follows it and says nothing,
162	    so it is taken along. Returns None when this plan has never been poured here — then the
163	    positional rule below is all there is to go on, and it stays exactly as it was.
164	    """
165	    if not plan_doc_id:
166	        return None
167	    report_types = set(AUTO_REPORT_MAP.values())
168	    block: list[dict] = []
169	    inside = False
170	    for item in _ordered(items):
171	        own = str(item.get("source_doc_id") or "") == str(plan_doc_id)
172	        attached = inside and str(item.get("type") or "").upper() in report_types
173	        if own or attached:
174	            block.append(item)
175	            inside = True
176	        else:
177	            inside = False
178	    return block or None

181	def _slot_pool(items: Iterable[dict],
182	               plan_doc_id: Optional[str]) -> tuple[list[dict], list[int]]:
183	    """The rows this plan may fill, and the rows left out of the count (0444 NR0003 §4-1).
184	
185	    The contract, in one sentence:
186	
187	        A sequence cell this plan may fill is either (a) a row that is still ``pending`` or
188	        (b) a row this very plan poured. A row somebody else already started or finished
189	        drops out of the ordinal count and out of the candidate set.
190	
191	    Half (b) is what ``_poured_block`` already does, and it is why re-applying the same plan
192	    still maps its own finished rows to their own places and reports ``already_started``
193	    instead of duplicating them. Half (a) is the fix: the positional fallback used to hand a
194	    new plan's N#1 to a finished N row an earlier plan had left behind, and ``project`` then
195	    classified that row ``already_started`` and threw the provider and the note away while the
196	    genuinely new pending row sat untouched.
197	
198	    Returns (candidate rows, ``item_seq`` of the rows that were skipped). The second element
199	    is what ``done_rows_skipped`` reports, so the skip is never silent.
200	    """
201	    poured = _poured_block(items, plan_doc_id)
202	    if poured is not None:
203	        return poured, []
204	    ordered = _ordered(items)
205	    candidates = [item for item in ordered if _progress(item) == "pending"]
206	    skipped = [_int(item.get("item_seq")) for item in ordered if _progress(item) != "pending"]
207	    return candidates, skipped

210	# L0010 §2.5
211	def build_step_map(plan_steps: Iterable[dict], items: Iterable[dict],
212	                   plan_doc_id: Optional[str] = None) -> list[dict]:
213	    slots: dict[tuple[str, int], dict] = {}
214	    seen: dict[str, int] = {}
215	    pool, _skipped = _slot_pool(items, plan_doc_id)
216	    for item in pool:
217	        code = str(item.get("type") or "").upper()
218	        seen[code] = seen.get(code, 0) + 1
219	        slots[(code, seen[code])] = item
220	    result = []
221	    for step in plan_steps or []:
222	        code, ordinal = str(step.get("type") or "").upper(), _int(step.get("ordinal"))
223	        item = slots.get((code, ordinal))
224	        result.append({
225	            "key": step.get("key"), "type": code, "ordinal": ordinal,
226	            "matched": item is not None,
227	            "item_seq": _int(item.get("item_seq")) if item else None,
228	            "status": _progress(item) if item else "unmatched",
229	            **({"label": item.get("label") or ""} if item else {}),
230	        })
231	    return result

234	def _registry(value: Any) -> dict[str, dict]:
235	    if isinstance(value, dict):
236	        rows = value.get("providers")
237	        if rows is None:
238	            rows = [
239	                ({**v, "id": k} if isinstance(v, dict) else {"id": k, "name": str(v)})
240	                for k, v in value.items()
241	            ]
242	    else:
243	        rows = value or []
244	    return {
245	        str(row.get("id") or row.get("provider_id")): row for row in rows
246	        if row.get("id") or row.get("provider_id")
247	    }

250	def _usable_provider(step: dict, registry: dict[str, dict]) -> Optional[str]:
251	    provider_id = step.get("provider_id")
252	    row = registry.get(str(provider_id)) if provider_id else None
253	    if not row or row.get("enabled", True) is False or row.get("is_enabled", True) is False:
254	        return None
255	    return str(provider_id)

271	# L0010 §2.6 / §4.2
272	def project(plan_steps: Iterable[dict], step_map: Iterable[dict], items: Iterable[dict],
273	            instruction_mode: str, provider_registry: Any) -> dict:
274	    mode = instruction_mode if instruction_mode in INSTRUCTION_MODES else "auto_approved"
275	    rows = list(step_map or [])
276	    mapping = {str(row.get("key")): row for row in rows}
277	    registry = _registry(provider_registry)
278	    own, tucked, folded, unfilled = [], [], [], []
279	    for step in plan_steps or []:
280	        key = str(step.get("key") or "")
281	        mapped = mapping.get(key) or {"matched": False}
282	        if not mapped.get("matched"):
283	            unfilled.append({"key": key, "reason": "unmatched"})
284	            continue
285	        if step.get("locked") or str(step.get("type") or "").upper() == "TSR":
286	            unfilled.append({"key": key, "reason": "locked", "item_seq": mapped.get("item_seq")})
287	            continue
288	        if mapped.get("status") != "pending":
289	            unfilled.append({"key": key, "reason": "already_started", "item_seq": mapped.get("item_seq")})
290	            continue
291	        source_seq = _int(mapped.get("item_seq"))
292	        target_seq, is_folded = source_seq, False
293	        code = str(step.get("type") or "").upper()
294	        if mode == "auto_approved" and code in INSTRUCTION_AUTO_TYPES:
295	            target = _first_pair_after(items, source_seq, AUTO_REPORT_MAP.get(code, ""))
296	            if target:
297	                target_seq, is_folded = _int(target.get("item_seq")), True
298	                to_map = next((row for row in rows if row.get("item_seq") == target_seq), {})
299	                folded.append({
300	                    "from_key": key, "to_key": to_map.get("key"),
301	                    "to_item_seq": target_seq, "reason": "auto_approved_instruction",
302	                })
303	        (tucked if is_folded else own).append((step, target_seq, source_seq))
304	
305	    provider_out: dict[str, str] = {}
306	    note_out: dict[str, str] = {}
307	
308	    def put(step: dict, target_seq: int, absent_only: bool) -> None:
309	        target = str(target_seq)
310	        provider, note = _usable_provider(step, registry), _usable_note(step)
311	        if provider is not None and (not absent_only or target not in provider_out):
312	            provider_out[target] = provider
313	        if note is not None and (not absent_only or target not in note_out):
314	            note_out[target] = note
315	        if provider is None:
316	            reason = "provider_unset" if not step.get("provider_id") else "provider_not_registered"
317	            entry = {"key": step.get("key"), "reason": reason}
318	            if step.get("provider_id"):
319	                entry["provider_id"] = step.get("provider_id")
320	            unfilled.append(entry)
321	        if note is None:
322	            unfilled.append({"key": step.get("key"), "reason": "note_empty"})
323	
324	    for step, target, _source in own:  # own values win
325	        put(step, target, False)
326	    for step, target, source in sorted(tucked, key=lambda row: row[2], reverse=True):
327	        put(step, target, True)  # folded values fill only empty cells
328	    filled = sorted({int(x) for x in provider_out} | {int(x) for x in note_out})
329	    return {
330	        "provider_overrides": provider_out, "note_overrides": note_out,
331	        "filled_item_seqs": filled, "folded": folded, "unfilled": unfilled,
332	    }

335	# L0010 §2.7
336	def suggest_target_seq(plan_steps: Iterable[dict], step_map: Iterable[dict],
337	                       folded: Iterable[dict], provider_registry: Any) -> Optional[int]:
338	    rows = list(step_map or [])
339	    mapping = {str(row.get("key")): row for row in rows}
340	    folded_keys = {str(row.get("from_key")) for row in folded or []}
341	    registry, candidates = _registry(provider_registry), []
342	    for step in plan_steps or []:
343	        key, mapped = str(step.get("key") or ""), mapping.get(str(step.get("key") or ""))
344	        if not mapped or not mapped.get("matched") or mapped.get("status") != "pending":
345	            continue
346	        if step.get("locked") or str(step.get("type") or "").upper() == "TSR" or key in folded_keys:
347	            continue
348	        if step.get("provider_id") and _usable_provider(step, registry) is None:
349	            continue
350	        candidates.append(_int(mapped.get("item_seq")))
351	    if candidates:
352	        return max(candidates)
353	    pending = [_int(row.get("item_seq")) for row in rows
354	               if row.get("matched") and row.get("status") == "pending"]
355	    return max(pending) if pending else None

358	def _warning(code: str, locale: str, keys=(), item_seqs=(), detail=None) -> dict:
359	    keys = list(dict.fromkeys(str(x) for x in keys if x not in (None, "")))
360	    item_seqs = list(dict.fromkeys(_int(x) for x in item_seqs if x is not None))
361	    count = max(len(keys), len(item_seqs), 1 if code in {
362	        "workflow_not_decided", "nothing_to_fill", "wp_not_approved"
363	    } else 0)
364	    lang = locale if locale in _COPY else "ko"
365	    result = {
366	        "code": code, "severity": WARNING_SEVERITY[code], "count": count,
367	        "keys": keys, "item_seqs": item_seqs,
368	        "message": _COPY[lang][code].format(count=count),
369	    }
370	    if detail:
371	        result["detail"] = detail
372	    return result

375	# L0010 §2.8 / §4.3
376	def build_warnings(*, plan_steps: list[dict], step_map: list[dict], provider_registry: Any,
377	                   projection: dict, sequence_decided: bool, added: list[dict],
378	                   extra_item_seqs: list[int], unplaceable_keys: list[str],
379	                   order_differs_keys: list[str], wp_review_status: Optional[str],
380	                   unmatched_keys: list[str], skipped_done_item_seqs: list[int] = (),
381	                   locale: str = "ko") -> list[dict]:
382	    registry, result = _registry(provider_registry), []
383	    if not sequence_decided:
384	        result.append(_warning("workflow_not_decided", locale))
385	    if added:
386	        result.append(_warning("steps_added", locale, [row.get("plan_key") for row in added]))
387	    if extra_item_seqs:
388	        result.append(_warning("extra_workflow_steps", locale, item_seqs=extra_item_seqs))
389	    already = [row.get("item_seq") for row in step_map
390	               if row.get("matched") and row.get("status") != "pending"]
391	    if already:
392	        result.append(_warning("steps_already_done", locale, item_seqs=already))
393	    # A different fact from the line above: those rows are not merely left alone, they were
394	    # dropped from the position count, so this plan's values land on a row further down.
395	    if skipped_done_item_seqs:
396	        result.append(_warning("done_rows_skipped", locale, item_seqs=skipped_done_item_seqs))
397	    if unplaceable_keys:
398	        result.append(_warning("type_not_placeable", locale, unplaceable_keys))
399	    if order_differs_keys:
400	        result.append(_warning("order_differs", locale, order_differs_keys))
401	    unset, unavailable, renamed, empty_notes, locked_values = [], [], [], [], []
402	    for step in plan_steps:
403	        key = str(step.get("key") or "")
404	        if step.get("locked") or str(step.get("type") or "").upper() == "TSR":
405	            if step.get("provider_id") or str(step.get("note") or "").strip():
406	                locked_values.append(key)
407	            continue
408	        provider_id = step.get("provider_id")
409	        if not provider_id:
410	            unset.append(key)
411	        elif _usable_provider(step, registry) is None:
412	            unavailable.append(key)
413	        else:
414	            current = (registry.get(str(provider_id)) or {}).get("name")
415	            if current and step.get("provider_display_name") and current != step.get("provider_display_name"):
416	                renamed.append(key)
417	        if not str(step.get("note") or "").strip():
418	            empty_notes.append(key)
419	    for code, keys in (
420	        ("provider_unset", unset), ("provider_not_registered", unavailable),
421	        ("provider_renamed", renamed), ("note_empty", empty_notes),
422	        ("locked_step_has_value", locked_values),
423	    ):
424	        if keys:
425	            result.append(_warning(code, locale, keys))
426	    if not projection.get("provider_overrides") and not projection.get("note_overrides"):
427	        result.append(_warning("nothing_to_fill", locale))
428	    folded = [row.get("from_key") for row in projection.get("folded") or []]
429	    if folded:
430	        result.append(_warning("instructions_folded", locale, folded))
431	    if wp_review_status != "approved":
432	        result.append(_warning("wp_not_approved", locale))
433	    if unmatched_keys:
434	        result.append(_warning("unmatched_plan_steps", locale, unmatched_keys))
435	    return result

442	# L0010 §4.4
443	def _missing_items(plan_steps: list[dict], items: list[dict], locale: str,
444	                   plan_doc_id: Optional[str] = None) -> tuple[list[dict], list[str]]:
445	    # 0444 NR0003 §4-1: the same rule as _slot_pool, or the two disagree and the fix breaks
446	    # pouring outright. Only rows this plan may fill count towards "the plan already has an
447	    # Nth row of type X" — otherwise a sequence holding nothing but another plan's finished
448	    # rows counts them, proposes no new row, and every plan step stays unmatched forever.
449	    poured = _poured_block(items, plan_doc_id) or []
450	    fillable = {_int(x.get("item_seq")) for x in poured}
451	    counts: dict[str, int] = {}
452	    for item in items:
453	        if _progress(item) != "pending" and _int(item.get("item_seq")) not in fillable:
454	            continue
455	        code = str(item.get("type") or "").upper()
456	        counts[code] = counts.get(code, 0) + 1
457	    # Positions stay whole-sequence: a new row is always appended after everything present.
458	    next_seq = max((_int(x.get("item_seq")) for x in items), default=0)
459	    next_order = max((_int(x.get("sort_order"), -1) for x in items), default=-1) + 1
460	    added, unplaceable = [], []
461	    for step in plan_steps:
462	        code, ordinal = str(step.get("type") or "").upper(), _int(step.get("ordinal"))
463	        if counts.get(code, 0) >= ordinal:
464	            continue
465	        if code not in WORK_PLAN_STEP_TYPES:
466	            unplaceable.append(str(step.get("key") or ""))
467	            continue
468	        # Reuse the workflow-decision draft rule; it is idempotent for an attached report.
469	        draft = expand_steps_with_reports([{"type": code, "label": _label(code, locale)}], locale)
470	        for row in draft:
471	            row_code = str(row.get("type") or "").upper()
472	            if counts.get(row_code, 0) >= ordinal:
473	                continue
474	            next_seq += 1
475	            added.append({
476	                "item_seq": next_seq, "type": row_code,
477	                "label": row.get("label") or _label(row_code, locale),
478	                "status": "pending", "sort_order": next_order, "result_doc_id": None,
479	                "plan_key": f"{row_code}#{ordinal}", "position": next_seq,
480	            })
481	            next_order += 1
482	            counts[row_code] = counts.get(row_code, 0) + 1
483	    return added, unplaceable

486	def _extra_items(plan_steps: list[dict], items: list[dict]) -> list[int]:
487	    wanted = {(str(x.get("type") or "").upper(), _int(x.get("ordinal"))) for x in plan_steps}
488	    seen, extra = {}, []
489	    for item in _ordered(items):
490	        code = str(item.get("type") or "").upper()
491	        seen[code] = seen.get(code, 0) + 1
492	        if code in WORK_PLAN_STEP_TYPES and (code, seen[code]) not in wanted:
493	            extra.append(_int(item.get("item_seq")))
494	    return extra

497	def _order_differs(rows: list[dict]) -> list[str]:
498	    matched = [row for row in rows if row.get("matched")]
499	    seqs = [_int(row.get("item_seq")) for row in matched]
500	    return [] if seqs == sorted(seqs) else [str(row.get("key")) for row in matched]

520	def _preview_apply_blocker(*, sequence_decided: bool, target_seq: Optional[int],
521	                           projection: dict, has_unmatched_steps: bool) -> Optional[str]:
522	    if not sequence_decided:
523	        return "workflow_not_decided"
524	    if target_seq is None:
525	        return "unmatched_plan_steps" if has_unmatched_steps else "no_target"
526	    if not projection.get("provider_overrides") and not projection.get("note_overrides"):
527	        return "nothing_to_fill"
528	    return None

531	def preview(*, doc: dict, plan: dict, providers: Any,
532	            instruction_mode: str, locale: str = "ko") -> dict:
533	    mode = instruction_mode if instruction_mode in INSTRUCTION_MODES else "auto_approved"
534	    owner = doc.get("target_id") or doc.get("triggered_by")
535	    sequence, current = _sequence(owner)
536	    steps = list(plan.get("steps") or [])
537	    added, unplaceable = _missing_items(steps, current, locale, doc.get("doc_id"))
538	    current_mapping = build_step_map(steps, current, doc.get("doc_id"))
539	    current_projection = project(steps, current_mapping, current, mode, providers)
540	    current_target_seq = suggest_target_seq(
541	        steps, current_mapping, current_projection["folded"], providers,
542	    )
543	    keep_blocker = _preview_apply_blocker(
544	        sequence_decided=sequence is not None,
545	        target_seq=current_target_seq,
546	        projection=current_projection,
547	        has_unmatched_steps=bool(added),
548	    )
549	    projected_items = current + added
550	    mapping = build_step_map(steps, projected_items, doc.get("doc_id"))
551	    projection = project(steps, mapping, projected_items, mode, providers)
552	    target_seq = suggest_target_seq(steps, mapping, projection["folded"], providers)
553	    change_blocker = _preview_apply_blocker(
554	        sequence_decided=True,
555	        target_seq=target_seq,
556	        projection=projection,
557	        has_unmatched_steps=False,
558	    )
559	    by_seq = {_int(x.get("item_seq")): x for x in projected_items}
560	    current_max = max((_int(x.get("item_seq")) for x in current), default=0)
561	    preview_map = []
562	    for row in mapping:
563	        view = {**row, "position_after_apply": row.get("item_seq")}
564	        if row.get("matched") and _int(row.get("item_seq")) > current_max:
565	            view.update({"matched": False, "item_seq": None, "status": "to_be_added"})
566	        preview_map.append(view)
567	    warnings = build_warnings(
568	        plan_steps=steps, step_map=mapping, provider_registry=providers,
569	        projection=projection, sequence_decided=sequence is not None, added=added,
570	        extra_item_seqs=_extra_items(steps, current), unplaceable_keys=unplaceable,
571	        order_differs_keys=_order_differs(mapping),
572	        wp_review_status=doc.get("doc_review_status"), unmatched_keys=[],
573	        skipped_done_item_seqs=_slot_pool(projected_items, doc.get("doc_id"))[1],
574	        locale=locale,
575	    )
576	    target = by_seq.get(target_seq or -1)
577	    return {
578	        "ok": True, "wp_doc_id": doc.get("doc_id"),
579	        "wp_revision_no": _int(doc.get("revision_no")),
580	        "wp_review_status": doc.get("doc_review_status"), "instruction_mode": mode,
581	        "workflow": {
582	            "owner_doc_id": owner, "sequence_id": sequence.get("id") if sequence else None,
583	            "decided": sequence is not None, "workflow_tag": build_workflow_tag(sequence, current),
584	            "current_items": current,
585	        },
586	        "comparison": _comparison(steps, current, added), "step_map": preview_map,
587	        "fill_preview": {
588	            "target_seq": target_seq,
589	            "target_key": next((x.get("key") for x in mapping if x.get("item_seq") == target_seq), None),
590	            "target_label": (target or {}).get("label"), **projection,
591	            "default_note": str((plan.get("defaults") or {}).get("note") or "").strip(),
592	        },
593	        "warnings": warnings,
594	        "can_apply": change_blocker is None,
595	        "can_apply_without_workflow": keep_blocker is None,
596	        "can_apply_with_workflow": change_blocker is None,
597	        "can_change_workflow": True,
598	        "apply_blockers": {
599	            "keep_workflow": keep_blocker,
600	            "change_workflow": change_blocker,
601	        },
602	    }

609	def append_application(plan_path: Path, doc_id: str, row: dict) -> None:
610	    path = _applications_path(plan_path, doc_id)
611	    path.parent.mkdir(parents=True, exist_ok=True)

616	def read_applications(plan_path: Path, doc_id: str, limit: int = APPLICATIONS_DEFAULT_LIMIT) -> dict:
617	    limit = max(1, min(_int(limit, APPLICATIONS_DEFAULT_LIMIT), APPLICATIONS_MAX_LIMIT))
618	    path = _applications_path(plan_path, doc_id)
619	    if not path.exists():
620	        return {"ok": True, "total": 0, "items": [], "broken_lines": 0}
621	    try:
622	        lines = path.read_text(encoding="utf-8").splitlines()
623	    except (OSError, UnicodeError):
624	        return {"ok": True, "total": 0, "items": [], "broken_lines": 1}
625	    readable, broken = [], 0
626	    for line in lines:
627	        if not line.strip():
628	            continue
629	        try:
630	            row = json.loads(line)
631	            if isinstance(row, dict):
632	                readable.append(row)
633	            else:
634	                broken += 1
635	        except (json.JSONDecodeError, ValueError):
636	            broken += 1
637	    return {"ok": True, "total": len(readable), "items": list(reversed(readable))[:limit],
638	            "broken_lines": broken}

641	def apply(*, doc: dict, owner_doc: dict, plan: dict, plan_path: Path, providers: Any,
642	          instruction_mode: str, change_workflow: bool, workflow_tag: str,
643	          wp_revision_no: int, applied_by: str, locale: str = "ko") -> dict:
644	    mode = instruction_mode if instruction_mode in INSTRUCTION_MODES else "auto_approved"
645	    owner_id = owner_doc.get("doc_id")
646	    sequence, current = _sequence(owner_id)
647	    before_tag = build_workflow_tag(sequence, current)
648	    if workflow_tag != before_tag:
649	        raise ApplyConflict("workflow_changed", {
650	            "code": "workflow_changed", "sent_workflow_tag": workflow_tag,
651	            "current_workflow_tag": before_tag,
652	        })
653	    current_revision = _int(doc.get("revision_no"))
654	    if _int(wp_revision_no) != current_revision:
655	        raise ApplyConflict("wp_changed", {
656	            "code": "wp_changed", "sent_wp_revision_no": _int(wp_revision_no),
657	            "current_wp_revision_no": current_revision,
658	        })
659	    steps = list(plan.get("steps") or [])
660	    proposed, unplaceable = _missing_items(steps, current, locale, doc.get("doc_id"))
661	    if change_workflow:
662	        for item in proposed:
663	            item["source_doc_id"] = doc.get("doc_id")
664	            item["source_revision_no"] = current_revision
665	    projected_items = list(current) + (list(proposed) if change_workflow else [])
666	    mapping = build_step_map(steps, projected_items, doc.get("doc_id"))
667	    projection = project(steps, mapping, projected_items, mode, providers)
668	    provider_registry = _registry(providers)
669	    added = []
670	    if change_workflow and proposed:
671	        with get_store().transaction():
672	            if sequence is None:
673	                db_wfseq.insert_sequence(owner_id)
674	                sequence = db_wfseq.get_sequence_by_doc_id(owner_id)
675	            for item in proposed:
676	                item_seq = _int(item.get("item_seq"))
677	                provider_id = projection["provider_overrides"].get(str(item_seq))
678	                provider = provider_registry.get(str(provider_id)) if provider_id else None
679	                db_wfseq.insert_sequence_item(
680	                    sequence_id=sequence["id"], item_seq=item_seq,
681	                    type_=str(item.get("type") or ""), label=str(item.get("label") or ""),
682	                    doc_class=str(owner_doc.get("type_code") or "R"),
683	                    sort_order=_int(item.get("sort_order")),
684	                    # 0434 T0004 F3: projection already decided these values. Persist that
685	                    # decision on the same rows instead of reporting a fill that never landed.
686	                    note=projection["note_overrides"].get(str(item_seq), ""),
687	                    provider_id=provider_id,
688	                    provider_display_name=(
689	                        str(provider.get("display_name") or provider.get("name") or provider_id)
690	                        if provider is not None else None
691	                    ),
692	                    source_doc_id=doc.get("doc_id"),
693	                    source_revision_no=current_revision,
694	                )
695	            added = proposed
696	        sequence, current = _sequence(owner_id)
697	    target_seq = suggest_target_seq(steps, mapping, projection["folded"], providers)
698	    unmatched = [str(x.get("key")) for x in mapping if not x.get("matched")] if not change_workflow else []
699	    warnings = build_warnings(
700	        plan_steps=steps, step_map=mapping, provider_registry=providers,
701	        projection=projection, sequence_decided=sequence is not None, added=added,
702	        extra_item_seqs=_extra_items(steps, current), unplaceable_keys=unplaceable,
703	        order_differs_keys=_order_differs(mapping), wp_review_status=doc.get("doc_review_status"),
704	        unmatched_keys=unmatched,
705	        skipped_done_item_seqs=_slot_pool(projected_items, doc.get("doc_id"))[1],
706	        locale=locale,
707	    )
708	    after_tag = build_workflow_tag(sequence, current)
709	    target = next((x for x in current if _int(x.get("item_seq")) == target_seq), {})
710	    applied_at = datetime.now(timezone.utc).astimezone().isoformat(timespec="seconds")
711	    fill = {
712	        "source_doc_id": doc.get("doc_id"), "source_revision_no": current_revision,
713	        "instruction_mode": mode, "target_seq": target_seq,
714	        "target_type": target.get("type"), "target_label": target.get("label"),
715	        "provider_overrides": projection["provider_overrides"],
716	        "note_overrides": projection["note_overrides"],
717	        "default_note": str((plan.get("defaults") or {}).get("note") or "").strip(),
718	        "filled_item_seqs": projection["filled_item_seqs"],
719	        "folded": projection["folded"], "unfilled": projection["unfilled"],
720	    }
721	    history = {
722	        "applied_at": applied_at, "applied_by": applied_by,
723	        "wp_revision_no": current_revision, "instruction_mode": mode,
724	        "workflow_changed": bool(added), "workflow_tag_before": before_tag,
725	        "workflow_tag_after": after_tag,
726	        "added_item_seqs": [_int(x.get("item_seq")) for x in added],
727	        "filled_item_seqs": projection["filled_item_seqs"], "target_seq": target_seq,
728	        "warning_codes": [x["code"] for x in warnings],
729	    }
730	    append_application(plan_path, str(doc.get("doc_id")), history)
731	    return {
732	        "ok": True, "applied_at": applied_at, "applied_by": applied_by,
733	        "wp_doc_id": doc.get("doc_id"), "wp_revision_no": current_revision,
734	        "workflow_changed": bool(added),
735	        "workflow": {"owner_doc_id": owner_id, "sequence_id": sequence.get("id") if sequence else None},
736	        "workflow_tag": after_tag, "added_items": added, "step_map": mapping,
737	        "fill": fill, "warnings": warnings,
738	    }
