109	@router.post("/storage/folder", response_class=JSONResponse)
110	async def api_create_folder(request: Request):
111	    """Create folder. Body: {project_id, parent_path, name, group_id?}
112	
113	    0327 T0004 (B0001): the optional group_id creates inside that group's worktree
114	    so the explorer's "new folder" works while a group branch is selected.
115	    """
116	    body = await request.json()
117	    result = await anyio.to_thread.run_sync(
118	        lambda: process_service.create_storage_folder(
119	            project_id=body.get("project_id", ""),
120	            parent_path=body.get("parent_path", ""),
121	            name=body.get("name", ""),
122	            group_id=body.get("group_id") or None,
123	        )
124	    )
125	    if result.get("status") == "error":
126	        return JSONResponse(status_code=400, content=result)
127	    return JSONResponse(content=result)

200	def _pick_folder_dialog(initial_dir: str = "") -> str:
201	    """Server-side Windows folder selection dialog. Returns selected path or empty string."""
202	    try:
203	        import tkinter as tk
204	        from tkinter import filedialog
205	        root = tk.Tk()
206	        root.withdraw()
207	        root.wm_attributes("-topmost", True)
208	        result = filedialog.askdirectory(
209	            parent=root,
210	            initialdir=initial_dir if (initial_dir and os.path.isdir(initial_dir)) else "/",
211	        )
212	        root.destroy()
213	        return os.path.normpath(result) if result else ""
214	    except Exception:
215	        return ""

218	@router.post("/project-settings/pick-folder", response_class=JSONResponse)
219	async def api_pick_folder(initial_dir: str = Form("")):
220	    """Invoke server-side folder selection dialog."""
221	    loop = asyncio.get_running_loop()
222	    path = await loop.run_in_executor(None, _pick_folder_dialog, initial_dir.strip())
223	    if path:
224	        return JSONResponse(content={"cancelled": False, "path": path})
225	    return JSONResponse(content={"cancelled": True, "path": ""})
