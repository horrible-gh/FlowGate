1	"""Storage/DB migration logic.
2	
3	Move an old-schema DB (`project`, `type`, etc.) and old storage directories
4	to a new-schema DB (`project_id`, `type_code`, etc.) and new storage directories.
5	
6	Used by both the router and the CLI. The CLI entry point is
7	`server/_tools/storage_migrate.py`.
8	
9	Key functions:
10	    migrate(...)        — migrate data (DB INSERT + file copy)
11	    verify(...)         — self-verify migration results (row count, file count, sample hash)
12	    delete_legacy(...)  — permanently delete old DB/files after verification passes
13	    run_full(...)       — run all 3 steps together, including skipping deletion on verification failure
14	"""
15	from __future__ import annotations

37	def _connect(path: str, ro: bool = False) -> sqlite3.Connection:
38	    if ro:
39	        conn = sqlite3.connect(f"file:{path}?mode=ro", uri=True)
40	    else:
41	        conn = sqlite3.connect(path)
42	    conn.row_factory = sqlite3.Row
43	    return conn

46	def _columns(conn: sqlite3.Connection, table: str) -> set[str]:
47	    return {r["name"] for r in conn.execute(f"PRAGMA table_info({table})").fetchall()}

50	def _file_sha256(path: Path) -> str:
51	    h = hashlib.sha256()
52	    with open(path, "rb") as f:
53	        for chunk in iter(lambda: f.read(8192), b""):
54	            h.update(chunk)
55	    return h.hexdigest()

62	def _migrate_groups(old: sqlite3.Connection, new: sqlite3.Connection,
63	                    dry_run: bool) -> dict:
64	    rows = old.execute("SELECT * FROM groups").fetchall()
65	    project_ids = {r["project_id"] for r in new.execute(
66	        "SELECT project_id FROM projects").fetchall()}
67	    new_cols = _columns(new, "groups")
68	    inserted = 0
69	    skipped = 0
70	    unmatched: list[str] = []
71	    for r in rows:
72	        proj = r["project"] if "project" in r.keys() else r["project_id"]
73	        if proj not in project_ids:
74	            unmatched.append(f"{r['group_id']} (project={proj})")
75	            continue
76	        exists = new.execute(
77	            "SELECT 1 FROM groups WHERE group_id = ?", (r["group_id"],)
78	        ).fetchone()
79	        if exists:
80	            skipped += 1
81	            continue
82	        values = {
83	            "group_id": r["group_id"],
84	            "project_id": proj,
85	            "module": r["module"] if "module" in r.keys() else "none",
86	            "parent_id": None,
87	            "title": r["title"],
88	            "priority": r["priority"] if "priority" in r.keys() else None,
89	            "status": (r["status"] or "OPEN") if "status" in r.keys() else "OPEN",
90	            "created_at": r["created_at"],
91	            "updated_at": r["updated_at"],
92	            "closed_at": r["closed_at"] if "closed_at" in r.keys() else None,
93	        }
94	        if values["status"] == "DISCARDED":
95	            values["status"] = "CANCELLED"
96	        cols = [c for c in values.keys() if c in new_cols]
97	        placeholders = ",".join(["?"] * len(cols))
98	        sql = f"INSERT INTO groups ({','.join(cols)}) VALUES ({placeholders})"
99	        if not dry_run:
100	            new.execute(sql, [values[c] for c in cols])
101	        inserted += 1
102	    return {"inserted": inserted, "skipped": skipped, "unmatched": unmatched}

196	def _copy_storage(src_root: str, dst_root: str, dry_run: bool) -> dict:
197	    src = Path(src_root)
198	    dst = Path(dst_root)
199	    if not src.exists():
200	        return {"copied_dirs": 0, "copied_files": 0,
201	                "error": f"src missing: {src}"}
202	    copied_dirs = 0
203	    copied_files = 0
204	    for sub in COPY_SUBDIRS:
205	        s = src / sub
206	        if not s.is_dir():
207	            continue
208	        d = dst / sub
209	        if dry_run:
210	            copied_files += sum(1 for _ in s.rglob("*") if _.is_file())
211	            copied_dirs += 1
212	            continue
213	        d.mkdir(parents=True, exist_ok=True)
214	        for item in s.rglob("*"):
215	            if item.is_file():
216	                rel = item.relative_to(s)
217	                target = d / rel
218	                target.parent.mkdir(parents=True, exist_ok=True)
219	                shutil.copy2(item, target)
220	                copied_files += 1
221	        copied_dirs += 1
222	    return {"copied_dirs": copied_dirs, "copied_files": copied_files}

225	def migrate(src: str, dst: str, db_old: str, db_new: str,
226	            dry_run: bool = False) -> dict:
227	    """Migrate old DB/storage → new DB/storage. Return the result dict."""
228	    if not os.path.exists(db_old):
229	        return {"ok": False, "error": f"Legacy DB not found: {db_old}"}
230	    if not os.path.exists(db_new):
231	        return {"ok": False, "error": f"New DB not found: {db_new}"}
232	
233	    old = _connect(db_old, ro=True)
234	    new = _connect(db_new, ro=False)
235	    try:
236	        if not dry_run:
237	            new.execute("BEGIN")
238	        try:
239	            g = _migrate_groups(old, new, dry_run)
240	            d = _migrate_documents(old, new, dry_run)
241	            e = _migrate_events(old, new, dry_run)
242	            if not dry_run:
243	                new.commit()
244	        except Exception:
245	            if not dry_run:
246	                new.rollback()
247	            raise
248	        f = _copy_storage(src, dst, dry_run)
249	    finally:
250	        old.close()
251	        new.close()
252	
253	    return {
254	        "ok": "error" not in f,
255	        "dry_run": dry_run,
256	        "groups": g,
257	        "documents": d,
258	        "events": e,
259	        "files": f,
260	    }

264	# Verification

267	def verify(src: str, dst: str, db_old: str, db_new: str,
268	           migrate_result: Optional[dict] = None) -> dict:
269	    """Self-verify migration results. Only ok=True is a safe signal to delete legacy data."""
270	    checks: list[dict] = []
271	    failures: list[str] = []
272	
273	    if not os.path.exists(db_old):
274	        return {"ok": False, "checks": [], "failures": [f"Legacy DB not found: {db_old}"]}
275	    if not os.path.exists(db_new):
276	        return {"ok": False, "checks": [], "failures": [f"New DB not found: {db_new}"]}
277	
278	    old = _connect(db_old, ro=True)
279	    new = _connect(db_new, ro=True)
280	    try:
281	        # Compare row counts (excluding unmatched)
282	        for table, mr_key in (("groups", "groups"),
283	                              ("documents", "documents"),
284	                              ("events", "events")):
285	            try:
286	                old_n = old.execute(f"SELECT COUNT(*) FROM {table}").fetchone()[0]
287	            except sqlite3.Error:
288	                old_n = 0
289	            new_n = new.execute(f"SELECT COUNT(*) FROM {table}").fetchone()[0]
290	            unmatched = 0
291	            if migrate_result and mr_key in migrate_result:
292	                unmatched = len(migrate_result[mr_key].get("unmatched", []))
293	            expected_min = old_n - unmatched
294	            ok = new_n >= expected_min
295	            checks.append({
296	                "name": f"row_count[{table}]",
297	                "old": old_n, "new": new_n, "unmatched": unmatched,
298	                "expected_min": expected_min, "ok": ok,
299	            })
300	            if not ok:
301	                failures.append(
302	                    f"{table}: new={new_n} < expected_min={expected_min}")
303	    finally:
304	        old.close()
305	        new.close()
306	
307	    # Verify file hashes
308	    src_p = Path(src)
309	    dst_p = Path(dst)
310	    file_total = 0
311	    file_match = 0
312	    sample_failures: list[str] = []
313	    for sub in COPY_SUBDIRS:
314	        s = src_p / sub
315	        if not s.is_dir():
316	            continue
317	        for item in s.rglob("*"):
318	            if not item.is_file():
319	                continue
320	            file_total += 1
321	            rel = item.relative_to(s)
322	            target = dst_p / sub / rel
323	            if not target.exists():
324	                sample_failures.append(f"missing: {sub}/{rel}")
325	                continue
326	            try:
327	                if _file_sha256(item) == _file_sha256(target):
328	                    file_match += 1
329	                else:
330	                    sample_failures.append(f"hash mismatch: {sub}/{rel}")
331	            except OSError as exc:
332	                sample_failures.append(f"read err {sub}/{rel}: {exc}")
333	    files_ok = (file_total == 0 and not sample_failures) or (
334	        file_match == file_total and not sample_failures)
335	    checks.append({
336	        "name": "files",
337	        "total": file_total, "matched": file_match,
338	        "failures": sample_failures[:10], "ok": files_ok,
339	    })
340	    if not files_ok:
341	        failures.extend(sample_failures[:10])
342	
343	    ok = not failures
344	    return {"ok": ok, "checks": checks, "failures": failures}

347	# ─────────────────────────────────────────────────────────────
348	# Permanently delete old data
349	# ─────────────────────────────────────────────────────────────
350	
351	def delete_legacy(src: str, db_old: str,
352	                  protect_paths: Optional[list[str]] = None) -> dict:
353	    """Call after verification passes. Permanently delete the old DB and old storage
354	    subdirectories (COPY_SUBDIRS).
355	
356	    Never touch paths listed in `protect_paths` (to prevent mistakes).
357	    """
358	    protect = {os.path.normpath(p) for p in (protect_paths or [])}
359	    deleted_files = 0
360	    deleted_dirs = 0
361	    removed_db = None
362	    errors: list[str] = []
363	
364	    src_p = Path(src)
365	    for sub in COPY_SUBDIRS:
366	        s = src_p / sub
367	        if not s.is_dir():
368	            continue
369	        if os.path.normpath(str(s)) in protect:
370	            continue
371	        for item in s.rglob("*"):
372	            if item.is_file():
373	                try:
374	                    item.unlink()
375	                    deleted_files += 1
376	                except OSError as exc:
377	                    errors.append(f"{item}: {exc}")
378	        try:
379	            shutil.rmtree(s)
380	            deleted_dirs += 1
381	        except OSError as exc:
382	            errors.append(f"{s}: {exc}")
383	
384	    if os.path.exists(db_old) and os.path.normpath(db_old) not in protect:
385	        try:
386	            os.remove(db_old)
387	            removed_db = db_old
388	        except OSError as exc:
389	            errors.append(f"{db_old}: {exc}")
390	
391	    return {
392	        "ok": not errors,
393	        "deleted_files": deleted_files,
394	        "deleted_dirs": deleted_dirs,
395	        "removed_db": removed_db,
396	        "errors": errors,
397	    }

400	# ─────────────────────────────────────────────────────────────
401	# Full flow
402	# ─────────────────────────────────────────────────────────────
403	
404	def run_full(src: str, dst: str, db_old: str, db_new: str,
405	             delete_legacy_after: bool = True) -> dict:
406	    """Run migration → verification → legacy deletion on verification success in one pass.
407	
408	    Skip legacy deletion when verification fails. The result dict includes details for each stage.
409	    """
410	    # When the paths are identical (src==dst), file copy is a no-op.
411	    # Legacy deletion must be skipped because the new data is there.
412	    same_root = os.path.normpath(src) == os.path.normpath(dst)
413	
414	    m = migrate(src, dst, db_old, db_new, dry_run=False)
415	    if not m.get("ok"):
416	        return {"ok": False, "stage": "migrate", "migrate": m}
417	
418	    v = verify(src, dst, db_old, db_new, migrate_result=m)
419	    result: dict[str, Any] = {
420	        "ok": v["ok"],
421	        "stage": "verify",
422	        "migrate": m,
423	        "verify": v,
424	    }
425	
426	    if v["ok"] and delete_legacy_after and not same_root:
427	        d = delete_legacy(src, db_old, protect_paths=[dst, db_new])
428	        result["delete"] = d
429	        result["stage"] = "delete"
430	        result["ok"] = d["ok"]
431	    elif v["ok"] and same_root:
432	        result["delete"] = {
433	            "ok": True, "deleted_files": 0, "deleted_dirs": 0,
434	            "removed_db": None, "errors": [],
435	            "note": "src==dst: file deletion skipped (legacy DB preserved)",
436	        }
437	
438	    return result

483	def _cleanup_dst(dst: str) -> None:
484	    """Clean up partially copied target directories on verification failure."""
485	    dst_p = Path(dst)
486	    for sub in COPY_SUBDIRS:
487	        d = dst_p / sub
488	        if d.is_dir():
489	            try:
490	                shutil.rmtree(d)
491	            except OSError:
492	                pass

504	def apply_storage_change(project_id: str, new_root: str) -> dict:
505	    """Change project-settings storage_root_override to new_root, then migrate old
506	    data to the new location → verify → permanently delete old data.
507	
508	    Old root: the current project_settings.storage_root_override or the default
509	    (`<server>/storage`).
510	    If a legacy DB (`<old_root>/flow_gate.db`) exists, migrate both DB + files.
511	    If there is no legacy DB, copy files only → verify → delete the source.
512	    Update DB storage paths only after verification succeeds.
513	    On verification failure, clean the target path and preserve legacy data.
514	    """
515	    from modules.flow_gate.db import projects as _proj
516	
517	    new_root_p = Path(new_root).resolve()
518	    new_root_p.mkdir(parents=True, exist_ok=True)
519	
520	    cur = _proj.get_settings(project_id) or {}
521	    old_root_str = (cur.get("storage_root_override") or "").strip()
522	    if not old_root_str:
523	        old_root_str = str(_server_dir() / "storage")
524	
525	    db_new = str(_server_dir() / "flowgate.db")
526	    db_old = str(Path(old_root_str) / "flow_gate.db")
527	
528	    if Path(old_root_str).resolve() == new_root_p:
529	        # No-op: same path
530	        if cur.get("storage_root_override") != str(new_root_p):
531	            _proj.upsert_settings(
532	                project_id, {**cur, "storage_root_override": str(new_root_p)}
533	            )
534	        return {
535	            "ok": True,
536	            "stage": "noop",
537	            "message": "Same path — migration skipped",
538	            "old_root": old_root_str,
539	            "new_root": str(new_root_p),
540	        }
541	
542	    if os.path.exists(db_old):
543	        # Legacy-schema DB exists → migrate both DB + files
544	        result = run_full(
545	            old_root_str, str(new_root_p), db_old, db_new,
546	            delete_legacy_after=True,
547	        )
548	    else:
549	        # Already using the new schema → copy files only → verify → delete the source
550	        f = _copy_storage(old_root_str, str(new_root_p), dry_run=False)
551	        if "error" in f:
552	            return {
553	                "ok": False,
554	                "stage": "migrate",
555	                "migrate": {"files": f},
556	                "old_root": old_root_str,
557	                "new_root": str(new_root_p),
558	            }
559	        v = _verify_files_only(old_root_str, str(new_root_p))
560	        result = {
561	            "ok": v["ok"],
562	            "stage": "verify",
563	            "migrate": {"files": f},
564	            "verify": v,
565	        }
566	        if v["ok"]:
567	            d = delete_legacy(old_root_str, db_old, protect_paths=[str(new_root_p), db_new])
568	            result["delete"] = d
