1	"""FlowGate DB migration script.
2	
3	Three steps: backup -> transform -> validate.
4	Idempotency guaranteed (safe to run again).
5	"""
6	from __future__ import annotations
7	
8	import shutil
9	import sqlite3
10	import sys
11	from datetime import datetime, timezone, timedelta
12	from pathlib import Path
13	
14	JST = timezone(timedelta(hours=9))
15	
16	_SERVER_DIR = Path(__file__).resolve().parents[3]  # server/
17	_SCHEMA_SQL = _SERVER_DIR / "sql" / "migrations" / "sqlite" / "001_flowgate_schema.sql"

20	def now_iso() -> str:
21	    return datetime.now(JST).isoformat(timespec="seconds")
22	
23	
24	def backup_db(db_path: Path) -> Path | None:
25	    if not db_path.exists():
26	        return None
27	    ts = datetime.now(JST).strftime("%Y%m%d_%H%M%S")
28	    bak = db_path.with_name(f"{db_path.stem}_bak_{ts}{db_path.suffix}")
29	    shutil.copy2(db_path, bak)
30	    print(f"[backup] {db_path} → {bak}")
31	    return bak

34	def get_existing_tables(conn: sqlite3.Connection) -> set[str]:
35	    rows = conn.execute(
36	        "SELECT name FROM sqlite_master WHERE type='table'"
37	    ).fetchall()
38	    return {r[0] for r in rows}
39

41	def apply_schema(conn: sqlite3.Connection) -> None:
42	    sql = _SCHEMA_SQL.read_text(encoding="utf-8")
43	    conn.executescript(sql)
44	    print("[schema] Schema applied")

47	def migrate_legacy_data(
48	    legacy_conn: sqlite3.Connection, new_conn: sqlite3.Connection
49	) -> None:
50	    """Attempt to preserve data from the legacy DB in the new schema."""
51	    legacy_tables = get_existing_tables(legacy_conn)
52	    now = now_iso()
53	
54	    # documents migration
55	    if "documents" in legacy_tables:
56	        rows = legacy_conn.execute("SELECT * FROM documents").fetchall()
57	        cols_info = legacy_conn.execute("PRAGMA table_info(documents)").fetchall()
58	        cols = [c[1] for c in cols_info]
59	        migrated = 0
60	        for row in rows:
61	            d = dict(zip(cols, row))
62	            try:
63	                project_id = d.get("project") or "LEGACY"
64	                new_conn.execute(
65	                    "INSERT OR IGNORE INTO projects "
66	                    "(project_id, project_name, is_active, created_at, updated_at) "
67	                    "VALUES (?, ?, 1, ?, ?)",
68	                    [project_id, project_id, now, now],
69	                )
70	                new_conn.execute(
71	                    "INSERT OR IGNORE INTO documents "
72	                    "(doc_id, project_id, module, type_code, seq, title, status, "
73	                    "owner_id, priority, due_date, direction, review_required, "
74	                    "tv_type, pass_criteria, worker_tier, target_id, triggered_by, "
75	                    "superseded_by, previous_tv, previous_t, previous_ds, created_at, updated_at) "
76	                    "VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)",
77	                    [
78	                        d.get("doc_id"), project_id,
79	                        d.get("module") or "none",
80	                        d.get("type") or "R",
81	                        d.get("seq_num") or 0,
82	                        d.get("title") or "(migration)",
83	                        d.get("status") or "draft",
84	                        d.get("owner"), d.get("priority"), d.get("due_date"),
85	                        d.get("direction"), d.get("review_required") or 0,
86	                        d.get("tv_type"), d.get("pass_criteria") or "all",
87	                        d.get("worker_tier"), d.get("target_id"),
88	                        d.get("triggered_by"), d.get("superseded_by"),
89	                        d.get("previous_tv"), d.get("previous_t"), d.get("previous_ds"),
90	                        d.get("created_at") or now, d.get("updated_at") or now,
91	                    ],
92	                )
93	                migrated += 1
94	            except Exception as e:
95	                print(f"[warn] documents migration failed: {e}", file=sys.stderr)
96	        print(f"[migrate] documents: {migrated}/{len(rows)} rows migrated")
97	
98	    # events migration
99	    if "events" in legacy_tables:
100	        rows = legacy_conn.execute("SELECT * FROM events").fetchall()
101	        cols_info = legacy_conn.execute("PRAGMA table_info(events)").fetchall()
102	        cols = [c[1] for c in cols_info]
103	        migrated = 0
104	        for row in rows:

108	                    "INSERT OR IGNORE INTO events "
109	                    "(doc_id, event_type, memo_file, file_hash, reason, "
110	                    "related_doc_id, related_target_id, note, created_at) "
111	                    "VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?)",
112	                    [
113	                        d.get("doc_id"), d.get("event_type") or "legacy",
114	                        d.get("memo_file"), d.get("file_hash"), d.get("reason"),
115	                        d.get("related_doc_id"), d.get("related_target_id"),
116	                        d.get("note"), d.get("created_at") or now,
117	                    ],
118	                )
119	                migrated += 1
120	            except Exception as e:
121	                print(f"[warn] events migration failed: {e}", file=sys.stderr)
122	        print(f"[migrate] events: {migrated}/{len(rows)} rows migrated")
123	
124	    # tv_scenarios migration
125	    if "tv_scenarios" in legacy_tables:
126	        rows = legacy_conn.execute("SELECT * FROM tv_scenarios").fetchall()
127	        cols_info = legacy_conn.execute("PRAGMA table_info(tv_scenarios)").fetchall()
128	        cols = [c[1] for c in cols_info]
129	        migrated = 0
130	        for row in rows:
131	            d = dict(zip(cols, row))
132	            try:
133	                new_conn.execute(
134	                    "INSERT OR IGNORE INTO tv_scenarios "
135	                    "(tv_doc_id, scenario_idx, source, title, result, note, "
136	                    "disabled, disabled_reason, updated_at) "
137	                    "VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?)",
138	                    [
139	                        d.get("tv_doc_id"), d.get("scenario_idx") or 0,
140	                        d.get("source") or "worker",
141	                        d.get("title") or "(migration)",
142	                        d.get("result"), d.get("note"),

154	        rows = legacy_conn.execute("SELECT * FROM groups").fetchall()
155	        cols_info = legacy_conn.execute("PRAGMA table_info(groups)").fetchall()
156	        cols = [c[1] for c in cols_info]

167	                )

178	                        d.get("title") or "(migration)",

181	                        d.get("closed_at"),

183	                )
184	                migrated += 1

190	def verify(conn: sqlite3.Connection) -> bool:
191	    required_tables = {
192	        "users", "roles", "permissions", "role_permissions",
193	        "projects", "project_settings", "groups", "sub_groups",
194	        "user_project_roles", "documents", "events", "tv_scenarios",
195	        "document_types", "document_type_templates", "id_counter",
196	        "token_blacklist", "refresh_tokens", "workflow_events",
197	        "numbering_jobs", "system_settings",
198	    }
199	    existing = get_existing_tables(conn)
200	    missing = required_tables - existing
201	    if missing:
202	        print(f"[verify] ❌ Missing tables: {missing}", file=sys.stderr)
203	        return False
204	    role_count = conn.execute(
205	        "SELECT COUNT(*) FROM roles WHERE is_system=1"
206	    ).fetchone()[0]
207	    perm_count = conn.execute("SELECT COUNT(*) FROM permissions").fetchone()[0]
208	    dt_count = conn.execute(
209	        "SELECT COUNT(*) FROM document_types WHERE project_id IS NULL"
210	    ).fetchone()[0]
211	    print(
212	        f"[verify] {len(existing)} tables ✓, "
213	        f"system roles {role_count}, permissions {perm_count}, doc types {dt_count}"
214	    )
215	    return True

218	def run(db_path: str | None = None, legacy_db_path: str | None = None) -> None:
219	    """
220	    db_path: path to the new / migration target DB
221	    legacy_db_path: path to the legacy DB (if missing, only create a new DB)
222	    """
223	    if db_path is None:
224	        server_dir = Path(__file__).resolve().parents[3]
225	        db_path_obj = server_dir / "storage" / "flow_gate_new.db"
226	    else:
227	        db_path_obj = Path(db_path)
228	
229	    db_path_obj.parent.mkdir(parents=True, exist_ok=True)
230	
231	    # 1. Backup
232	    backup_db(db_path_obj)
233	
234	    # 2. Apply the new schema
235	    new_conn = sqlite3.connect(str(db_path_obj))
236	    new_conn.execute("PRAGMA foreign_keys = ON")
237	    new_conn.execute("PRAGMA journal_mode = WAL")
238	    try:
239	        apply_schema(new_conn)
240	
241	        # 3. Migrate legacy data
242	        if legacy_db_path and Path(legacy_db_path).exists():
243	            legacy_conn = sqlite3.connect(legacy_db_path)
244	            try:
245	                migrate_legacy_data(legacy_conn, new_conn)
246	            except Exception as e:
247	                new_conn.rollback()
248	                print(f"[error] Migration failed, rollback: {e}", file=sys.stderr)
249	                raise
250	            finally:
251	                legacy_conn.close()
252	
253	        new_conn.commit()
254	
255	        # 4. Validation
256	        if not verify(new_conn):
257	            raise RuntimeError("Verification failed")
258	        print("[done] Migration complete")
259	    except Exception:
260	        new_conn.rollback()
261	        raise
262	    finally:
263	        new_conn.close()

269	    parser = argparse.ArgumentParser(description="FlowGate DB migration")
