2	
3	flowgate.default.0060 L0012 §2-2 (S1~S9), §2-3 (E1~E4, U1~U3), §2-4 (D1~D6), §2-7.

5	What this replaces: ``documents.py`` used to sanitize with a single ``Path(name).name``.
6	On Linux that lets ``..\\..\\etc\\passwd`` through as ONE filename component — the exact
7	defect ``tree_routes.py:151`` already had to fix by normalizing a backslash to ``/`` first.

31	_CONTROL_CHARS = {chr(c) for c in range(0x00, 0x20)} | {chr(0x7F)}
32	# Bidi overrides: a right-to-left mark can make `photo_gnp.exe` render as `photo_exe.png`.
33	_BIDI_CONTROLS = {chr(c) for c in range(0x202A, 0x202F)} | {chr(c) for c in range(0x2066, 0x206A)}
34	_WINDOWS_ILLEGAL = ('<', '>', ':', '"', '|', '?', '*')

37	def _truncate_utf8(text: str, max_bytes: int) -> str:
38	    """Cut *text* to at most *max_bytes* UTF-8 bytes, on a character boundary.
39	
40	    L0012 §2-2 note 4: cutting on a raw byte index splits a Hangul syllable in half and the
41	    name on disk stops matching the name in the registry.
42	    """
43	    if max_bytes <= 0:
44	        return ""
45	    encoded = text.encode("utf-8")
46	    if len(encoded) <= max_bytes:
47	        return text
48	    return encoded[:max_bytes].decode("utf-8", errors="ignore")

51	def _strip_controls(name: str) -> str:
52	    return "".join(ch for ch in name if ch not in _CONTROL_CHARS and ch not in _BIDI_CONTROLS)

55	def _apply_s3_to_s7(name: str) -> str:
56	    """S3~S7 — the part shared by ``sanitize_attachment_name`` and ``sanitize_path_component``."""
57	    # S3 — backslash first, then keep only the last path component.
58	    name = name.replace("\\", "/")
59	    name = name.rsplit("/", 1)[-1]
60	    # S4 — control / bidi characters out, Windows-illegal characters to '_'.
61	    name = _strip_controls(name)
62	    for ch in _WINDOWS_ILLEGAL:
63	        name = name.replace(ch, "_")
64	    # S5 — Windows silently drops trailing dots and spaces, so drop them ourselves.
65	    #      Must run BEFORE S6 or `CON.` never matches the reserved-name list.
66	    name = name.rstrip(" .").lstrip(" ")
67	    # S6 — device names.
68	    if Path(name).stem.upper() in WINDOWS_RESERVED_NAMES:
69	        name = "_" + name
70	    # S7 — nothing usable survived.
71	    if not name or name in (".", ".."):
72	        name = "upload"
73	    return name

76	def sanitize_path_component(raw: str) -> str:
77	    """S3~S7 only — used for the attachment room name (the document code)."""
78	    return _apply_s3_to_s7(raw or "")

81	def sanitize_attachment_name(raw: Optional[str]) -> str:
82	    """S1~S9 — the name a part is actually stored under."""
83	    # S1
84	    name = raw or ""
85	    if not name.strip():
86	        name = "upload"
87	    # S2 — macOS sends NFD Hangul; fold to NFC so the registry and the disk agree.
88	    name = unicodedata.normalize("NFC", name)
89	    # S3~S7
90	    name = _apply_s3_to_s7(name)
91	    # S8 — truncate LAST, on a character boundary, because earlier steps shorten the name.
92	    if len(name.encode("utf-8")) > ATTACH_MAX_FILENAME_BYTES:
93	        ext = Path(name).suffix
94	        if len(ext.encode("utf-8")) > 32:
95	            ext = _truncate_utf8(ext, 32)
96	        budget = ATTACH_MAX_FILENAME_BYTES - len(ext.encode("utf-8"))
97	        stem = _truncate_utf8(Path(name).stem, budget)
98	        name = stem + ext
99	        if not Path(name).stem.strip():
100	            name = "upload" + ext
101	    # S9
102	    return name

105	def original_display_name(raw: Optional[str]) -> str:
106	    """``original_filename`` — S2 only (P0011 §1-3 / L0012 §2-2 last paragraph).
107	
108	    The user's own name is shown back to them verbatim; only characters that would corrupt
109	    a log line or spoof an extension are removed.
110	    """
111	    name = raw or "upload"
112	    return _strip_controls(unicodedata.normalize("NFC", name)) or "upload"

117	def check_request_size(content_length: Optional[str]) -> None:
118	    """U1 — the cheap pre-check on the declared body size.
119	
120	    A missing or lying header is expected; U3 counts the real bytes as they arrive.
121	    """
122	    if not content_length:
123	        return
124	    try:
125	        declared = int(content_length)
126	    except (TypeError, ValueError):
127	        return
128	    if declared > ATTACH_MAX_REQUEST_BYTES:
129	        raise AttachmentError(
130	            413,
131	            "ATTACHMENT_TOO_LARGE",
132	            "Attachment request exceeds the total size limit.",
133	            limit_bytes=ATTACH_MAX_REQUEST_BYTES,
134	        )

137	def resolve_content_type(safe_name: str) -> str:
138	    """E1~E2 — the media type this attachment is stored under. Refuses nothing.
139	
140	    This was ``check_extension``, and it raised ``400 UNSUPPORTED_EXTENSION`` for every
141	    kind in §1-3. TR0017 rev3 was rejected for that wall, so the extension no longer decides
142	    whether a file is accepted — only how it is served (see ``constants`` §1-3).
143	
144	    The part's own ``Content-Type`` header is deliberately ignored either way: keeping it
145	    would let the uploader pick the media type the download response later serves.
146	    """
147	    ext = Path(safe_name).suffix.lstrip(".").lower()
148	    if not ext:
149	        return ATTACH_DEFAULT_CONTENT_TYPE
150	    if ext in ATTACH_EXECUTABLE_EXTENSIONS:
151	        # ``.js`` guesses as ``text/javascript``, ``.jar`` as ``application/java-archive``.
152	        # Writing that into the registry would mean every later reader of the row has to
153	        # remember to undo it; one row that nobody undid is one inline-rendered script.
154	        return ATTACH_DEFAULT_CONTENT_TYPE
155	    guessed, _ = mimetypes.guess_type(safe_name)
156	    return guessed or ATTACH_DEFAULT_CONTENT_TYPE

159	def is_executable_extension(filename: str) -> bool:
160	    """§1-3 — "a kind the receiver would click and run".
161	
162	    Answers a serving question, not an admission one: download forces these back to
163	    octet-stream even when the stored row says otherwise (a legacy-migrated row can, since
164	    it never went through :func:`resolve_content_type`).
165	    """
166	    return Path(filename).suffix.lstrip(".").lower() in ATTACH_EXECUTABLE_EXTENSIONS

174	def _exists_case_insensitive(room: Path, candidate: str) -> bool:
175	    """D3 — compare case-folded.
176	
177	    ``A.txt`` and ``a.txt`` are the same file on Windows and two files on Linux. Without
178	    this the identical request produces different results per host.
179	    """
180	    folded = candidate.casefold()
181	    try:
182	        return any(entry.name.casefold() == folded for entry in room.iterdir())
183	    except OSError:
184	        return False

187	def resolve_unique_name(
188	    room: Path,
189	    doc_id: str,
190	    safe_name: str,
191	    reserved_in_request: set,
192	    registry_has,
193	) -> tuple[str, int]:
194	    """D1~D6 — pick a free name and atomically reserve it. Returns (name, open fd).
195	
196	    The old rule appended ``_{epoch}`` exactly once, so two parts with the same name in one
197	    request both computed the same candidate in the same second and the second silently
198	    overwrote the first. D2 (in-request reservation) and D5 (``O_CREAT|O_EXCL``) close that;
199	    D5 is the only one that also closes the race between two concurrent requests, because an
200	    ``exists()`` followed by a ``write()`` always has a gap.
201	    """
202	    stem = Path(safe_name).stem
203	    suffix = Path(safe_name).suffix
204	    epoch = int(time.time())
205	
206	    for attempt in range(ATTACH_DEDUPE_MAX_ATTEMPTS + 1):
207	        if attempt == 0:
208	            candidate = safe_name
209	        elif attempt == 1:
210	            candidate = f"{stem}_{epoch}{suffix}"
211	        else:
212	            candidate = f"{stem}_{epoch}_{attempt}{suffix}"
213	
214	        if candidate in reserved_in_request:          # D2
215	            continue
216	        if _exists_case_insensitive(room, candidate):  # D3
217	            continue
218	        if registry_has(doc_id, candidate):            # D4
219	            continue
220	        try:                                           # D5 — the real reservation
221	            fd = os.open(
222	                str(room / candidate),
223	                os.O_CREAT | os.O_EXCL | os.O_WRONLY | _O_BINARY,
224	            )
225	        except FileExistsError:
226	            continue
227	        except OSError as exc:
228	            raise AttachmentError(
229	                500,
230	                "ATTACHMENT_STORE_FAILED",
231	                "Could not create the attachment file.",
232	                filename=safe_name,
233	            ) from exc
234	        reserved_in_request.add(candidate)
235	        return candidate, fd
236	
237	    # D6 — stop inventing names.
238	    raise AttachmentError(
239	        409,
240	        "ATTACHMENT_EXISTS",
241	        "An attachment with this name already exists.",
242	        doc_id=doc_id,
243	        filename=safe_name,
244	    )

249	def rfc5987_encode(name: str) -> str:
250	    """RFC 5987 UTF-8 percent-encoding — same rule as ``tree_routes._rfc5987_encode``."""
251	    return "UTF-8''" + quote(name, safe="")

254	def rfc5987_disposition(name: str) -> str:
255	    """``filename*`` always, plus the ASCII ``filename`` fallback when the name is ASCII.
256	
257	    NR0015 §5 measured that ``tree_routes`` has the encoder as a standalone function but
258	    leaves the fallback inline in ``download_file:400-407``. This is that pair lifted into
259	    one shared helper, per NR0015 §7 settled directives 4 and 5 — nothing imports the route.
260	    """
261	    encoded = rfc5987_encode(name)
262	    try:
263	        ascii_name = name.encode("ascii").decode("ascii")
264	    except UnicodeEncodeError:
265	        return f"attachment; filename*={encoded}"
266	    # A quote or backslash inside the ASCII fallback would end the quoted-string early.
267	    ascii_name = ascii_name.replace("\\", "_").replace('"', "_")
268	    return f'attachment; filename="{ascii_name}"; filename*={encoded}'

271	def has_forbidden_name_chars(name: str) -> bool:
272	    """J1 — the URL segment must be a bare filename, never a path."""
273	    if not name or name in (".", ".."):
274	        return True
275	    if "/" in name or "\\" in name:
276	        return True
277	    return any(ch in _CONTROL_CHARS for ch in name)
