1	"""numbering_jobs queue worker — single-process polling mode."""
2	from __future__ import annotations
3	
4	import threading
5	import time
6	import logging
7	from pathlib import Path
8	from typing import Optional
9	
10	from ..db import numbering_jobs as db_jobs
11	from .migration_service import process_job
12	
13	logger = logging.getLogger(__name__)
14	
15	
16	class NumberingWorker:
17	    """Worker that processes the numbering_jobs queue with a single thread.
18	
19	    Usage::
20	
21	        worker = NumberingWorker()
22	        worker.start()          # Start the background thread
23	        ...
24	        worker.stop()           # Stop the worker
25	    """
26	
27	    def __init__(
28	        self,
29	        poll_interval: float = 5.0,
30	        backup_dir: Optional[Path] = None,
31	    ) -> None:
32	        self._poll_interval = poll_interval
33	        self._backup_dir = backup_dir
34	        self._stop_event = threading.Event()
35	        self._thread: Optional[threading.Thread] = None
36	
37	    def start(self) -> None:
38	        """Start the worker thread."""
39	        if self._thread and self._thread.is_alive():
40	            return
41	        self._stop_event.clear()
42	        self._thread = threading.Thread(
43	            target=self._run, daemon=True, name="numbering-worker"
44	        )
45	        self._thread.start()
46	        logger.info("[NumberingWorker] started")
47	
48	    def stop(self, timeout: float = 10.0) -> None:
49	        """Stop the worker thread."""
50	        self._stop_event.set()
51	        if self._thread:
52	            self._thread.join(timeout=timeout)
53	        logger.info("[NumberingWorker] stopped")
54	
55	    def _run(self) -> None:
56	        while not self._stop_event.is_set():
57	            try:
58	                self._process_one_batch()
59	            except Exception as exc:
60	                logger.exception(f"[NumberingWorker] Exception: {exc}")
61	            self._stop_event.wait(self._poll_interval)
62	
63	    def _process_one_batch(self) -> None:
64	        """Fetch and process one job in queued status."""
65	        # Process the oldest queued job across all projects
66	        from ..db.connection import get_store
67	        row = get_store()._fetch_one(
68	            "SELECT id FROM numbering_jobs WHERE status='queued' ORDER BY created_at LIMIT 1"
69	        )
70	        if row is None:
71	            return
72	        job_id = row["id"]
73	        try:
74	            result = process_job(job_id, backup_dir=self._backup_dir)
75	            logger.info(
76	                f"[NumberingWorker] job={job_id} → {result.get('status')}"
77	            )
78	        except Exception as exc:
79	            logger.error(f"[NumberingWorker] job={job_id} processing error: {exc}")
80	
81	
82	# Singleton worker instance
83	_worker: Optional[NumberingWorker] = None
84	_worker_meta = threading.Lock()
85	
86	
87	def get_worker(poll_interval: float = 5.0) -> NumberingWorker:
88	    """Return the singleton worker instance."""
89	    global _worker
90	    with _worker_meta:
91	        if _worker is None:
92	            _worker = NumberingWorker(poll_interval=poll_interval)
93	        return _worker
94
