1	"""Move attachments left in the old side tree into the document's own room.
2	
3	flowgate.default.0060 L0012 §2-11 (M1~M8), DB0013 §3-4.
4	
5	The old location is ``{storage_root}/projects/{project_dir}/attachments/{doc_id}/`` and it
6	was written without a registry, so the ONLY clue about which document a file belongs to is
7	the directory name (D0010 §3-8).
8	
9	This is an operational procedure, not part of server start-up. DB0013 §3-4 gives two
10	reasons: the source cannot be expressed as ``INSERT ... SELECT`` (it is a directory listing),
11	and the copy is heavy I/O that would block boot. Run it from
12	``server/tools/migrate_legacy_attachments.py`` after 080 is applied.
13	"""
14	from __future__ import annotations
15	
16	import hashlib
17	import os
18	import shutil
19	from datetime import datetime, timezone
20	from pathlib import Path
21	
22	from modules.flow_gate.storage import paths as storage_paths
23	
24	from .locator import resolve_attach_dir, storage_relative
25	from .naming import resolve_content_type, resolve_unique_name, sanitize_attachment_name
26	from .registry import registry_has, registry_has_same, registry_insert
27	
28	MIGRATED_DIR_NAME = "_migrated"
29	
30	
31	def legacy_root(project_id: str) -> Path:
32	    """Where ``documents.py:2763`` used to put things."""
33	    return (
34	        storage_paths.get_storage_root(project_id)
35	        / "projects"
36	        / storage_paths.project_dir_name(project_id)
37	        / "attachments"
38	    )
39	
40	
41	def _sha256_of(path: Path) -> str:
42	    digest = hashlib.sha256()
43	    with open(path, "rb") as fh:
44	        for chunk in iter(lambda: fh.read(1048576), b""):
45	            digest.update(chunk)
46	    return digest.hexdigest()
47	
48	
49	def _mtime_rfc3339(path: Path) -> str:
50	    return (
51	        datetime.fromtimestamp(path.stat().st_mtime, tz=timezone.utc)
52	        .strftime("%Y-%m-%dT%H:%M:%SZ")
53	    )
54	
55	
56	def migrate_legacy_attachments(project_id: str, dry_run: bool = True) -> dict:
57	    """M1~M8. Returns ``{moved, skipped, unresolved}``.
58	
59	    Two rules that are not negotiable:
60	
61	    * A file whose document cannot be found (M2) is not touched — not deleted, not moved.
62	      The directory name is the only clue to its owner, and the document may still be
63	      restored later; erasing the clue loses the file for good.
64	    * The original is MOVED to ``_migrated/``, never deleted (M7). If anything about the
65	      migration turns out to be wrong there is still something to go back to. This product
66	      does not do destructive deletes.
67	
68	    Re-running is safe: M4 skips any file whose sha256 is already registered for that
69	    document. The extension deny-list is deliberately NOT applied — the goal is not to erase
70	    material that is already uploaded; ``read`` blocks those at §2-9 R3 instead.
71	    """
72	    from modules.flow_gate.documents import document_service
73	
74	    report: dict[str, list] = {"moved": [], "skipped": [], "unresolved": []}
75	    root = legacy_root(project_id)
76	    if not root.is_dir():
77	        return report
78	
79	    for directory in sorted(p for p in root.iterdir() if p.is_dir()):
80	        if directory.name.startswith("_"):        # M1 — _migrated and friends
81	            continue
82	        doc = document_service.get_document(directory.name)   # dir name == old doc_id
83	        if doc is None:                                        # M2
84	            report["unresolved"].append(directory.name)
85	            continue
86	
87	        doc_id = doc.get("doc_id")
88	        room = resolve_attach_dir(doc)                         # M3
89	        if not dry_run:
90	            room.mkdir(parents=True, exist_ok=True)
91	
92	        reserved: set = set()
93	        for entry in sorted(p for p in directory.iterdir() if p.is_file()):
94	            safe = sanitize_attachment_name(entry.name)        # §2-2 re-applied
95	            sha = _sha256_of(entry)
96	            if registry_has_same(doc_id, sha):                 # M4 — idempotent
97	                report["skipped"].append({"doc_id": doc_id, "filename": entry.name})
98	                continue
99	            if dry_run:
100	                report["moved"].append({
101	                    "doc_id": doc_id, "filename": entry.name, "planned_name": safe,
102	                })
103	                continue
104	
105	            final_name, fd = resolve_unique_name(              # M5
106	                room, doc_id, safe, reserved, registry_has
107	            )
108	            dest = room / final_name
109	            try:
110	                with open(entry, "rb") as src:
111	                    while True:
112	                        chunk = src.read(1048576)
113	                        if not chunk:
114	                            break
115	                        os.write(fd, chunk)
116	                os.fsync(fd)
117	            finally:
118	                try:
119	                    os.close(fd)
120	                except OSError:
121	                    pass
122	
123	            # A legacy executable keeps its row and reads back like any other binary; the
124	            # extension only decides that it is stored and served as octet-stream (§1-3).
125	            content_type = resolve_content_type(final_name)
126	
127	            registry_insert(                                   # M6
128	                doc_id=doc_id,
129	                original_filename=entry.name,
130	                filename=final_name,
131	                file_path=storage_relative(dest, doc.get("project_id")),
132	                size=dest.stat().st_size,
133	                content_type=content_type,
134	                content_sha256=sha,
135	                uploaded_by=None,                              # no record of who; do not invent one
136	                uploaded_at=_mtime_rfc3339(entry),             # mtime, an estimate
137	                is_legacy_migrated=True,
138	            )
139	
140	            keep = root / MIGRATED_DIR_NAME / directory.name   # M7 — preserve the original
141	            keep.mkdir(parents=True, exist_ok=True)
142	            shutil.move(str(entry), str(keep / entry.name))
143	            report["moved"].append({
144	                "doc_id": doc_id, "filename": entry.name, "stored_as": final_name,
145	            })
146	
147	    return report                                              # M8
