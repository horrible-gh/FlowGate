17	def _sanitize_part(s: str) -> str:
18	    """Normalize a filesystem path segment (only Korean/Latin/digits/underscore/hyphen allowed)."""
19	    return re.sub(r"[^\w\-]", "_", s, flags=re.UNICODE) or "_"

22	def default_storage_root() -> Path:
23	    """Return the default storage root relative to the server's working directory."""
24	    return Path.cwd() / "storage"

27	def _project_override_root(project_id: Optional[str]) -> Optional[Path]:
28	    """Read the storage_root_override value from project settings.
29	
30	    Returns None on failure. Uses lazy import to avoid circular imports with
31	    config/db modules.
32	    """
33	    if not project_id:
34	        return None
35	    try:
36	        from modules.flow_gate.db import projects as _proj  # lazy
37	    except Exception:
38	        return None
39	    try:
40	        settings = _proj.get_settings(project_id)
41	    except Exception:
42	        return None
43	    if not settings:
44	        return None
45	    override = (settings.get("storage_root_override") or "").strip()
46	    return Path(override) if override else None

49	def _system_storage_root() -> Optional[Path]:
50	    """Read the system_settings.storage_root value.
51	
52	    Returns None on failure. Uses lazy import to avoid circular imports.
53	    """
54	    try:
55	        from modules.flow_gate.db import system_settings as _sys  # lazy
56	    except Exception:
57	        return None
58	    try:
59	        value = _sys.get_value("storage_root")
60	    except Exception:
61	        return None
62	    if not value:
63	        return None
64	    value = value.strip()
65	    return Path(value) if value else None

68	def get_storage_root(
69	    project_id: Optional[str] = None,
70	    create: bool = False,
71	) -> Path:
72	    """Return the storage root path by priority.
73	
74	    ① FLOWGATE_STORAGE_DIR environment variable
75	    ② project_settings.storage_root_override (when project_id is given)
76	    ③ system_settings.storage_root
77	    ④ default_storage_root()
78	    """
79	    env = os.environ.get("FLOWGATE_STORAGE_DIR", "").strip()
80	    if env:
81	        root = Path(env)
82	    else:
83	        override = _project_override_root(project_id)
84	        if override:
85	            root = override
86	        else:
87	            system_root = _system_storage_root()
88	            root = system_root if system_root else default_storage_root()
89	    if create:
90	        root.mkdir(parents=True, exist_ok=True)
91	    return root

94	def _slugify_project_name(name: str) -> str:
95	    """Convert a project name to a directory-safe slug, preserving Korean/Unicode."""
96	    slug = name.strip()
97	    slug = re.sub(r"\s+", "_", slug)
98	    slug = re.sub(r"[^\w\-]", "", slug, flags=re.UNICODE)
99	    slug = slug.strip("_-")
100	    return slug

104	    """Project directory name = sanitized project_id."""
105	    if not project_id:
106	        return project_id
107	    return _sanitize_part(project_id)

118	def _group_worktree_on_disk(
119	    project_id: Optional[str], group_id: Optional[str]
120	) -> Optional[Path]:
121	    """The group's branch worktree path when it physically exists, else None.
122	
123	    0284 T0005 (structural fix for B0001): recovery used by
124	    resolve_project_src_root / classify_src_root when the git ledger's
125	    worktree_registered flag has been cleared — merge/push cleanup resets it to
126	    0 (git_integration.unregister_worktree), which is what silently dropped every
127	    post-merge / in-flight re-run to the base(main) tree (NR0003 §4). The flag gates
128	    slot *accounting*, not which tree the runner must read: while the group's branch
129	    worktree is still on disk it stays authoritative, so a fix-verification suite is
130	    checked against the tree that actually holds the fix. Mirrors the gates of
131	    git_service.effective_src_root_ex (integration on, state, branch, project
132	    name, directory present) EXCEPT the registration flag. Pure lookup — never
133	    raises; a genuinely pruned directory returns None so the caller still falls back.
134	
135	    0287 NR0004 §5: "present" additionally requires the worktree's `.git` link. A
136	    teardown interrupted mid-delete leaves the directory standing with its link and
137	    most of its files gone, and that corpse must not be recovered as authoritative.
138	    """
139	    if not project_id or not group_id:
140	        return None
141	    try:
142	        from modules.flow_gate.services import git_service  # lazy — import cycle
143	
144	        cfg = git_service.db_git.get_config(project_id)
145	        if cfg is None or not cfg.get("enabled"):
146	            return None
147	        state = git_service.db_git.get_state(group_id)
148	        if not state:
149	            return None
150	        branch = (state.get("branch") or "").strip()
151	        if not branch:
152	            return None
153	        project_name = git_service._project_name(project_id)
154	        if not project_name:
155	            return None
156	        wt = src_root(project_name, branch)
157	        # 0287 NR0004 §5: "still on disk" has to mean a real tree, not a directory
158	        # that happens to exist. An interrupted `worktree remove` leaves the path
159	        # in place minus its `.git` link and most of its content, and recovering
160	        # ONTO that corpse is worse than the fallback this guard was written to
161	        # avoid — the suite runs against a tree missing the modules under test.
162	        if wt.is_dir() and git_service._worktree_link_ok(wt):
163	            return wt.resolve()
164	    except Exception:
165	        _log.warning(
166	            "on-disk worktree recovery failed for group %s", group_id, exc_info=True
167	        )
168	    return None

171	def resolve_project_src_root(
172	    project_id: Optional[str],
173	    fallback_branch: str = "main",
174	    group_id: Optional[str] = None,
175	) -> Optional[Path]:
176	    """Resolve a project's source mirror root from its *project_id*.
177	
178	    ``src_root()`` takes the human-facing project_name, so callers holding only
179	    a project_id must translate through the projects table and take the branch
180	    from project settings (falling back to *fallback_branch*). Passing a
181	    project_id straight into ``src_root()`` resolves a nonexistent directory
182	    whenever project_id != project_name (the 0152 test-runner outage).
183	    Returns None when the project row or its name is missing.
184	
185	    0115: when *group_id* is given and the project is git-integrated with a
186	    registered worktree for that group, the group worktree path wins (L0006
187	    §2.2). Every other case — no group, no config, disabled, missing worktree —
188	    falls back to the ordinary project-branch folder below (fallback first).
189	    """
190	    if not project_id:
191	        return None
192	    if group_id:
193	        try:
194	            from modules.flow_gate.services import git_service  # lazy — import cycle
195	
196	            wt = git_service.effective_src_root(project_id, group_id)
197	            if wt is not None:
198	                return wt
199	        except Exception:
200	            # 0280 NR0003 §4-B: worktree resolution must never break the fallback
201	            # path — but swallowing it silently made a git-integrated group run in
202	            # base(main) with no trace anywhere. Fall back loudly.
203	            _log.warning(
204	                "worktree resolution failed for group %s (project %s) — falling back "
205	                "to the base project-branch tree",
206	                group_id,
207	                project_id,
208	                exc_info=True,
209	            )
210	        # 0284 T0005 (structural fix for B0001 / NR0003 §6-1): effective_src_root()
211	        # returns None once the ledger's worktree_registered flag is cleared (merge/
212	        # push cleanup), which used to drop the run to the base(main) tree even while
213	        # the group's branch worktree was still on disk — so a fix-verification suite
214	        # ran against a tree lacking the very fix it verifies. Recover the on-disk
215	        # branch worktree before falling back; a pruned directory — or one left
216	        # broken by an interrupted teardown (0287 NR0004) — falls through.
217	        recovered = _group_worktree_on_disk(project_id, group_id)
218	        if recovered is not None:
219	            _log.info(
220	                "resolve_project_src_root: group %s using on-disk worktree %s despite "
221	                "cleared registration ledger (post-merge recovery)",
222	                group_id,
223	                recovered,
224	            )
225	            return recovered
226	    try:
227	        from modules.flow_gate.db import projects as _proj  # lazy — import cycle
228	    except Exception:
229	        return None
230	    try:
231	        row = _proj.get_by_id(project_id)
232	    except Exception:
233	        return None
234	    project_name = (row.get("project_name") or "").strip() if row else ""
235	    if not project_name:
236	        return None
237	    branch = (fallback_branch or "main").strip() or "main"
238	    try:
239	        settings = _proj.get_settings(project_id)
240	    except Exception:
241	        settings = None
242	    if settings:
243	        configured = (settings.get("branch") or "").strip()
244	        if configured:
245	            branch = configured
246	    return src_root(project_name, branch).resolve()

249	def classify_src_root(
250	    project_id: Optional[str],
251	    group_id: Optional[str],
252	    root: Optional[Path],
253	) -> str:
254	    """Classify an *already resolved* src root into a ``SRC_ROOT_*`` kind.
255	
256	    0280 NR0003 §6-2: callers persist and display the root they actually used, so
257	    a "the tests ran in main" report becomes checkable instead of arguable. Kind is
258	    ``git_service.SRC_ROOT_WORKTREE`` when *root* is the group's worktree, else the
259	    ``SRC_ROOT_*`` reason the worktree was skipped.
260	
261	    Deliberately classifies the root it is *given* rather than re-deriving it: when
262	    the two disagree (a test double, a caller that resolved elsewhere) the answer is
263	    ``"unknown"``, never a confident lie. Never raises — bookkeeping must not be able
264	    to fail a run.
265	    """
266	    if root is None:
267	        return "unknown"
268	    try:
269	        from modules.flow_gate.services import git_service  # lazy — import cycle
270	
271	        wt, reason = git_service.effective_src_root_ex(project_id, group_id)
272	        if wt is not None:
273	            return reason if Path(root).resolve(strict=False) == wt else "unknown"
274	        # 0284 T0005: effective_src_root_ex still reports the ledger fallback reason
275	        # (e.g. worktree_unregistered after merge/cleanup) even though the run may have
276	        # executed in the group's on-disk branch worktree via the recovery above.
277	        # Classify by the root ACTUALLY used so the TSR reads "worktree", not the
278	        # stale ledger reason.
279	        recovered = _group_worktree_on_disk(project_id, group_id)
280	        if recovered is not None and Path(root).resolve(strict=False) == recovered:
281	            return git_service.SRC_ROOT_WORKTREE
282	        return reason
283	    except Exception:
284	        _log.warning(
285	            "src_root classification failed for group %s", group_id, exc_info=True
286	        )
287	        return "unknown"

318	def group_dir_name(group_id: str) -> str:
319	    """Group directory name = sanitized last segment (group_num) of group_id."""
320	    if not group_id:
321	        return group_id
322	    if "." in group_id:
323	        return _sanitize_part(group_id.rsplit(".", 1)[-1])
324	    if "-" in group_id:
325	        return _sanitize_part(group_id.rsplit("-", 1)[-1])
326	    return _sanitize_part(group_id)

360	def document_path(
361	    project_id: str,
362	    group_code: str,
363	    doc_code: str,
364	    filename: str,
365	    subgroup_code: Optional[str] = None,
366	    storage_root: Optional[Path] = None,
367	    module: str = "none",
368	    *,
369	    branch: str = "main",
370	) -> Path:
371	    """Document file path.
372	
373	    With subgroup:    {project_root}/{module}/{group_code}/{subgroup_code}/{doc_code}_{filename}
374	    Without subgroup: {project_root}/{module}/{group_code}/{doc_code}_{filename}
375	    """
376	    if subgroup_code:
377	        base = subgroup_path(project_id, group_code, subgroup_code, storage_root, module, branch=branch)

380	    return base / f"{doc_code}_{filename}"

383	# ── Storage-relative path persistence (B0054.0001 / L0054.0002) ──────────────
384	# Single source of truth: writers persist storage-root-relative POSIX paths,
385	# readers resolve them back through resolve_storage_path(). This makes path
386	# columns invariant across host/OS migration (the B0001 root cause: Windows
387	# absolute paths like ``E:\...`` were stored verbatim and broke on Linux).
388	# These two functions absorb the three legacy resolve variants
389	# (documents._document_file_path, process_service._resolve_storage_path,
390	# document_routes._fallback_file_path).
391	
392	_DRIVE_RE = re.compile(r"^[A-Za-z]:")
393	
394	
395	def _is_relative_to(path: Path, base: Path) -> bool:
396	    try:
397	        path.relative_to(base)
398	        return True
399	    except ValueError:
400	        return False

403	def _allowed_roots(project_id: Optional[str] = None) -> list[Path]:
404	    """Filesystem jail: storage root plus the special storage subtrees.
405	
406	    Lazy-imports db to avoid a circular import (db/__init__ imports this module
407	    at load time). Mirrors the allow-list previously inlined in
408	    documents._document_file_path.
409	    """
410	    roots = [get_storage_root(project_id).resolve()]
411	    try:
412	        from modules.flow_gate import db  # lazy — avoid import cycle
413	        for attr in (
414	            "STORAGE_DIR",
415	            "TEST_REPORTS_DIR",
416	            "TEST_REPORTS_ARCHIVE_DIR",
417	            "DESIGN_REOPEN_DIR",
418	        ):
419	            val = getattr(db, attr, None)
420	            if val:
421	                roots.append(Path(val).resolve())
422	    except Exception:
423	        pass
424	    return roots

427	def _within_allowed_roots(path: Path, project_id: Optional[str] = None) -> bool:
428	    resolved = path.resolve(strict=False)
429	    return any(
430	        resolved == root or _is_relative_to(resolved, root)
431	        for root in _allowed_roots(project_id)
432	    )

435	def within_allowed_roots(path: Path, project_id: Optional[str] = None) -> bool:
436	    """Public wrapper over ``_within_allowed_roots`` — the storage filesystem jail.
437	
438	    flowgate.default.0060 NR0015 §7 settled directive 5: the predicate itself is private, and the
439	    document-attachment paths (L0012 §2-1, §2-6 J4) need exactly this judgement. Importing a
440	    private name across modules would make the reuse boundary invisible, so the boundary is
441	    named here instead. Same behaviour, no second copy of the rule.
442	    """
443	    return _within_allowed_roots(path, project_id)

446	def to_storage_relative(abs_path, project_id: Optional[str] = None) -> str:
447	    """Write helper → storage-root-relative POSIX path for DB persistence.
448	
449	    Idempotent: a value that is already relative is returned normalized (so this
450	    can never double-relativize a path that was loaded from the DB). Absolute
451	    paths outside the storage root are preserved as POSIX with a warning — this
452	    should not happen for managed files.
453	    """
454	    raw = str(abs_path).replace("\\", "/")
455	    p_in = Path(abs_path)
456	    # Already storage-relative (no leading slash, no drive letter) → no-op.
457	    if not p_in.is_absolute() and not _DRIVE_RE.match(raw):
458	        return Path(raw).as_posix()
459	    root = get_storage_root(project_id).resolve()
460	    p = p_in.resolve(strict=False)
461	    try:
462	        return p.relative_to(root).as_posix()
463	    except ValueError:
464	        _log.warning("path outside storage root: %s (root=%s)", p, root)
465	        return p.as_posix()

468	def _apply_branch_segment(stored: str, branch: str) -> Optional[str]:
469	    """Insert or remove the branch segment after ``/documents/<project>/``.
470	
471	    Absorbs document_routes._fallback_file_path: handles paths that predate (or
472	    double-count) the branch directory level.
473	    """
474	    if not stored or not branch:
475	        return None
476	    norm = stored.replace("\\", "/")
477	    # Match an optional prefix, the documents/<project>/ segment, then the rest.
478	    # Works for both canonical relative ("documents/<proj>/...") and legacy
479	    # absolute ("/home/.../documents/<proj>/...") forms.
480	    m = re.search(r"(.*?)(documents/[^/]+/)(.*)", norm)
481	    if not m:
482	        return None
483	    prefix, doc_seg, after_project = m.group(1), m.group(2), m.group(3)
484	    if not after_project:
485	        return None
486	    first_seg = after_project.split("/")[0]
487	    if first_seg == branch:
488	        rest = "/".join(after_project.split("/")[1:])
489	        return prefix + doc_seg + rest
490	    return prefix + doc_seg + branch + "/" + after_project

493	def resolve_storage_path(
494	    stored: str,
495	    project_id: Optional[str] = None,
496	    branch: str = "main",
497	) -> Optional[Path]:
498	    """Read helper → resolve a stored file path to a real, jailed Path, or None.
499	
500	    Handles, in order:
501	      ① legacy ``/storage/...`` prefix (old process_service form),
502	      ② legacy absolute path (POSIX or ``C:`` Windows drive),
503	      ③ canonical storage-relative path (the new standard),
504	      ④ branch-segment drift (absorbs _fallback_file_path).
505	    Returns None when nothing resolves to a file inside the allowed roots.
506	    """
507	    if not stored:
508	        return None
509	    s = stored.strip().replace("\\", "/")
510	    root = get_storage_root(project_id).resolve()
511	
512	    if s.startswith("/storage/"):
513	        cand = root / s[len("/storage/"):]
514	    elif Path(s).is_absolute() or _DRIVE_RE.match(s):
515	        cand = Path(s)
516	    else:
517	        cand = root / s
518	
519	    cand = cand.resolve(strict=False)
520	    if _within_allowed_roots(cand, project_id) and cand.is_file():
521	        return cand
522	
523	    fb = _apply_branch_segment(s, branch)
524	    if fb:
525	        fcand = (
526	            Path(fb) if (Path(fb).is_absolute() or _DRIVE_RE.match(fb)) else root / fb
527	        ).resolve(strict=False)
528	        if _within_allowed_roots(fcand, project_id) and fcand.is_file():
529	            return fcand
530	
531	    return None

534	def resolve_storage_dir(stored: str, project_id: Optional[str] = None) -> Optional[Path]:
535	    """Resolve a stored *directory* path (e.g. tokens.scratch_dir) to an absolute Path.
536	
537	    Directory analogue of resolve_storage_path: no ``is_file`` check and it
538	    tolerates a not-yet-created directory. Relative values join the storage root;
539	    legacy absolute values pass through. Used so consumers that need a usable OS
540	    path (the doc_path security jail, worker scratch staging) keep working after
541	    the column is stored relative.
542	    """
543	    if not stored:
544	        return None
545	    s = stored.strip().replace("\\", "/")
546	    if Path(s).is_absolute() or _DRIVE_RE.match(s):
547	        return Path(s).resolve(strict=False)
548	    root = get_storage_root(project_id).resolve()
549	    return (root / s).resolve(strict=False)
